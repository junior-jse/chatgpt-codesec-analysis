```c
#include <stdio.h>
#include <stdlib.h>

int countRotations(int arr[], int n) {
    if (n <= 1) {
        return 0;
    }
    
    int min_index = 0;
    for (int i = 1; i < n; i++) {
        if (arr[i] < arr[min_index]) {
            min_index = i;
        }
    }
    
    return min_index;
}

int main(void) {
    int arr[] = {15, 18, 2, 3, 6, 12};
    int n = sizeof(arr) / sizeof(arr[0]);
    
    int rotations = countRotations(arr, n);
    printf("Number of rotations: %d\n", rotations);
    
    return 0;
}
```
