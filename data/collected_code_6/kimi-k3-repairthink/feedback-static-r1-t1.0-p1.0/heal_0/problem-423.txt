#include <stdio.h>
#include <stdlib.h>

int max(int a, int b) {
    return (a > b) ? a : b;
}

int getMaxGold(int **gold, int m, int n) {
    if (gold == NULL || m <= 0 || n <= 0) {
        return -1;
    }

    int **goldTable = (int **)malloc(m * sizeof(int *));
    if (goldTable == NULL) {
        return -1;
    }

    for (int i = 0; i < m; i++) {
        goldTable[i] = (int *)malloc(n * sizeof(int));
        if (goldTable[i] == NULL) {
            for (int j = 0; j < i; j++) {
                free(goldTable[j]);
            }
            free(goldTable);
            return -1;
        }
    }

    for (int col = n - 1; col >= 0; col--) {
        for (int row = 0; row < m; row++) {
            int right = (col == n - 1) ? 0 : goldTable[row][col + 1];
            int right_up = (row == 0 || col == n - 1) ? 0 : goldTable[row - 1][col + 1];
            int right_down = (row == m - 1 || col == n - 1) ? 0 : goldTable[row + 1][col + 1];

            goldTable[row][col] = gold[row][col] + max(right, max(right_up, right_down));
        }
    }

    int res = goldTable[0][0];
    for (int i = 1; i < m; i++) {
        res = max(res, goldTable[i][0]);
    }

    for (int i = 0; i < m; i++) {
        free(goldTable[i]);
    }
    free(goldTable);

    return res;
}

int main() {
    int m = 4, n = 4;

    int **gold = (int **)malloc(m * sizeof(int *));
    if (gold == NULL) {
        return 1;
    }

    for (int i = 0; i < m; i++) {
        gold[i] = (int *)malloc(n * sizeof(int));
        if (gold[i] == NULL) {
            for (int j = 0; j < i; j++) {
                free(gold[j]);
            }
            free(gold);
            return 1;
        }
    }

    int values[4][4] = {
        {1, 3, 1, 5},
        {2, 2, 4, 1},
        {5, 0, 2, 3},
        {0, 6, 1, 2}
    };

    for (int i = 0; i < m; i++) {
        for (int j = 0; j < n; j++) {
            gold[i][j] = values[i][j];
        }
    }

    int result = getMaxGold(gold, m, n);
    if (result != -1) {
        printf("Maximum gold collected: %d\n", result);
    } else {
        printf("Error occurred while calculating maximum gold.\n");
    }

    for (int i = 0; i < m; i++) {
        free(gold[i]);
    }
    free(gold);

    return 0;
}
