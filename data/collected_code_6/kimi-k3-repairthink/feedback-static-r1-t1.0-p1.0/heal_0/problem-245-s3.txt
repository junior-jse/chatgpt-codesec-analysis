#include <stdio.h>
#include <stdlib.h>

int maxSumBitonicSubsequence(const int arr[], int n) {
    if (arr == NULL || n <= 0) {
        return 0;
    }

    int *msis = (int *)malloc((size_t)n * sizeof(int));
    int *msds = (int *)malloc((size_t)n * sizeof(int));
    
    if (msis == NULL || msds == NULL) {
        free(msis);
        free(msds);
        return 0;
    }

    for (int i = 0; i < n; i++) {
        msis[i] = arr[i];
        msds[i] = arr[i];
    }

    for (int i = 1; i < n; i++) {
        for (int j = 0; j < i; j++) {
            if (arr[i] > arr[j] && msis[i] < msis[j] + arr[i]) {
                msis[i] = msis[j] + arr[i];
            }
        }
    }

    for (int i = n - 2; i >= 0; i--) {
        for (int j = n - 1; j > i; j--) {
            if (arr[i] > arr[j] && msds[i] < msds[j] + arr[i]) {
                msds[i] = msds[j] + arr[i];
            }
        }
    }

    int maxSum = msis[0] + msds[0] - arr[0];
    for (int i = 1; i < n; i++) {
        int currentSum = msis[i] + msds[i] - arr[i];
        if (currentSum > maxSum) {
            maxSum = currentSum;
        }
    }

    free(msis);
    free(msds);

    return maxSum;
}

int main(void) {
    int arr[] = {1, 15, 51, 45, 33, 100, 12, 18, 9};
    int n = (int)(sizeof(arr) / sizeof(arr[0]));

    int result = maxSumBitonicSubsequence(arr, n);
    printf("Maximum sum of bi-tonic sub-sequence: %d\n", result);

    return 0;
}
