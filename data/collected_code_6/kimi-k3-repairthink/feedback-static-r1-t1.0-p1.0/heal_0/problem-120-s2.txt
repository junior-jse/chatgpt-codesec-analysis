#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

typedef struct {
    int first;
    int second;
} Tuple;

int find_max_product(const Tuple *tuples, size_t count, int *result) {
    if (tuples == NULL || result == NULL || count < 2) {
        return 0;
    }

    long long max_product = LLONG_MIN;
    int found = 0;

    for (size_t i = 0; i < count; i++) {
        for (size_t j = i + 1; j < count; j++) {
            long long product = (long long)tuples[i].first * tuples[i].second;
            long long other_product = (long long)tuples[j].first * tuples[j].second;
            long long pair_product = product * other_product;
            
            if (pair_product > max_product) {
                max_product = pair_product;
                found = 1;
            }
        }
    }

    if (found && max_product <= INT_MAX && max_product >= INT_MIN) {
        *result = (int)max_product;
        return 1;
    }
    return 0;
}

int main(void) {
    Tuple tuples[] = {{2, 3}, {4, 5}, {1, 2}, {6, 7}};
    size_t count = sizeof(tuples) / sizeof(tuples[0]);
    int result;

    if (find_max_product(tuples, count, &result)) {
        printf("Maximum product: %d\n", result);
    } else {
        printf("Unable to compute maximum product.\n");
    }

    return 0;
}
