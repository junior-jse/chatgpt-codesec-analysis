#include <stdio.h>
#include <stdlib.h>

void rotateLeft(int* arr, int n, int k) {
    if (arr == NULL || n <= 0 || k < 0) {
        return;
    }
    
    k = k % n;
    if (k == 0) {
        return;
    }
    
    int* temp = (int*)malloc(k * sizeof(int));
    if (temp == NULL) {
        return;
    }
    
    for (int i = 0; i < k; i++) {
        temp[i] = arr[i];
    }
    
    for (int i = 0; i < n - k; i++) {
        arr[i] = arr[i + k];
    }
    
    for (int i = 0; i < k; i++) {
        arr[n - k + i] = temp[i];
    }
    
    free(temp);
    temp = NULL;
}

int main(void) {
    int n = 0;
    int k = 0;
    
    if (printf("Enter number of elements: ") < 0) {
        return EXIT_FAILURE;
    }
    
    if (scanf("%d", &n) != 1 || n <= 0) {
        fprintf(stderr, "Invalid input for number of elements\n");
        return EXIT_FAILURE;
    }
    
    int* arr = (int*)malloc(n * sizeof(int));
    if (arr == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    
    if (printf("Enter %d elements: ", n) < 0) {
        free(arr);
        return EXIT_FAILURE;
    }
    
    for (int i = 0; i < n; i++) {
        if (scanf("%d", &arr[i]) != 1) {
            fprintf(stderr, "Invalid input for element %d\n", i);
            free(arr);
            return EXIT_FAILURE;
        }
    }
    
    if (printf("Enter number of positions to rotate left: ") < 0) {
        free(arr);
        return EXIT_FAILURE;
    }
    
    if (scanf("%d", &k) != 1 || k < 0) {
        fprintf(stderr, "Invalid input for rotation count\n");
        free(arr);
        return EXIT_FAILURE;
    }
    
    rotateLeft(arr, n, k);
    
    if (printf("Rotated array: ") < 0) {
        free(arr);
        return EXIT_FAILURE;
    }
    
    for (int i = 0; i < n; i++) {
        if (printf("%d ", arr[i]) < 0) {
            free(arr);
            return EXIT_FAILURE;
        }
    }
    
    if (printf("\n") < 0) {
        free(arr);
        return EXIT_FAILURE;
    }
    
    free(arr);
    arr = NULL;
    
    return EXIT_SUCCESS;
}
