#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>
#include <stdint.h>

#define MAX_INPUT_LENGTH 4096

char *replace_with_colon(const char *input) {
    regex_t regex;
    regmatch_t match;
    const char *pattern = "[ ,.]";
    char *result;
    char *safe_input;
    size_t input_len;
    size_t result_len = 0;
    size_t i = 0;
    size_t j;
    int ret;

    if (input == NULL) {
        return NULL;
    }

    input_len = strnlen(input, MAX_INPUT_LENGTH);

    if (input_len >= SIZE_MAX) {
        return NULL;
    }

    safe_input = malloc(input_len + 1);
    if (safe_input == NULL) {
        return NULL;
    }

    for (j = 0; j < input_len; j++) {
        safe_input[j] = input[j];
    }
    safe_input[input_len] = '\0';

    result = malloc(input_len + 1);
    if (result == NULL) {
        free(safe_input);
        return NULL;
    }

    ret = regcomp(&regex, pattern, REG_EXTENDED);
    if (ret != 0) {
        free(safe_input);
        free(result);
        return NULL;
    }

    while (i < input_len) {
        ret = regexec(&regex, safe_input + i, 1, &match, 0);
        if (ret == 0 && match.rm_so == 0 && match.rm_eo > 0) {
            result[result_len++] = ':';
            i += (size_t)match.rm_eo;
        } else {
            result[result_len++] = safe_input[i++];
        }
    }

    result[result_len] = '\0';
    regfree(&regex);
    free(safe_input);

    return result;
}

int main(void) {
    const char *test_string = "Hello, world. This is a test";
    char *modified;

    modified = replace_with_colon(test_string);
    if (modified == NULL) {
        fprintf(stderr, "Error: Failed to process string\n");
        return EXIT_FAILURE;
    }

    printf("Original: %s\n", test_string);
    printf("Modified: %s\n", modified);

    free(modified);
    return EXIT_SUCCESS;
}
