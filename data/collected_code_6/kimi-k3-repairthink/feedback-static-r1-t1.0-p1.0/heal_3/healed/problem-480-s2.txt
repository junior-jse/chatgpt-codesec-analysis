```c
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <limits.h>
#include <ctype.h>

#define INITIAL_CAPACITY 128
#define MAX_INPUT_LENGTH ((size_t)16 * 1024 * 1024)

char findMaxOccurringChar(const char *str, int *maxCount)
{
    int freq[256] = {0};
    size_t i;
    int max = 0;
    char result = '\0';

    if (str == NULL || maxCount == NULL) {
        if (maxCount != NULL) {
            *maxCount = 0;
        }
        return '\0';
    }

    for (i = 0; str[i] != '\0'; i++) {
        unsigned char c = (unsigned char)str[i];
        if (freq[c] < INT_MAX) {
            freq[c]++;
        }
    }

    for (i = 0; i < 256; i++) {
        if (freq[i] > max) {
            max = freq[i];
            result = (char)i;
        }
    }

    *maxCount = max;
    return result;
}

int main(void)
{
    size_t capacity = INITIAL_CAPACITY;
    size_t length = 0;
    char *buffer;
    int maxCount = 0;
    char maxChar;
    int ch;

    buffer = malloc(capacity);
    if (buffer == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    printf("Enter a string: ");
    fflush(stdout);

    while ((ch = getchar()) != EOF && ch != '\n') {
        if (length + 1 >= capacity) {
            size_t newCapacity;
            char *resized;

            if (capacity > SIZE_MAX / 2 || capacity >= MAX_INPUT_LENGTH) {
                fprintf(stderr, "Input too large\n");
                free(buffer);
                return EXIT_FAILURE;
            }

            newCapacity = capacity * 2;
            if (newCapacity > MAX_INPUT_LENGTH) {
                newCapacity = MAX_INPUT_LENGTH;
            }

            resized = realloc(buffer, newCapacity);
            if (resized == NULL) {
                fprintf(stderr, "Memory allocation failed\n");
                free(buffer);
                return EXIT_FAILURE;
            }

            buffer = resized;
            capacity = newCapacity;
        }

        buffer[length] = (char)ch;
        length++;
    }

    buffer[length] = '\0';

    if (ferror(stdin)) {
        fprintf(stderr, "Error reading input\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    if (length == 0) {
        printf("Empty string provided\n");
        free(buffer);
        return EXIT_SUCCESS;
    }

    maxChar = findMaxOccurringChar(buffer, &maxCount);

    if (maxCount > 0) {
        if (isprint((unsigned char)maxChar)) {
            printf("Maximum occurring character: '%c' (appears %d times)\n",
                   maxChar, maxCount);
        } else {
            printf("Maximum occurring character: '\\x%02X' (appears %d times)\n",
                   (unsigned char)maxChar, maxCount);
        }
    } else {
        printf("No characters found\n");
    }

    free(buffer);
    return EXIT_SUCCESS;
}
```

Key fixes applied to address the flagged weaknesses (CWE-120, CWE-20):

1. **Buffer boundary enforcement**: The `length + 1 >= capacity` check runs before every write, guaranteeing space for both the new byte and the terminating `'\0'`, so the `getchar` loop can never write out of bounds.
2. **Input validation (CWE-20)**: A hard `MAX_INPUT_LENGTH` cap was added so unbounded input cannot force endless reallocation/memory exhaustion.
3. **Integer overflow protection**: `capacity > SIZE_MAX / 2` is verified before doubling, and the new capacity is clamped to the maximum allowed size.
4. **Safe `realloc` usage**: The result is stored in a temporary pointer so the original buffer is not leaked on failure, and the old buffer is freed on every error path.
5. **Stream error handling**: `ferror(stdin)` is checked after the loop to detect read errors distinct from `EOF`.
6. **Signedness safety**: Characters are cast to `unsigned char` before being used as array indices or passed to `isprint`, avoiding negative-index undefined behavior.
