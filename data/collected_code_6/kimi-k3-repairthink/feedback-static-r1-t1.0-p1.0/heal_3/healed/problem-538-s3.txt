#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STRING_LENGTH 4096

typedef struct {
    char **items;
    size_t count;
} Tuple;

static int safe_string_copy(char *dest, size_t dest_size, const char *src) {
    size_t i;
    size_t src_len;

    if (dest == NULL || src == NULL || dest_size == 0) {
        return -1;
    }

    src_len = strnlen(src, dest_size);
    if (src_len >= dest_size) {
        return -1;
    }

    for (i = 0; i < src_len; i++) {
        dest[i] = src[i];
    }
    dest[src_len] = '\0';

    return 0;
}

Tuple *string_list_to_tuple(char **list, size_t count) {
    Tuple *tuple;
    size_t i;

    if (list == NULL || count == 0) {
        return NULL;
    }

    if (count > SIZE_MAX / sizeof(char *)) {
        return NULL;
    }

    tuple = malloc(sizeof(Tuple));
    if (tuple == NULL) {
        return NULL;
    }

    tuple->items = calloc(count, sizeof(char *));
    if (tuple->items == NULL) {
        free(tuple);
        return NULL;
    }

    tuple->count = count;

    for (i = 0; i < count; i++) {
        size_t len;
        size_t dest_size;

        if (list[i] == NULL) {
            goto fail;
        }

        len = strnlen(list[i], MAX_STRING_LENGTH);
        if (len >= MAX_STRING_LENGTH) {
            goto fail;
        }

        dest_size = len + 1;
        if (dest_size > MAX_STRING_LENGTH) {
            goto fail;
        }

        tuple->items[i] = malloc(dest_size);
        if (tuple->items[i] == NULL) {
            goto fail;
        }

        if (safe_string_copy(tuple->items[i], dest_size, list[i]) != 0) {
            free(tuple->items[i]);
            tuple->items[i] = NULL;
            goto fail;
        }
    }

    return tuple;

fail:
    for (size_t j = 0; j < i; j++) {
        free(tuple->items[j]);
    }
    free(tuple->items);
    free(tuple);
    return NULL;
}

void free_tuple(Tuple *tuple) {
    size_t i;

    if (tuple == NULL) {
        return;
    }

    for (i = 0; i < tuple->count; i++) {
        free(tuple->items[i]);
    }
    free(tuple->items);
    free(tuple);
}

int main(void) {
    char *list[] = {"apple", "banana", "cherry", "date"};
    size_t count = sizeof(list) / sizeof(list[0]);
    Tuple *tuple;

    tuple = string_list_to_tuple(list, count);
    if (tuple == NULL) {
        fprintf(stderr, "Failed to convert list to tuple\n");
        return 1;
    }

    printf("(");
    for (size_t i = 0; i < tuple->count; i++) {
        printf("\"%s\"", tuple->items[i]);
        if (i < tuple->count - 1) {
            printf(", ");
        }
    }
    printf(")\n");

    free_tuple(tuple);

    return 0;
}
