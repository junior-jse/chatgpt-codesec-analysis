#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <limits.h>
#include <ctype.h>

#define INITIAL_CAPACITY 128
#define MAX_INPUT_LENGTH ((size_t)16 * 1024 * 1024)

char findMaxOccurringChar(const char *str, int *maxCount)
{
    int freq[256] = {0};
    size_t i;
    int max = 0;
    int resultIndex = -1;

    if (str == NULL || maxCount == NULL) {
        if (maxCount != NULL) {
            *maxCount = 0;
        }
        return '\0';
    }

    for (i = 0; str[i] != '\0'; i++) {
        unsigned char c = (unsigned char)str[i];
        if (freq[c] < INT_MAX) {
            freq[c]++;
        }
    }

    for (i = 0; i < 256; i++) {
        if (freq[i] > max) {
            max = freq[i];
            resultIndex = (int)i;
        }
    }

    *maxCount = max;

    if (resultIndex < 0) {
        return '\0';
    }

    return (char)(unsigned char)resultIndex;
}

int main(void)
{
    size_t capacity = INITIAL_CAPACITY;
    size_t length = 0;
    char *buffer;
    int maxCount = 0;
    char maxChar;
    int ch;

    buffer = malloc(capacity);
    if (buffer == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    printf("Enter a string: ");
    fflush(stdout);

    while ((ch = getchar()) != EOF && ch != '\n') {
        if (length >= capacity - 1) {
            size_t newCapacity;
            char *resized;

            if (capacity >= MAX_INPUT_LENGTH) {
                fprintf(stderr, "Input too large\n");
                free(buffer);
                return EXIT_FAILURE;
            }

            if (capacity > SIZE_MAX / 2) {
                newCapacity = MAX_INPUT_LENGTH;
            } else {
                newCapacity = capacity * 2;
                if (newCapacity > MAX_INPUT_LENGTH) {
                    newCapacity = MAX_INPUT_LENGTH;
                }
            }

            if (newCapacity <= capacity) {
                fprintf(stderr, "Input too large\n");
                free(buffer);
                return EXIT_FAILURE;
            }

            resized = realloc(buffer, newCapacity);
            if (resized == NULL) {
                fprintf(stderr, "Memory allocation failed\n");
                free(buffer);
                return EXIT_FAILURE;
            }

            buffer = resized;
            capacity = newCapacity;
        }

        buffer[length] = (char)ch;
        length++;
    }

    if (ferror(stdin)) {
        fprintf(stderr, "Error reading input\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    buffer[length] = '\0';

    if (length == 0) {
        printf("Empty string provided\n");
        free(buffer);
        return EXIT_SUCCESS;
    }

    maxChar = findMaxOccurringChar(buffer, &maxCount);

    if (maxCount > 0) {
        unsigned char outChar = (unsigned char)maxChar;
        if (isprint(outChar)) {
            printf("Maximum occurring character: '%c' (appears %d times)\n",
                   outChar, maxCount);
        } else {
            printf("Maximum occurring character: '\\x%02X' (appears %d times)\n",
                   outChar, maxCount);
        }
    } else {
        printf("No characters found\n");
    }

    free(buffer);
    return EXIT_SUCCESS;
}
