Key fixes applied:
- **CWE-126 (strlen over-read):** Replaced unbounded `strlen` with a bounded `safe_strnlen` helper that never reads past a fixed limit, rejecting non-terminated/over-long input.
- **CWE-120 (strcpy overflow):** Eliminated `strcpy` entirely; copies are done with `memcpy` using the exact pre-computed length, followed by explicit null termination (allocation is always `len + 1`).
- Also switched `strcmp` to a bounded `strncmp` for the same defensive reason.

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_STRING_LENGTH 4096

typedef struct {
    char **strings;
    size_t *counts;
    size_t size;
    size_t capacity;
} StringCounter;

static size_t safe_strnlen(const char *str, size_t max_len) {
    size_t len = 0;
    if (str == NULL) {
        return 0;
    }
    while (len < max_len && str[len] != '\0') {
        len++;
    }
    return len;
}

static char *safe_strdup(const char *str) {
    size_t len;
    char *copy;

    if (str == NULL) {
        return NULL;
    }

    len = safe_strnlen(str, MAX_STRING_LENGTH + 1);
    if (len > MAX_STRING_LENGTH) {
        return NULL;
    }

    copy = malloc(len + 1);
    if (copy == NULL) {
        return NULL;
    }

    memcpy(copy, str, len);
    copy[len] = '\0';
    return copy;
}

static int init_counter(StringCounter *counter, size_t capacity) {
    if (counter == NULL || capacity == 0) {
        return -1;
    }
    counter->strings = calloc(capacity, sizeof(char *));
    if (counter->strings == NULL) {
        return -1;
    }
    counter->counts = calloc(capacity, sizeof(size_t));
    if (counter->counts == NULL) {
        free(counter->strings);
        counter->strings = NULL;
        return -1;
    }
    counter->size = 0;
    counter->capacity = capacity;
    return 0;
}

static void free_counter(StringCounter *counter) {
    if (counter == NULL) {
        return;
    }
    if (counter->strings != NULL) {
        for (size_t i = 0; i < counter->size; i++) {
            free(counter->strings[i]);
        }
        free(counter->strings);
    }
    free(counter->counts);
    counter->strings = NULL;
    counter->counts = NULL;
    counter->size = 0;
    counter->capacity = 0;
}

static int add_string(StringCounter *counter, const char *str) {
    if (counter == NULL || str == NULL || counter->size >= counter->capacity) {
        return -1;
    }
    for (size_t i = 0; i < counter->size; i++) {
        if (strncmp(counter->strings[i], str, MAX_STRING_LENGTH + 1) == 0) {
            counter->counts[i]++;
            return 0;
        }
    }
    counter->strings[counter->size] = safe_strdup(str);
    if (counter->strings[counter->size] == NULL) {
        return -1;
    }
    counter->counts[counter->size] = 1;
    counter->size++;
    return 0;
}

char *find_second_most_frequent(const char **sequence, size_t length) {
    StringCounter counter;
    size_t first_idx = 0;
    size_t second_idx = 0;
    size_t first_count = 0;
    size_t second_count = 0;
    int found_second = 0;
    char *result = NULL;

    if (sequence == NULL || length == 0) {
        return NULL;
    }

    if (init_counter(&counter, length) != 0) {
        return NULL;
    }

    for (size_t i = 0; i < length; i++) {
        if (sequence[i] == NULL) {
            free_counter(&counter);
            return NULL;
        }
        if (add_string(&counter, sequence[i]) != 0) {
            free_counter(&counter);
            return NULL;
        }
    }

    if (counter.size < 2) {
        free_counter(&counter);
        return NULL;
    }

    for (size_t i = 0; i < counter.size; i++) {
        if (counter.counts[i] > first_count) {
            second_count = first_count;
            second_idx = first_idx;
            first_count = counter.counts[i];
            first_idx = i;
            found_second = (second_count > 0) ? 1 : found_second;
        } else if (counter.counts[i] < first_count && counter.counts[i] > second_count) {
            second_count = counter.counts[i];
            second_idx = i;
            found_second = 1;
        }
    }

    if (found_second && second_count > 0) {
        result = safe_strdup(counter.strings[second_idx]);
    }

    free_counter(&counter);
    return result;
}

int main(void) {
    const char *sequence[] = {
        "apple", "banana", "apple", "cherry",
        "banana", "apple", "date", "banana"
    };
    size_t length = sizeof(sequence) / sizeof(sequence[0]);

    char *result = find_second_most_frequent(sequence, length);
    if (result != NULL) {
        printf("Second most frequent string: %s\n", result);
        free(result);
    } else {
        printf("No second most frequent string found.\n");
    }

    return 0;
}
```

Notes on the changes:
- `safe_strnlen` guarantees reads never exceed `MAX_STRING_LENGTH + 1` bytes, so a missing null terminator cannot cause an over-read.
- `safe_strdup` centralizes allocation and copying: buffer size is derived from the bounded length, `memcpy` copies exactly that many bytes, and the terminator is written explicitly — no format-string or unbounded-copy pitfalls.
- All duplicated strings in the counter are guaranteed to be null-terminated, so subsequent operations on `counter.strings[...]` are safe.
- Memory-management behavior (cleanup paths, ownership of the returned string) is unchanged from the original.
