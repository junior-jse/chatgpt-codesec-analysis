```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_WORD_LEN 4096

char **shortlist_words(char **words, int count, int n, int *result_count)
{
    if (words == NULL || count < 0 || n < 0 || result_count == NULL) {
        if (result_count != NULL) {
            *result_count = 0;
        }
        return NULL;
    }

    char **result = malloc((size_t)count * sizeof(char *));
    if (result == NULL) {
        *result_count = 0;
        return NULL;
    }

    int j = 0;
    for (int i = 0; i < count; i++) {
        if (words[i] == NULL) {
            continue;
        }

        size_t len = strnlen(words[i], (size_t)MAX_WORD_LEN + 1);
        if (len > (size_t)MAX_WORD_LEN) {
            continue;
        }

        if ((int)len > n) {
            result[j] = malloc(len + 1);
            if (result[j] == NULL) {
                for (int k = 0; k < j; k++) {
                    free(result[k]);
                }
                free(result);
                *result_count = 0;
                return NULL;
            }
            memcpy(result[j], words[i], len + 1);
            j++;
        }
    }

    *result_count = j;
    return result;
}

void free_words(char **words, int count)
{
    if (words == NULL) {
        return;
    }
    for (int i = 0; i < count; i++) {
        free(words[i]);
    }
    free(words);
}

int main(void)
{
    char *words[] = {"apple", "hi", "banana", "cat", "elephant", "dog"};
    int count = 6;
    int n = 3;
    int result_count = 0;

    char **shortlisted = shortlist_words(words, count, n, &result_count);

    if (shortlisted == NULL && result_count == 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    printf("Words longer than %d:\n", n);
    for (int i = 0; i < result_count; i++) {
        printf("%s\n", shortlisted[i]);
    }

    free_words(shortlisted, result_count);

    return EXIT_SUCCESS;
}
```

Summary of the fixes applied:

1. **CWE-126 (buffer over-read via `strlen`)**: Replaced `strlen` with `strnlen(words[i], MAX_WORD_LEN + 1)`, which bounds the scan to a maximum number of bytes. If no null terminator is found within that bound, the string is treated as invalid and skipped, so a non-terminated string can never trigger an out-of-bounds read.

2. **CWE-120 (buffer overflow via `strcpy`)**: Replaced `strcpy` with `memcpy(result[j], words[i], len + 1)`. The destination buffer is allocated with exactly `len + 1` bytes, and `len` was already validated by `strnlen`, so the copy size is provably correct and includes the null terminator. This avoids the banned `strcpy` and the pitfalls of `strncpy` (which can leave the destination unterminated).
