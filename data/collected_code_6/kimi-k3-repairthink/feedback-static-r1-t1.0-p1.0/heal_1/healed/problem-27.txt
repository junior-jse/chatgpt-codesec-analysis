#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_STRING_SIZE 256

static size_t bounded_strlen(const char *str, size_t max_len) {
    size_t len = 0;

    while (len < max_len && str[len] != '\0') {
        len++;
    }

    return len;
}

void remove_digits(char *str, size_t max_len) {
    if (str == NULL || max_len == 0) {
        return;
    }

    size_t len = bounded_strlen(str, max_len);
    if (len == max_len) {
        len = max_len - 1;
    }

    size_t read_pos = 0;
    size_t write_pos = 0;

    while (read_pos < len) {
        if (!isdigit((unsigned char)str[read_pos])) {
            str[write_pos] = str[read_pos];
            write_pos++;
        }
        read_pos++;
    }

    str[write_pos] = '\0';
}

void process_string_list(char list[][MAX_STRING_SIZE], size_t count) {
    if (list == NULL) {
        return;
    }

    for (size_t i = 0; i < count; i++) {
        remove_digits(list[i], MAX_STRING_SIZE);
    }
}

int main(void) {
    char strings[][MAX_STRING_SIZE] = {
        "abc123def",
        "test456",
        "no_digits_here",
        "789mixed012",
        "999",
        ""
    };

    size_t count = sizeof(strings) / sizeof(strings[0]);

    printf("Original strings:\n");
    for (size_t i = 0; i < count; i++) {
        printf("[%zu]: %s\n", i, strings[i]);
    }

    process_string_list(strings, count);

    printf("\nAfter removing digits:\n");
    for (size_t i = 0; i < count; i++) {
        printf("[%zu]: %s\n", i, strings[i]);
    }

    return 0;
}
