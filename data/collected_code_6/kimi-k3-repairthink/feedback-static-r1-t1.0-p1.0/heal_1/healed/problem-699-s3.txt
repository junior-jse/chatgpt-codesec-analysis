#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 1000

static size_t bounded_strlen(const char *s, size_t max_len)
{
    size_t n = 0;

    while (n < max_len && s[n] != '\0') {
        n++;
    }
    return n;
}

int min_swaps(const char *s1, const char *s2, size_t max_len)
{
    size_t len1;
    size_t len2;
    size_t i;
    size_t count1_s1 = 0;
    size_t count1_s2 = 0;
    int swaps = 0;
    int diff = 0;

    if (s1 == NULL || s2 == NULL) {
        return -1;
    }

    len1 = bounded_strlen(s1, max_len);
    len2 = bounded_strlen(s2, max_len);

    if (len1 != len2) {
        return -1;
    }

    for (i = 0; i < len1; i++) {
        if (s1[i] != '0' && s1[i] != '1') {
            return -1;
        }
        if (s2[i] != '0' && s2[i] != '1') {
            return -1;
        }
        if (s1[i] == '1') {
            count1_s1++;
        }
        if (s2[i] == '1') {
            count1_s2++;
        }
    }

    if (count1_s1 != count1_s2) {
        return -1;
    }

    for (i = 0; i < len1; i++) {
        if (s1[i] != s2[i]) {
            if (s1[i] == '1') {
                diff++;
            } else {
                diff--;
            }
            if (diff > 0) {
                swaps += diff;
            }
        }
    }

    return swaps;
}

int main(void)
{
    char s1[MAX_INPUT_LEN];
    char s2[MAX_INPUT_LEN];
    int result;

    if (fgets(s1, sizeof(s1), stdin) == NULL) {
        return 1;
    }
    if (fgets(s2, sizeof(s2), stdin) == NULL) {
        return 1;
    }

    if (strchr(s1, '\n') == NULL && !feof(stdin)) {
        return 1;
    }
    if (strchr(s2, '\n') == NULL && !feof(stdin)) {
        return 1;
    }

    s1[strcspn(s1, "\n")] = '\0';
    s2[strcspn(s2, "\n")] = '\0';

    result = min_swaps(s1, s2, sizeof(s1));
    printf("%d\n", result);

    return 0;
}
