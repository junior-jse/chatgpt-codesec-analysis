#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

typedef struct {
    int id;
    char name[50];
    double value;
} Tuple;

char* flatten_tuple_list(const Tuple* tuples, size_t count) {
    if (tuples == NULL || count == 0) {
        char* empty = malloc(1);
        if (empty != NULL) {
            empty[0] = '\0';
        }
        return empty;
    }

    size_t total_size = 1;
    for (size_t i = 0; i < count; i++) {
        int needed = snprintf(NULL, 0, "(%d,%.49s,%.2f)",
                              tuples[i].id, tuples[i].name, tuples[i].value);
        if (needed < 0) {
            return NULL;
        }
        if (total_size > SIZE_MAX - (size_t)needed - 1) {
            return NULL;
        }
        total_size += (size_t)needed + 1;
    }

    char* result = malloc(total_size);
    if (result == NULL) {
        return NULL;
    }

    result[0] = '\0';
    size_t offset = 0;

    for (size_t i = 0; i < count; i++) {
        int written = snprintf(result + offset, total_size - offset,
                               "(%d,%.49s,%.2f)",
                               tuples[i].id, tuples[i].name, tuples[i].value);
        if (written < 0 || (size_t)written >= total_size - offset) {
            free(result);
            return NULL;
        }
        offset += (size_t)written;

        if (i < count - 1) {
            if (offset + 1 >= total_size) {
                free(result);
                return NULL;
            }
            result[offset] = ',';
            offset++;
            result[offset] = '\0';
        }
    }

    return result;
}

int main(void) {
    Tuple tuples[3];
    size_t count = sizeof(tuples) / sizeof(tuples[0]);

    tuples[0].id = 1;
    snprintf(tuples[0].name, sizeof(tuples[0].name), "%s", "apple");
    tuples[0].value = 2.5;

    tuples[1].id = 2;
    snprintf(tuples[1].name, sizeof(tuples[1].name), "%s", "banana");
    tuples[1].value = 1.75;

    tuples[2].id = 3;
    snprintf(tuples[2].name, sizeof(tuples[2].name), "%s", "cherry");
    tuples[2].value = 3.0;

    char* flattened = flatten_tuple_list(tuples, count);

    if (flattened == NULL) {
        fprintf(stderr, "Error: Failed to flatten tuple list\n");
        return EXIT_FAILURE;
    }

    printf("%s\n", flattened);
    free(flattened);

    return EXIT_SUCCESS;
}
