#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <stdlib.h>
#include <ctype.h>
#include <sys/types.h>

void reverse_substring(char *start, char *end) {
    while (start < end) {
        char temp = *start;
        *start = *end;
        *end = temp;
        start++;
        end--;
    }
}

void reverse_words(char *str) {
    if (str == NULL || *str == '\0') {
        return;
    }

    char *word_start = NULL;
    char *current = str;

    while (*current) {
        if (word_start == NULL && !isspace((unsigned char)*current)) {
            word_start = current;
        }
        if (word_start && (isspace((unsigned char)*(current + 1)) || *(current + 1) == '\0')) {
            reverse_substring(word_start, current);
            word_start = NULL;
        }
        current++;
    }

    reverse_substring(str, current - 1);

    current = str;
    char *write_ptr = str;
    int space_found = 0;

    while (*current && isspace((unsigned char)*current)) {
        current++;
    }

    while (*current) {
        if (isspace((unsigned char)*current)) {
            space_found = 1;
        } else {
            if (space_found && write_ptr != str) {
                *write_ptr++ = ' ';
            }
            space_found = 0;
            *write_ptr++ = *current;
        }
        current++;
    }
    *write_ptr = '\0';
}

int main(void) {
    char *buffer = NULL;
    size_t buffer_size = 0;
    ssize_t len;

    printf("Enter a string: ");
    len = getline(&buffer, &buffer_size, stdin);
    if (len < 0) {
        fprintf(stderr, "Error reading input\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
        len--;
    }

    if (len == 0) {
        printf("Empty string provided\n");
        free(buffer);
        return EXIT_SUCCESS;
    }

    printf("Original: \"%s\"\n", buffer);
    reverse_words(buffer);
    printf("Reversed: \"%s\"\n", buffer);

    free(buffer);
    return EXIT_SUCCESS;
}
