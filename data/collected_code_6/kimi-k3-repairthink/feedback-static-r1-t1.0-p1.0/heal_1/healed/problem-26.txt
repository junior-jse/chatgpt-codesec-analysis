```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define BUFFER_SIZE 4

typedef struct {
    char *data;
} Tuple;

static size_t bounded_strlen(const char *s, size_t max_len)
{
    const char *end;

    if (s == NULL) {
        return 0;
    }

    end = memchr(s, '\0', max_len);
    if (end == NULL) {
        return max_len;
    }

    return (size_t)(end - s);
}

int has_all_k_elements(Tuple *tuples, size_t tuple_count, size_t k)
{
    size_t i;
    size_t element_len;
    const char *element;

    if (tuples == NULL || k == 0) {
        return 0;
    }

    for (i = 0; i < tuple_count; i++) {
        if (tuples[i].data == NULL) {
            continue;
        }

        element = tuples[i].data;
        element_len = bounded_strlen(element, k + 1);

        if (element_len != k) {
            return 0;
        }
    }

    return 1;
}

int main(void)
{
    Tuple tuples[3];
    size_t k = 3;
    int result;

    tuples[0].data = malloc(BUFFER_SIZE);
    tuples[1].data = malloc(BUFFER_SIZE);
    tuples[2].data = malloc(BUFFER_SIZE);

    if (tuples[0].data == NULL || tuples[1].data == NULL || tuples[2].data == NULL) {
        free(tuples[0].data);
        free(tuples[1].data);
        free(tuples[2].data);
        return 1;
    }

    snprintf(tuples[0].data, BUFFER_SIZE, "abc");
    snprintf(tuples[1].data, BUFFER_SIZE, "def");
    snprintf(tuples[2].data, BUFFER_SIZE, "ghi");

    result = has_all_k_elements(tuples, 3, k);

    if (result) {
        printf("All tuples have %zu elements\n", k);
    } else {
        printf("Not all tuples have %zu elements\n", k);
    }

    free(tuples[0].data);
    free(tuples[1].data);
    free(tuples[2].data);

    return 0;
}
```
