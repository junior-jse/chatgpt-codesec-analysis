#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STRING_LENGTH 4096

static size_t bounded_strlen(const char* s, size_t max_len) {
    size_t len = 0;
    if (s == NULL) {
        return 0;
    }
    while (len < max_len && s[len] != '\0') {
        len++;
    }
    return len;
}

char* concatenate_strings(const char** list, size_t count, const char* separator) {
    if (list == NULL || count == 0) {
        char* empty = malloc(1);
        if (empty != NULL) {
            empty[0] = '\0';
        }
        return empty;
    }

    size_t total_len = 0;
    size_t sep_len = bounded_strlen(separator, MAX_STRING_LENGTH);

    for (size_t i = 0; i < count; i++) {
        if (list[i] != NULL) {
            size_t len = bounded_strlen(list[i], MAX_STRING_LENGTH);
            if (len > SIZE_MAX - total_len) {
                return NULL;
            }
            total_len += len;
        }
        if (i < count - 1 && separator != NULL) {
            if (sep_len > SIZE_MAX - total_len) {
                return NULL;
            }
            total_len += sep_len;
        }
    }

    if (total_len == SIZE_MAX) {
        return NULL;
    }

    char* result = malloc(total_len + 1);
    if (result == NULL) {
        return NULL;
    }

    size_t offset = 0;
    size_t capacity = total_len + 1;

    for (size_t i = 0; i < count; i++) {
        if (list[i] != NULL) {
            size_t len = bounded_strlen(list[i], MAX_STRING_LENGTH);
            if (len >= capacity - offset) {
                len = capacity - offset - 1;
            }
            if (len > 0) {
                memcpy(result + offset, list[i], len);
                offset += len;
            }
        }
        if (i < count - 1 && separator != NULL) {
            size_t len = sep_len;
            if (len >= capacity - offset) {
                len = capacity - offset - 1;
            }
            if (len > 0) {
                memcpy(result + offset, separator, len);
                offset += len;
            }
        }
    }
    result[offset] = '\0';

    return result;
}

int main(void) {
    const char* words[] = {"Hello", "World", "C", "Programming"};
    size_t count = sizeof(words) / sizeof(words[0]);

    char* result = concatenate_strings(words, count, " ");
    if (result == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    printf("%s\n", result);
    free(result);

    return EXIT_SUCCESS;
}
