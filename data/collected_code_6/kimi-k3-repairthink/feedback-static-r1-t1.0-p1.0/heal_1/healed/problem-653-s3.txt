## Analysis of Identified Weaknesses

1. **`duplicate_string` (CWE-126 / CWE-120):** Guard against `len + 1` integer overflow, allocate `len + 1`, copy exactly `len` bytes into the sufficiently-sized buffer, and explicitly NUL-terminate — so the destination provably holds the source data.
2. **Variable scope warnings:** `new_capacity` (both functions) and `group` moved into the narrowest possible scope (C99 declarations).
3. **Extra hardening (experienced-dev practice):** overflow checks before capacity doubling and before `new_capacity * sizeof(...)`; avoid the redundant second `find_group` lookup after `add_group`.

## Rewritten Code

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define INITIAL_CAPACITY 4

typedef struct {
    char *key;
    char **values;
    size_t value_count;
    size_t value_capacity;
} Group;

typedef struct {
    char *key;
    char *value;
} KeyValuePair;

typedef struct {
    Group *groups;
    size_t group_count;
    size_t group_capacity;
} GroupMap;

static char *duplicate_string(const char *str) {
    size_t len;
    char *copy;

    if (str == NULL) {
        return NULL;
    }

    len = strlen(str);
    if (len == SIZE_MAX) {
        return NULL;
    }

    copy = malloc(len + 1);
    if (copy == NULL) {
        return NULL;
    }

    memcpy(copy, str, len);
    copy[len] = '\0';

    return copy;
}

static void init_group_map(GroupMap *map) {
    if (map == NULL) {
        return;
    }
    map->groups = NULL;
    map->group_count = 0;
    map->group_capacity = 0;
}

static void free_group_map(GroupMap *map) {
    if (map == NULL) {
        return;
    }

    for (size_t i = 0; i < map->group_count; i++) {
        free(map->groups[i].key);
        for (size_t j = 0; j < map->groups[i].value_count; j++) {
            free(map->groups[i].values[j]);
        }
        free(map->groups[i].values);
    }
    free(map->groups);

    map->groups = NULL;
    map->group_count = 0;
    map->group_capacity = 0;
}

static Group *find_group(GroupMap *map, const char *key) {
    if (map == NULL || key == NULL) {
        return NULL;
    }

    for (size_t i = 0; i < map->group_count; i++) {
        if (strcmp(map->groups[i].key, key) == 0) {
            return &map->groups[i];
        }
    }
    return NULL;
}

static int add_value_to_group(Group *group, const char *value) {
    char *value_copy;

    if (group == NULL || value == NULL) {
        return -1;
    }

    if (group->value_count >= group->value_capacity) {
        if (group->value_capacity > SIZE_MAX / 2) {
            return -1;
        }
        size_t new_capacity = (group->value_capacity == 0)
                              ? INITIAL_CAPACITY
                              : group->value_capacity * 2;
        if (new_capacity > SIZE_MAX / sizeof(char *)) {
            return -1;
        }
        char **new_values = realloc(group->values, new_capacity * sizeof(char *));
        if (new_values == NULL) {
            return -1;
        }
        group->values = new_values;
        group->value_capacity = new_capacity;
    }

    value_copy = duplicate_string(value);
    if (value_copy == NULL) {
        return -1;
    }
    group->values[group->value_count] = value_copy;
    group->value_count++;

    return 0;
}

static int add_group(GroupMap *map, const char *key) {
    Group *new_group;

    if (map == NULL || key == NULL) {
        return -1;
    }

    if (map->group_count >= map->group_capacity) {
        if (map->group_capacity > SIZE_MAX / 2) {
            return -1;
        }
        size_t new_capacity = (map->group_capacity == 0)
                              ? INITIAL_CAPACITY
                              : map->group_capacity * 2;
        if (new_capacity > SIZE_MAX / sizeof(Group)) {
            return -1;
        }
        Group *new_groups = realloc(map->groups, new_capacity * sizeof(Group));
        if (new_groups == NULL) {
            return -1;
        }
        map->groups = new_groups;
        map->group_capacity = new_capacity;
    }

    new_group = &map->groups[map->group_count];
    new_group->key = duplicate_string(key);
    if (new_group->key == NULL) {
        return -1;
    }
    new_group->values = NULL;
    new_group->value_count = 0;
    new_group->value_capacity = 0;
    map->group_count++;

    return 0;
}

static int group_key_value_pairs(GroupMap *map, const KeyValuePair *pairs, size_t pair_count) {
    if (map == NULL || (pairs == NULL && pair_count > 0)) {
        return -1;
    }

    for (size_t i = 0; i < pair_count; i++) {
        if (pairs[i].key == NULL || pairs[i].value == NULL) {
            return -1;
        }

        Group *group = find_group(map, pairs[i].key);
        if (group == NULL) {
            if (add_group(map, pairs[i].key) != 0) {
                return -1;
            }
            group = &map->groups[map->group_count - 1];
        }

        if (add_value_to_group(group, pairs[i].value) != 0) {
            return -1;
        }
    }

    return 0;
}

static void print_group_map(const GroupMap *map) {
    if (map == NULL) {
        return;
    }

    for (size_t i = 0; i < map->group_count; i++) {
        printf("%s: [", map->groups[i].key);
        for (size_t j = 0; j < map->groups[i].value_count; j++) {
            printf("%s", map->groups[i].values[j]);
            if (j < map->groups[i].value_count - 1) {
                printf(", ");
            }
        }
        printf("]\n");
    }
}

int main(void) {
    KeyValuePair pairs[] = {
        {"fruit", "apple"},
        {"vegetable", "carrot"},
        {"fruit", "banana"},
        {"vegetable", "broccoli"},
        {"fruit", "cherry"},
        {"grain", "rice"},
        {"grain", "wheat"}
    };
    size_t pair_count = sizeof(pairs) / sizeof(pairs[0]);
    GroupMap map;
    int result;

    init_group_map(&map);

    result = group_key_value_pairs(&map, pairs, pair_count);
    if (result != 0) {
        fprintf(stderr, "Error: Failed to group key-value pairs\n");
        free_group_map(&map);
        return EXIT_FAILURE;
    }

    print_group_map(&map);

    free_group_map(&map);

    return EXIT_SUCCESS;
}
```

## Notes on the Fixes

- **`memcpy` safety:** The destination buffer (`len + 1` bytes) always holds the `len` copied bytes plus the terminator, and the terminator is written explicitly rather than relying on the source's NUL being copied.
- **`strlen` note:** CWE-126 cannot be fully eliminated for an arbitrary `const char *` (the buffer size is unknowable), so the contract remains that callers pass valid NUL-terminated strings — which `main` guarantees here. The `SIZE_MAX` guard prevents wraparound on `len + 1`.
- **Scope reduction:** `new_capacity`, `new_values`, `new_groups`, and `group` are now declared exactly where used.
- **Capacity growth:** Doubling and the `* sizeof(...)` multiplication are both overflow-checked before `realloc`.
