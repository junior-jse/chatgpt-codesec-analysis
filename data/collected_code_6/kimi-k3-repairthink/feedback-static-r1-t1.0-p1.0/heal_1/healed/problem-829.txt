#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_STRING_LENGTH 4096

typedef struct {
    char *str;
    size_t count;
} StringFreq;

static char *duplicate_string(const char *s) {
    if (s == NULL) {
        return NULL;
    }

    size_t len = strnlen(s, MAX_STRING_LENGTH + 1);
    if (len > MAX_STRING_LENGTH) {
        return NULL;
    }

    char *copy = malloc(len + 1);
    if (copy == NULL) {
        return NULL;
    }

    memcpy(copy, s, len);
    copy[len] = '\0';

    return copy;
}

static void free_string_freqs(StringFreq *freqs, size_t count) {
    if (freqs == NULL) {
        return;
    }
    for (size_t i = 0; i < count; i++) {
        free(freqs[i].str);
    }
    free(freqs);
}

char *second_most_frequent(char **sequence, size_t size) {
    if (sequence == NULL || size < 2) {
        return NULL;
    }

    StringFreq *freqs = calloc(size, sizeof(StringFreq));
    if (freqs == NULL) {
        return NULL;
    }

    size_t unique_count = 0;

    for (size_t i = 0; i < size; i++) {
        if (sequence[i] == NULL) {
            continue;
        }

        int found = 0;
        for (size_t j = 0; j < unique_count; j++) {
            if (strncmp(freqs[j].str, sequence[i], MAX_STRING_LENGTH + 1) == 0) {
                freqs[j].count++;
                found = 1;
                break;
            }
        }

        if (!found) {
            freqs[unique_count].str = duplicate_string(sequence[i]);
            if (freqs[unique_count].str == NULL) {
                free_string_freqs(freqs, unique_count);
                return NULL;
            }
            freqs[unique_count].count = 1;
            unique_count++;
        }
    }

    if (unique_count < 2) {
        free_string_freqs(freqs, unique_count);
        return NULL;
    }

    size_t first_idx = unique_count;
    size_t second_idx = unique_count;

    for (size_t i = 0; i < unique_count; i++) {
        if (first_idx == unique_count || freqs[i].count > freqs[first_idx].count) {
            second_idx = first_idx;
            first_idx = i;
        } else if (second_idx == unique_count || freqs[i].count > freqs[second_idx].count) {
            second_idx = i;
        }
    }

    char *result = NULL;
    if (second_idx != unique_count) {
        result = duplicate_string(freqs[second_idx].str);
    }

    free_string_freqs(freqs, unique_count);
    return result;
}

int main(void) {
    char *sequence[] = {
        "apple", "banana", "apple", "cherry",
        "banana", "apple", "cherry", "banana"
    };
    size_t size = sizeof(sequence) / sizeof(sequence[0]);

    char *result = second_most_frequent(sequence, size);
    if (result != NULL) {
        printf("Second most frequent: %s\n", result);
        free(result);
    } else {
        printf("No second most frequent string found\n");
    }

    return 0;
}
