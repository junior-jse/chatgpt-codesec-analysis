#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define INPUT_BUFFER_SIZE 1024

char* minimizeString(const char* str, size_t str_size, char ch) {
    if (str == NULL || str_size == 0) {
        return NULL;
    }

    size_t len = strnlen(str, str_size);
    if (len == str_size) {
        return NULL;
    }

    size_t count = 0;
    for (size_t i = 0; i < len; i++) {
        if (str[i] == ch) {
            count++;
        }
    }

    if (count == len) {
        char* result = (char*)malloc(1);
        if (result != NULL) {
            result[0] = '\0';
        }
        return result;
    }

    char* result = (char*)malloc(len - count + 1);
    if (result == NULL) {
        return NULL;
    }

    size_t j = 0;
    for (size_t i = 0; i < len; i++) {
        if (str[i] != ch) {
            result[j++] = str[i];
        }
    }
    result[j] = '\0';

    return result;
}

int main(void) {
    char input[INPUT_BUFFER_SIZE];
    char ch;

    printf("Enter a string: ");
    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    size_t len = strnlen(input, sizeof(input));
    if (len == sizeof(input)) {
        fprintf(stderr, "Input error: string is not null-terminated\n");
        return 1;
    }

    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else {
        int c;
        while ((c = getchar()) != '\n' && c != EOF) {
        }
    }

    printf("Enter the character to remove: ");
    if (scanf(" %c", &ch) != 1) {
        fprintf(stderr, "Error reading character\n");
        return 1;
    }

    char* result = minimizeString(input, sizeof(input), ch);
    if (result == NULL) {
        fprintf(stderr, "Failed to minimize string\n");
        return 1;
    }

    printf("Minimized string: %s\n", result);

    free(result);
    result = NULL;

    return 0;
}
