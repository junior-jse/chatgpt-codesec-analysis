#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int max_list_sum(int **lists, int num_lists, const int *sizes) {
    int max_sum = INT_MIN;
    int found_valid_list = 0;
    
    if (lists == NULL || sizes == NULL || num_lists <= 0) {
        return 0;
    }
    
    for (int i = 0; i < num_lists; i++) {
        if (lists[i] == NULL || sizes[i] <= 0) {
            continue;
        }
        
        int current_sum = 0;
        for (int j = 0; j < sizes[i]; j++) {
            current_sum += lists[i][j];
        }
        
        if (!found_valid_list || current_sum > max_sum) {
            max_sum = current_sum;
            found_valid_list = 1;
        }
    }
    
    return found_valid_list ? max_sum : 0;
}
