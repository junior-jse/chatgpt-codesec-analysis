```c
#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LENGTH 4096

char **split_at_lowercase(const char *str, size_t max_len, size_t *count) {
    if (str == NULL || count == NULL || max_len == 0) {
        return NULL;
    }

    *count = 0;

    size_t len = strnlen(str, max_len);
    if (len == 0) {
        return NULL;
    }

    char **result = malloc((len + 1) * sizeof(char *));
    if (result == NULL) {
        return NULL;
    }

    size_t start = 0;
    size_t idx = 0;

    for (size_t i = 0; i <= len; i++) {
        if (i == len || islower((unsigned char)str[i])) {
            size_t segment_len = i - start;
            if (segment_len > 0) {
                result[idx] = malloc((segment_len + 1) * sizeof(char));
                if (result[idx] == NULL) {
                    for (size_t j = 0; j < idx; j++) {
                        free(result[j]);
                    }
                    free(result);
                    return NULL;
                }
                memcpy(result[idx], str + start, segment_len);
                result[idx][segment_len] = '\0';
                idx++;
            }
            start = i + 1;
        }
    }

    *count = idx;
    return result;
}

void free_split_result(char **result, size_t count) {
    if (result == NULL) {
        return;
    }
    for (size_t i = 0; i < count; i++) {
        free(result[i]);
    }
    free(result);
}

int main(void) {
    const char *input = "HelloWorldThisIsATest";
    size_t count = 0;

    char **segments = split_at_lowercase(input, MAX_INPUT_LENGTH, &count);

    if (segments == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    printf("Input: %s\n", input);
    printf("Segments found: %zu\n", count);

    for (size_t i = 0; i < count; i++) {
        printf("Segment %zu: %s\n", i + 1, segments[i]);
    }

    free_split_result(segments, count);

    return EXIT_SUCCESS;
}
```

Key fixes applied:

1. **CWE-126 (strlen over-read):** Replaced `strlen(str)` with `strnlen(str, max_len)`, bounding the scan to `MAX_INPUT_LENGTH` so an unterminated string can never trigger an over-read. The split loop was also adjusted (`i == len ||` with short-circuit evaluation) so `str[len]` is never dereferenced when the bound is hit without finding a `'\0'`.

2. **CWE-120 (strncpy misuse):** Replaced `strncpy` with `memcpy` using the exactly-known `segment_len`, followed by explicit `'\0'` termination. This eliminates `strncpy`'s problematic semantics (no guaranteed termination, silent truncation, zero-padding overhead).

3. **Supporting hardening:** `*count` is zeroed up-front so failure paths never leave it indeterminate; the dead error check in `main` (`segments == NULL && count > 0`, which could never be true) was corrected; `size_t` is used consistently for lengths/indices/counts to avoid signedness and truncation issues; and `_POSIX_C_SOURCE 200809L` guarantees `strnlen` is declared even under strict `-std=c11` compilation.
