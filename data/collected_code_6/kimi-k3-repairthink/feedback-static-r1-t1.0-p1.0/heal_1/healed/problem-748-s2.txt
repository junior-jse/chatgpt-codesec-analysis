Here is the rewritten code. Summary of fixes applied:

- **`strlen` (CWE-126, both occurrences):** the function now takes an explicit `input_len` parameter, and a NUL-terminated working copy (`work`) is made so `regexec` can never over-read. The tail length is computed with a bounded `memchr` search instead of `strlen`. In `main`, `sizeof - 1` is used on a string literal array, eliminating `strlen` entirely.
- **`strcpy` (CWE-120):** replaced with a bounded `memcpy` that copies `remaining + 1` bytes (including the terminator) only after an explicit capacity check.
- **`strncpy` (CWE-120):** replaced with `memcpy` of exactly `prefix_len` bytes; termination is handled explicitly afterward.
- **Dead stores:** removed the unread assignments to `result_capacity` and `result_len` in the no-match exit path.
- **Additional hardening:** integer-overflow guard on `input_len * 2 + 1`, corrected an off-by-one in the capacity calculation (`+ 3` accounts for prefix + space + capital + NUL), and all error paths now free every allocated resource.

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
#include <regex.h>

char* add_spaces_before_capitals(const char* input, size_t input_len) {
    regex_t regex;
    regmatch_t match;
    const char* cursor;
    const char* nul;
    char* work;
    char* result;
    size_t work_len;
    size_t result_len;
    size_t result_capacity;
    int ret;

    if (input == NULL) {
        return NULL;
    }

    if (input_len > (SIZE_MAX - 1) / 2) {
        return NULL;
    }

    ret = regcomp(&regex, "[A-Z]", REG_EXTENDED);
    if (ret != 0) {
        return NULL;
    }

    result_capacity = input_len * 2 + 1;
    result = malloc(result_capacity);
    if (result == NULL) {
        regfree(&regex);
        return NULL;
    }

    work = malloc(input_len + 1);
    if (work == NULL) {
        free(result);
        regfree(&regex);
        return NULL;
    }

    memcpy(work, input, input_len);
    work[input_len] = '\0';
    nul = memchr(work, '\0', input_len + 1);
    work_len = (size_t)(nul - work);

    result[0] = '\0';
    result_len = 0;
    cursor = work;

    while (*cursor != '\0') {
        size_t prefix_len;
        size_t needed;

        ret = regexec(&regex, cursor, 1, &match, 0);

        if (ret == REG_NOMATCH) {
            size_t remaining = work_len - (size_t)(cursor - work);
            if (result_len + remaining + 1 > result_capacity) {
                char* temp = realloc(result, result_len + remaining + 1);
                if (temp == NULL) {
                    free(work);
                    free(result);
                    regfree(&regex);
                    return NULL;
                }
                result = temp;
            }
            memcpy(result + result_len, cursor, remaining + 1);
            break;
        } else if (ret != 0) {
            free(work);
            free(result);
            regfree(&regex);
            return NULL;
        }

        prefix_len = (size_t)match.rm_so;
        needed = result_len + prefix_len + 3;

        if (needed > result_capacity) {
            char* temp = realloc(result, needed * 2);
            if (temp == NULL) {
                free(work);
                free(result);
                regfree(&regex);
                return NULL;
            }
            result = temp;
            result_capacity = needed * 2;
        }

        memcpy(result + result_len, cursor, prefix_len);
        result_len += prefix_len;

        if (result_len > 0) {
            result[result_len++] = ' ';
        }

        result[result_len++] = cursor[match.rm_so];
        result[result_len] = '\0';

        cursor += match.rm_eo;
    }

    free(work);
    regfree(&regex);
    return result;
}

int main(void) {
    const char test_string[] = "HelloWorldThisIsATest";
    char* result;

    result = add_spaces_before_capitals(test_string, sizeof(test_string) - 1);

    if (result == NULL) {
        fprintf(stderr, "Error processing string\n");
        return EXIT_FAILURE;
    }

    printf("Original: %s\n", test_string);
    printf("Modified: %s\n", result);

    free(result);

    return EXIT_SUCCESS;
}
