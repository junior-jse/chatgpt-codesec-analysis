#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STRING_LENGTH ((size_t)16 * 1024 * 1024)

typedef struct {
    char **items;
    size_t count;
} StringList;

static int bounded_strlen(const char *s, size_t *out_length) {
    size_t length = 0;

    if (s == NULL || out_length == NULL) {
        return -1;
    }

    while (length < MAX_STRING_LENGTH) {
        if (s[length] == '\0') {
            *out_length = length;
            return 0;
        }
        length++;
    }

    return -1;
}

char *join_tuple_with_string(char **tuple, size_t count, const char *separator) {
    size_t total_length;
    size_t separator_length;
    char *result;
    char *current_pos;
    size_t remaining;

    if (tuple == NULL || separator == NULL || count == 0) {
        return NULL;
    }

    if (bounded_strlen(separator, &separator_length) != 0) {
        return NULL;
    }

    total_length = 1;

    for (size_t i = 0; i < count; i++) {
        size_t item_length;

        if (tuple[i] == NULL) {
            return NULL;
        }
        if (bounded_strlen(tuple[i], &item_length) != 0) {
            return NULL;
        }
        if (item_length > SIZE_MAX - separator_length) {
            return NULL;
        }
        if (total_length > SIZE_MAX - (item_length + separator_length)) {
            return NULL;
        }
        total_length += item_length + separator_length;
    }

    result = malloc(total_length);
    if (result == NULL) {
        return NULL;
    }

    current_pos = result;
    remaining = total_length - 1;

    for (size_t i = 0; i < count; i++) {
        size_t item_length;

        if (bounded_strlen(tuple[i], &item_length) != 0) {
            free(result);
            return NULL;
        }
        if (item_length > remaining) {
            free(result);
            return NULL;
        }
        memcpy(current_pos, tuple[i], item_length);
        current_pos += item_length;
        remaining -= item_length;

        if (separator_length > remaining) {
            free(result);
            return NULL;
        }
        memcpy(current_pos, separator, separator_length);
        current_pos += separator_length;
        remaining -= separator_length;
    }
    *current_pos = '\0';

    return result;
}

StringList *tuple_to_list_with_string(char **tuple, size_t count, const char *separator) {
    StringList *list;
    size_t separator_length;

    if (tuple == NULL || separator == NULL || count == 0) {
        return NULL;
    }

    if (bounded_strlen(separator, &separator_length) != 0) {
        return NULL;
    }

    if (count > SIZE_MAX / 2) {
        return NULL;
    }

    list = malloc(sizeof(StringList));
    if (list == NULL) {
        return NULL;
    }

    list->count = count * 2;

    if (list->count > SIZE_MAX / sizeof(char *)) {
        free(list);
        return NULL;
    }

    list->items = malloc(list->count * sizeof(char *));
    if (list->items == NULL) {
        free(list);
        return NULL;
    }

    for (size_t i = 0; i < count; i++) {
        size_t item_length;
        size_t item_alloc_size;
        size_t separator_alloc_size;

        if (tuple[i] == NULL || bounded_strlen(tuple[i], &item_length) != 0) {
            for (size_t j = 0; j < i * 2; j++) {
                free(list->items[j]);
            }
            free(list->items);
            free(list);
            return NULL;
        }

        item_alloc_size = item_length + 1;
        list->items[i * 2] = malloc(item_alloc_size);
        if (list->items[i * 2] == NULL) {
            for (size_t j = 0; j < i * 2; j++) {
                free(list->items[j]);
            }
            free(list->items);
            free(list);
            return NULL;
        }
        if (item_length + 1 > item_alloc_size) {
            for (size_t j = 0; j <= i * 2; j++) {
                free(list->items[j]);
            }
            free(list->items);
            free(list);
            return NULL;
        }
        memcpy(list->items[i * 2], tuple[i], item_length + 1);

        separator_alloc_size = separator_length + 1;
        list->items[i * 2 + 1] = malloc(separator_alloc_size);
        if (list->items[i * 2 + 1] == NULL) {
            for (size_t j = 0; j <= i * 2; j++) {
                free(list->items[j]);
            }
            free(list->items);
            free(list);
            return NULL;
        }
        if (separator_length + 1 > separator_alloc_size) {
            for (size_t j = 0; j <= i * 2 + 1; j++) {
                free(list->items[j]);
            }
            free(list->items);
            free(list);
            return NULL;
        }
        memcpy(list->items[i * 2 + 1], separator, separator_length + 1);
    }

    return list;
}

void free_string_list(StringList *list) {
    if (list == NULL) {
        return;
    }

    if (list->items != NULL) {
        for (size_t i = 0; i < list->count; i++) {
            free(list->items[i]);
        }
        free(list->items);
    }

    free(list);
}

int main(void) {
    char *tuple[] = {"apple", "banana", "cherry"};
    size_t tuple_count = sizeof(tuple) / sizeof(tuple[0]);
    const char *separator = "fruit";
    StringList *result;

    result = tuple_to_list_with_string(tuple, tuple_count, separator);
    if (result == NULL) {
        fprintf(stderr, "Error: Failed to convert tuple to list\n");
        return EXIT_FAILURE;
    }

    printf("[");
    for (size_t i = 0; i < result->count; i++) {
        printf("\"%s\"", result->items[i]);
        if (i < result->count - 1) {
            printf(", ");
        }
    }
    printf("]\n");

    free_string_list(result);

    return EXIT_SUCCESS;
}
