#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_ROMAN_LENGTH 16
#define MAX_SYMBOL_LENGTH 3

char* int_to_roman(int num) {
    if (num < 1 || num > 3999) {
        return NULL;
    }

    const int values[] = {1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1};
    const char* symbols[] = {"M", "CM", "D", "CD", "C", "XC", "L", "XL", "X", "IX", "V", "IV", "I"};
    const int num_symbols = (int)(sizeof(values) / sizeof(values[0]));

    char* roman = (char*)malloc(MAX_ROMAN_LENGTH * sizeof(char));
    if (roman == NULL) {
        return NULL;
    }

    roman[0] = '\0';
    int remaining = num;
    size_t current_len = 0;

    for (int i = 0; i < num_symbols; i++) {
        while (remaining >= values[i]) {
            size_t sym_len = strnlen(symbols[i], MAX_SYMBOL_LENGTH);
            if (sym_len == 0 || sym_len >= MAX_SYMBOL_LENGTH ||
                current_len + sym_len >= MAX_ROMAN_LENGTH) {
                free(roman);
                return NULL;
            }
            int written = snprintf(roman + current_len,
                                   MAX_ROMAN_LENGTH - current_len,
                                   "%.*s", (int)sym_len, symbols[i]);
            if (written < 0 || (size_t)written != sym_len) {
                free(roman);
                return NULL;
            }
            current_len += (size_t)written;
            remaining -= values[i];
        }
    }

    roman[current_len] = '\0';
    return roman;
}

int main(void) {
    int num = 1994;
    char* roman = int_to_roman(num);
    if (roman != NULL) {
        printf("%d -> %s\n", num, roman);
        free(roman);
    } else {
        printf("Invalid input or memory allocation failed.\n");
    }
    return 0;
}
