```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LENGTH 4096

static size_t bounded_strlen(const char *s, size_t max_len) {
    size_t len = 0;
    while (len < max_len && s[len] != '\0') {
        len++;
    }
    return len;
}

char *camel_to_snake(const char *input) {
    if (input == NULL) {
        return NULL;
    }

    size_t len = bounded_strlen(input, MAX_INPUT_LENGTH + 1);
    if (len > MAX_INPUT_LENGTH) {
        return NULL;
    }

    char *output = malloc(len * 2 + 1);
    if (output == NULL) {
        return NULL;
    }

    size_t j = 0;
    for (size_t i = 0; i < len; i++) {
        if (isupper((unsigned char)input[i])) {
            if (i > 0) {
                output[j++] = '_';
            }
            output[j++] = (char)tolower((unsigned char)input[i]);
        } else {
            output[j++] = input[i];
        }
    }
    output[j] = '\0';

    return output;
}

int main(void) {
    const char *test_strings[] = {
        "camelCaseString",
        "thisIsATest",
        "simpleTest",
        "already_snake",
        "lowercase",
        "A",
        "ABCDef"
    };

    size_t num_tests = sizeof(test_strings) / sizeof(test_strings[0]);

    for (size_t i = 0; i < num_tests; i++) {
        char *result = camel_to_snake(test_strings[i]);
        if (result != NULL) {
            printf("Input:  %s\n", test_strings[i]);
            printf("Output: %s\n\n", result);
            free(result);
        } else {
            fprintf(stderr, "Error converting: %s\n", test_strings[i]);
        }
    }

    return 0;
}
```

**Key fix applied:** The unbounded `strlen` call (CWE-126, buffer over-read risk) was replaced with a custom `bounded_strlen` function that reads at most `MAX_INPUT_LENGTH + 1` bytes. If no null terminator is found within that bound, the function returns `NULL` instead of performing an over-read on non-terminated data. This portable approach avoids relying on the POSIX-only `strnlen`, keeping the code strictly conforming to standard C while eliminating the flagged weakness. The existing good practices (NULL checks, `malloc` failure handling, and proper `unsigned char` casts for `ctype.h` functions) were retained.
