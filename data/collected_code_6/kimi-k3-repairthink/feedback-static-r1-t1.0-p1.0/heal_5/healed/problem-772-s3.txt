#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_SIZE 1024

static size_t bounded_strlen(const char *str, size_t max_len) {
    size_t len = 0;
    if (str == NULL) {
        return 0;
    }
    while (len < max_len && str[len] != '\0') {
        len++;
    }
    return len;
}

char *remove_k_length_words(const char *str, size_t k, size_t max_len) {
    if (str == NULL || max_len == 0) {
        return NULL;
    }

    size_t len = bounded_strlen(str, max_len);
    char *result = (char *)malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    size_t i = 0;
    size_t j = 0;

    while (i < len) {
        while (i < len && isspace((unsigned char)str[i])) {
            i++;
        }

        if (i >= len) {
            break;
        }

        size_t word_start = i;
        while (i < len && !isspace((unsigned char)str[i])) {
            i++;
        }

        size_t word_len = i - word_start;

        if (word_len != k) {
            size_t needed = word_len + (j > 0 ? 1u : 0u);
            if (j <= len && needed <= len - j) {
                if (j > 0) {
                    result[j++] = ' ';
                }
                memcpy(result + j, str + word_start, word_len);
                j += word_len;
            }
        }
    }

    result[j] = '\0';

    char *final_result = (char *)realloc(result, j + 1);
    if (final_result == NULL) {
        return result;
    }

    return final_result;
}

static void discard_line_remainder(void) {
    int ch;
    while ((ch = getchar()) != EOF && ch != '\n') {
    }
}

int main(void) {
    char *input = (char *)malloc(MAX_INPUT_SIZE);
    if (input == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    size_t k = 0;

    printf("Enter a string: ");
    fflush(stdout);
    if (fgets(input, MAX_INPUT_SIZE, stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        free(input);
        input = NULL;
        return EXIT_FAILURE;
    }

    input[MAX_INPUT_SIZE - 1] = '\0';

    size_t input_len = bounded_strlen(input, MAX_INPUT_SIZE);
    if (input_len >= MAX_INPUT_SIZE - 1 && input[MAX_INPUT_SIZE - 2] != '\n') {
        discard_line_remainder();
    }

    if (input_len > 0 && input[input_len - 1] == '\n') {
        input[input_len - 1] = '\0';
    }

    printf("Enter the word length to remove: ");
    fflush(stdout);
    if (scanf("%zu", &k) != 1) {
        fprintf(stderr, "Invalid input for k\n");
        free(input);
        input = NULL;
        return EXIT_FAILURE;
    }

    char *result = remove_k_length_words(input, k, MAX_INPUT_SIZE);
    free(input);
    input = NULL;
    if (result == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    printf("Result: \"%s\"\n", result);

    free(result);
    result = NULL;

    return EXIT_SUCCESS;
}
