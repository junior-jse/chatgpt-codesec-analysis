#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STRING_LENGTH ((size_t)16 * 1024 * 1024)

typedef struct {
    char **items;
    size_t count;
} StringList;

static int bounded_strlen(const char *s, size_t *out_length) {
    size_t length = 0;

    if (s == NULL || out_length == NULL) {
        return -1;
    }

    while (length < MAX_STRING_LENGTH) {
        if (s[length] == '\0') {
            *out_length = length;
            return 0;
        }
        length++;
    }

    return -1;
}

static int ranges_overlap(const char *dest, const char *src, size_t length) {
    uintptr_t dest_addr = (uintptr_t)dest;
    uintptr_t src_addr = (uintptr_t)src;

    if (length == 0) {
        return 0;
    }

    if (dest_addr == src_addr) {
        return 1;
    }
    if (dest_addr < src_addr) {
        return (uintptr_t)(src_addr - dest_addr) < (uintptr_t)length;
    }
    return (uintptr_t)(dest_addr - src_addr) < (uintptr_t)length;
}

static int safe_memcpy(char *dest, size_t dest_capacity, const char *src, size_t copy_length) {
    if (dest == NULL || src == NULL) {
        return -1;
    }
    if (copy_length > dest_capacity) {
        return -1;
    }
    if (copy_length > 0) {
        if (ranges_overlap(dest, src, copy_length)) {
            return -1;
        }
        memcpy(dest, src, copy_length);
    }
    return 0;
}

static void free_partial_list(StringList *list, size_t populated) {
    if (list == NULL) {
        return;
    }
    if (list->items != NULL) {
        for (size_t j = 0; j < populated; j++) {
            free(list->items[j]);
        }
        free(list->items);
    }
    free(list);
}

char *join_tuple_with_string(char **tuple, size_t count, const char *separator) {
    size_t total_length;
    size_t separator_length;
    char *result;
    char *current_pos;
    size_t remaining;
    size_t *item_lengths;

    if (tuple == NULL || separator == NULL || count == 0) {
        return NULL;
    }

    if (bounded_strlen(separator, &separator_length) != 0) {
        return NULL;
    }

    if (count > SIZE_MAX / sizeof(size_t)) {
        return NULL;
    }

    item_lengths = (size_t *)calloc(count, sizeof(size_t));
    if (item_lengths == NULL) {
        return NULL;
    }

    total_length = 1;

    for (size_t i = 0; i < count; i++) {
        if (tuple[i] == NULL) {
            free(item_lengths);
            return NULL;
        }
        if (bounded_strlen(tuple[i], &item_lengths[i]) != 0) {
            free(item_lengths);
            return NULL;
        }
        if (item_lengths[i] > SIZE_MAX - separator_length) {
            free(item_lengths);
            return NULL;
        }
        if (total_length > SIZE_MAX - (item_lengths[i] + separator_length)) {
            free(item_lengths);
            return NULL;
        }
        total_length += item_lengths[i] + separator_length;
    }

    result = (char *)malloc(total_length);
    if (result == NULL) {
        free(item_lengths);
        return NULL;
    }

    current_pos = result;
    remaining = total_length - 1;

    for (size_t i = 0; i < count; i++) {
        if (safe_memcpy(current_pos, remaining, tuple[i], item_lengths[i]) != 0) {
            free(result);
            free(item_lengths);
            return NULL;
        }
        current_pos += item_lengths[i];
        remaining -= item_lengths[i];

        if (safe_memcpy(current_pos, remaining, separator, separator_length) != 0) {
            free(result);
            free(item_lengths);
            return NULL;
        }
        current_pos += separator_length;
        remaining -= separator_length;
    }

    *current_pos = '\0';
    free(item_lengths);
    return result;
}

StringList *tuple_to_list_with_string(char **tuple, size_t count, const char *separator) {
    StringList *list;
    size_t separator_length;

    if (tuple == NULL || separator == NULL || count == 0) {
        return NULL;
    }

    if (bounded_strlen(separator, &separator_length) != 0) {
        return NULL;
    }

    if (count > SIZE_MAX / 2) {
        return NULL;
    }

    list = (StringList *)malloc(sizeof(StringList));
    if (list == NULL) {
        return NULL;
    }

    list->count = count * 2;
    list->items = NULL;

    if (list->count > SIZE_MAX / sizeof(char *)) {
        free(list);
        return NULL;
    }

    list->items = (char **)calloc(list->count, sizeof(char *));
    if (list->items == NULL) {
        free(list);
        return NULL;
    }

    for (size_t i = 0; i < count; i++) {
        size_t item_length;
        size_t item_alloc_size;
        size_t separator_alloc_size;

        if (tuple[i] == NULL || bounded_strlen(tuple[i], &item_length) != 0) {
            free_partial_list(list, i * 2);
            return NULL;
        }

        if (item_length == SIZE_MAX) {
            free_partial_list(list, i * 2);
            return NULL;
        }
        item_alloc_size = item_length + 1;
        list->items[i * 2] = (char *)malloc(item_alloc_size);
        if (list->items[i * 2] == NULL) {
            free_partial_list(list, i * 2);
            return NULL;
        }
        if (safe_memcpy(list->items[i * 2], item_alloc_size - 1, tuple[i], item_length) != 0) {
            free_partial_list(list, i * 2 + 1);
            return NULL;
        }
        list->items[i * 2][item_length] = '\0';

        if (separator_length == SIZE_MAX) {
            free_partial_list(list, i * 2 + 1);
            return NULL;
        }
        separator_alloc_size = separator_length + 1;
        list->items[i * 2 + 1] = (char *)malloc(separator_alloc_size);
        if (list->items[i * 2 + 1] == NULL) {
            free_partial_list(list, i * 2 + 1);
            return NULL;
        }
        if (safe_memcpy(list->items[i * 2 + 1], separator_alloc_size - 1, separator, separator_length) != 0) {
            free_partial_list(list, i * 2 + 2);
            return NULL;
        }
        list->items[i * 2 + 1][separator_length] = '\0';
    }

    return list;
}

void free_string_list(StringList *list) {
    if (list == NULL) {
        return;
    }

    if (list->items != NULL) {
        for (size_t i = 0; i < list->count; i++) {
            free(list->items[i]);
        }
        free(list->items);
    }

    free(list);
}

int main(void) {
    char *tuple[] = {"apple", "banana", "cherry"};
    size_t tuple_count = sizeof(tuple) / sizeof(tuple[0]);
    const char *separator = "fruit";
    StringList *result;

    result = tuple_to_list_with_string(tuple, tuple_count, separator);
    if (result == NULL) {
        fprintf(stderr, "Error: Failed to convert tuple to list\n");
        return EXIT_FAILURE;
    }

    printf("[");
    for (size_t i = 0; i < result->count; i++) {
        printf("\"%s\"", result->items[i]);
        if (i < result->count - 1) {
            printf(", ");
        }
    }
    printf("]\n");

    free_string_list(result);

    return EXIT_SUCCESS;
}
