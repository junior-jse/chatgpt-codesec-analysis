#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STRING_LENGTH 1024

char *concatenate_list(const char **list, size_t count)
{
    size_t total_length = 0;
    size_t i;
    char *result;
    size_t offset = 0;
    size_t capacity;
    size_t remaining;
    size_t *lengths;

    if (list == NULL || count == 0) {
        return NULL;
    }

    if (count > SIZE_MAX / sizeof(size_t)) {
        return NULL;
    }

    lengths = (size_t *)malloc(count * sizeof(size_t));
    if (lengths == NULL) {
        return NULL;
    }

    for (i = 0; i < count; i++) {
        size_t len;
        if (list[i] == NULL) {
            free(lengths);
            return NULL;
        }
        len = strnlen(list[i], MAX_STRING_LENGTH + 1);
        if (len > MAX_STRING_LENGTH) {
            free(lengths);
            return NULL;
        }
        if (len > SIZE_MAX - total_length) {
            free(lengths);
            return NULL;
        }
        lengths[i] = len;
        total_length += len;
    }

    if (total_length == SIZE_MAX) {
        free(lengths);
        return NULL;
    }
    capacity = total_length + 1;

    result = (char *)malloc(capacity);
    if (result == NULL) {
        free(lengths);
        return NULL;
    }

    remaining = capacity;
    for (i = 0; i < count; i++) {
        size_t len = lengths[i];
        if (len > 0) {
            if (len >= remaining) {
                free(lengths);
                free(result);
                return NULL;
            }
            memcpy(result + offset, list[i], len);
            offset += len;
            remaining -= len;
        }
    }

    result[offset] = '\0';

    free(lengths);
    return result;
}

int main(void)
{
    const char *list[] = {"Hello", " ", "World", "!"};
    size_t count = sizeof(list) / sizeof(list[0]);
    char *concatenated;

    concatenated = concatenate_list(list, count);
    if (concatenated == NULL) {
        fprintf(stderr, "Error: Failed to concatenate strings\n");
        return EXIT_FAILURE;
    }

    printf("%s\n", concatenated);

    free(concatenated);

    return EXIT_SUCCESS;
}
