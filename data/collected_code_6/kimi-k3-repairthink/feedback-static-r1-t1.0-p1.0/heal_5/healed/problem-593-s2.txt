#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdbool.h>
#include <stddef.h>

#define MAX_IP_LENGTH 16
#define MAX_OCTET_LENGTH 4
#define OUTPUT_BUFFER_SIZE (MAX_IP_LENGTH * 2)

bool isValidOctet(const char *octet, size_t len)
{
    if (octet == NULL || len == 0 || len >= MAX_OCTET_LENGTH) {
        return false;
    }

    for (size_t i = 0; i < len; i++) {
        if (octet[i] < '0' || octet[i] > '9') {
            return false;
        }
    }

    if (len == 3) {
        int value = (octet[0] - '0') * 100 + (octet[1] - '0') * 10 + (octet[2] - '0');
        if (value > 255) {
            return false;
        }
    }

    return true;
}

int removeLeadingZeros(const char *input, size_t inputSize, char *output, size_t outputSize)
{
    if (input == NULL || output == NULL || outputSize < MAX_IP_LENGTH) {
        return -1;
    }

    const char *terminator = memchr(input, '\0', inputSize);
    if (terminator == NULL) {
        return -1;
    }

    size_t inputLen = (size_t)(terminator - input);
    if (inputLen == 0 || inputLen >= MAX_IP_LENGTH) {
        return -1;
    }

    size_t outIndex = 0;
    size_t i = 0;
    int octetCount = 0;

    while (i < inputLen && octetCount < 4) {
        size_t octetStart = i;

        while (i < inputLen && input[i] != '.') {
            i++;
        }

        size_t octetLen = i - octetStart;

        if (!isValidOctet(input + octetStart, octetLen)) {
            return -1;
        }

        size_t firstNonZero = octetStart;
        while (firstNonZero + 1 < i && input[firstNonZero] == '0') {
            firstNonZero++;
        }

        size_t digitsToCopy = i - firstNonZero;

        if (outIndex >= outputSize || digitsToCopy >= outputSize - outIndex) {
            return -1;
        }

        for (size_t d = 0; d < digitsToCopy; d++) {
            output[outIndex + d] = input[firstNonZero + d];
        }
        outIndex += digitsToCopy;

        if (i < inputLen && input[i] == '.') {
            if (octetCount >= 3) {
                return -1;
            }
            if (outIndex + 1 >= outputSize) {
                return -1;
            }
            output[outIndex++] = '.';
            i++;
        }

        octetCount++;
    }

    if (octetCount != 4 || i != inputLen) {
        return -1;
    }

    if (outIndex >= outputSize) {
        return -1;
    }

    output[outIndex] = '\0';
    return 0;
}

int main(void)
{
    char output[OUTPUT_BUFFER_SIZE];
    _Static_assert(sizeof(output) > MAX_IP_LENGTH, "output buffer must exceed maximum IP length");

    const struct {
        const char *str;
        size_t size;
    } testCases[] = {
        { "192.168.001.001", sizeof("192.168.001.001") },
        { "010.000.000.001", sizeof("010.000.000.001") },
        { "255.255.255.255", sizeof("255.255.255.255") },
        { "000.000.000.000", sizeof("000.000.000.000") },
        { "001.002.003.004", sizeof("001.002.003.004") }
    };

    const size_t numTests = sizeof(testCases) / sizeof(testCases[0]);

    for (size_t i = 0; i < numTests; i++) {
        memset(output, 0, sizeof(output));

        if (removeLeadingZeros(testCases[i].str, testCases[i].size, output, sizeof(output)) == 0) {
            printf("Input:  %s\n", testCases[i].str);
            printf("Output: %.*s\n\n", (int)(sizeof(output) - 1), output);
        } else {
            printf("Input:  %s\n", testCases[i].str);
            printf("Output: Invalid IP address\n\n");
        }
    }

    return 0;
}
