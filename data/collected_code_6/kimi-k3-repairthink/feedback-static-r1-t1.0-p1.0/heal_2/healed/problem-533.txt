```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdbool.h>

#define MAX_STRING_LENGTH 1024

typedef enum {
    TYPE_INT,
    TYPE_FLOAT,
    TYPE_STRING,
    TYPE_CHAR,
    TYPE_DOUBLE
} DataType;

typedef struct {
    DataType type;
    union {
        int i_val;
        float f_val;
        char *s_val;
        char c_val;
        double d_val;
    } data;
} TupleElement;

typedef struct {
    TupleElement *elements;
    size_t count;
    size_t capacity;
} Tuple;

void init_tuple(Tuple *t, size_t capacity) {
    t->elements = malloc(capacity * sizeof(TupleElement));
    if (t->elements == NULL) {
        t->count = 0;
        t->capacity = 0;
        return;
    }
    t->count = 0;
    t->capacity = capacity;
}

void free_tuple(Tuple *t) {
    for (size_t i = 0; i < t->count; i++) {
        if (t->elements[i].type == TYPE_STRING) {
            free(t->elements[i].data.s_val);
            t->elements[i].data.s_val = NULL;
        }
    }
    free(t->elements);
    t->elements = NULL;
    t->count = 0;
    t->capacity = 0;
}

bool add_int(Tuple *t, int val) {
    if (t->count >= t->capacity) return false;
    t->elements[t->count].type = TYPE_INT;
    t->elements[t->count].data.i_val = val;
    t->count++;
    return true;
}

bool add_float(Tuple *t, float val) {
    if (t->count >= t->capacity) return false;
    t->elements[t->count].type = TYPE_FLOAT;
    t->elements[t->count].data.f_val = val;
    t->count++;
    return true;
}

bool add_string(Tuple *t, const char *val) {
    if (t->count >= t->capacity) return false;
    if (val == NULL) return false;
    size_t len = strnlen(val, MAX_STRING_LENGTH);
    if (len == MAX_STRING_LENGTH) return false;
    char *copy = malloc(len + 1);
    if (copy == NULL) return false;
    memcpy(copy, val, len + 1);
    t->elements[t->count].type = TYPE_STRING;
    t->elements[t->count].data.s_val = copy;
    t->count++;
    return true;
}

bool add_char(Tuple *t, char val) {
    if (t->count >= t->capacity) return false;
    t->elements[t->count].type = TYPE_CHAR;
    t->elements[t->count].data.c_val = val;
    t->count++;
    return true;
}

bool add_double(Tuple *t, double val) {
    if (t->count >= t->capacity) return false;
    t->elements[t->count].type = TYPE_DOUBLE;
    t->elements[t->count].data.d_val = val;
    t->count++;
    return true;
}

void remove_by_type(Tuple *t, DataType type) {
    size_t write_idx = 0;
    for (size_t read_idx = 0; read_idx < t->count; read_idx++) {
        if (t->elements[read_idx].type != type) {
            if (write_idx != read_idx) {
                t->elements[write_idx] = t->elements[read_idx];
            }
            write_idx++;
        } else {
            if (t->elements[read_idx].type == TYPE_STRING) {
                free(t->elements[read_idx].data.s_val);
                t->elements[read_idx].data.s_val = NULL;
            }
        }
    }
    t->count = write_idx;
}

void print_tuple(const Tuple *t) {
    printf("(");
    for (size_t i = 0; i < t->count; i++) {
        switch (t->elements[i].type) {
            case TYPE_INT:
                printf("%d", t->elements[i].data.i_val);
                break;
            case TYPE_FLOAT:
                printf("%.2f", t->elements[i].data.f_val);
                break;
            case TYPE_STRING:
                printf("\"%s\"", t->elements[i].data.s_val);
                break;
            case TYPE_CHAR:
                printf("'%c'", t->elements[i].data.c_val);
                break;
            case TYPE_DOUBLE:
                printf("%.2f", t->elements[i].data.d_val);
                break;
        }
        if (i < t->count - 1) {
            printf(", ");
        }
    }
    printf(")\n");
}

int main(void) {
    Tuple t;
    init_tuple(&t, 20);

    add_int(&t, 42);
    add_string(&t, "hello");
    add_float(&t, 3.14f);
    add_int(&t, 100);
    add_char(&t, 'A');
    add_string(&t, "world");
    add_double(&t, 2.718);
    add_int(&t, 7);
    add_float(&t, 1.5f);

    printf("Original tuple: ");
    print_tuple(&t);

    remove_by_type(&t, TYPE_INT);
    printf("After removing ints: ");
    print_tuple(&t);

    remove_by_type(&t, TYPE_STRING);
    printf("After removing strings: ");
    print_tuple(&t);

    free_tuple(&t);
    return 0;
}
```

The fix applied in `add_string`: instead of `memcpy(copy, val, len)` followed by a separate null-termination write, the copy now uses `memcpy(copy, val, len + 1)`. Since `len` is validated via `strnlen` (guaranteeing a null terminator exists within the first `MAX_STRING_LENGTH` bytes of `val`) and the destination buffer is allocated with exactly `len + 1` bytes, the copy is provably bounded — the destination always holds the full source including its null terminator, addressing the CWE-120 concern.
