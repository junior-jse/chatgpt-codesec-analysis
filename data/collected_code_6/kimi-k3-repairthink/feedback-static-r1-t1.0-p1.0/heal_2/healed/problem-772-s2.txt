#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <errno.h>
#include <stdint.h>
#include <sys/types.h>

void remove_words_of_length(char *str, size_t k, size_t max_len)
{
    if (str == NULL || max_len == 0) {
        return;
    }

    size_t read_idx = 0;
    size_t write_idx = 0;
    size_t str_len = strnlen(str, max_len);

    while (read_idx < str_len) {
        while (read_idx < str_len && isspace((unsigned char)str[read_idx])) {
            str[write_idx++] = str[read_idx++];
        }

        if (read_idx >= str_len) {
            break;
        }

        size_t word_start = read_idx;
        while (read_idx < str_len && !isspace((unsigned char)str[read_idx])) {
            read_idx++;
        }
        size_t word_len = read_idx - word_start;

        if (word_len != k) {
            if (write_idx > 0 && write_idx < str_len && !isspace((unsigned char)str[write_idx - 1])) {
                str[write_idx++] = ' ';
            }
            memmove(&str[write_idx], &str[word_start], word_len);
            write_idx += word_len;
        }
    }

    if (write_idx < max_len) {
        str[write_idx] = '\0';
    } else {
        str[max_len - 1] = '\0';
    }
}

int main(void)
{
    char *buffer = NULL;
    size_t buffer_size = 0;
    ssize_t len;

    printf("Enter a string: ");
    len = getline(&buffer, &buffer_size, stdin);
    if (len < 0) {
        fprintf(stderr, "Error reading input\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
        len--;
    }

    char *kbuf = NULL;
    size_t kbuf_size = 0;
    size_t k = 0;

    printf("Enter word length to remove: ");
    if (getline(&kbuf, &kbuf_size, stdin) < 0) {
        fprintf(stderr, "Error reading input\n");
        free(buffer);
        free(kbuf);
        return EXIT_FAILURE;
    }

    errno = 0;
    char *endptr = NULL;
    unsigned long long k_val = strtoull(kbuf, &endptr, 10);
    int valid = (endptr != kbuf);
    if (valid) {
        while (isspace((unsigned char)*endptr)) {
            endptr++;
        }
        valid = (errno == 0 && *endptr == '\0' && k_val <= SIZE_MAX);
    }

    free(kbuf);

    if (!valid) {
        fprintf(stderr, "Invalid input\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    k = (size_t)k_val;

    remove_words_of_length(buffer, k, (size_t)len + 1);

    printf("Result: %s\n", buffer);

    free(buffer);
    return EXIT_SUCCESS;
}
