#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/types.h>

static void reverse_segment(char *start, char *end)
{
    while (start < end) {
        char temp = *start;
        *start = *end;
        *end = temp;
        start++;
        end--;
    }
}

int reverse_words(char *str, size_t max_len)
{
    if (str == NULL || max_len == 0) {
        return -1;
    }

    size_t len = strnlen(str, max_len);
    if (len == 0) {
        return 0;
    }

    reverse_segment(str, str + len - 1);

    size_t word_start = 0;

    for (size_t i = 0; i < len; i++) {
        if (str[i] == ' ') {
            if (i > word_start) {
                reverse_segment(str + word_start, str + i - 1);
            }
            word_start = i + 1;
        }
    }
    if (len > word_start) {
        reverse_segment(str + word_start, str + len - 1);
    }

    return 0;
}

int main(void)
{
    char *buffer = NULL;
    size_t buffer_size = 0;
    ssize_t read;

    printf("Enter a string: ");
    read = getline(&buffer, &buffer_size, stdin);
    if (read < 0) {
        fprintf(stderr, "Error reading input\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    if (read > 0 && buffer[read - 1] == '\n') {
        buffer[read - 1] = '\0';
    }

    if (reverse_words(buffer, buffer_size) != 0) {
        fprintf(stderr, "Error processing string\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    printf("Reversed words: %s\n", buffer);

    free(buffer);
    return EXIT_SUCCESS;
}

**Key fixes applied:**

1. **Eliminated the statically-sized array (CWE-119/CWE-120):** Replaced `char buffer[256]` with a dynamically-allocated buffer via `getline()`, which automatically resizes to accommodate input of any length. This removes the fixed-size restriction entirely rather than just working around it.

2. **Proper length-limited handling:** `getline()` is bounds-safe by design (it manages its own memory), and `strnlen()` is retained inside `reverse_words()` as a defense-in-depth measure so the string is never scanned past the allocated capacity (`buffer_size`).

3. **Correct newline handling:** The newline is stripped using the actual number of bytes read (`read`), which is exact, rather than re-scanning the buffer.

4. **Memory hygiene:** The buffer is freed on every exit path (input error, processing error, and success), preventing leaks.

5. **Robust error checking:** The return value of `getline()` is checked with `ssize_t` (requiring `<sys/types.h>`), correctly handling EOF and read errors.
