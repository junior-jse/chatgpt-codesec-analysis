```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define DATE_PATTERN "[0-9]{4}-[0-9]{2}-[0-9]{2}"
#define DATE_STR_SIZE 16

int extract_date_from_url(const char *url, int *year, int *month, int *day)
{
    regex_t regex;
    regmatch_t matches[1];
    char date_str[DATE_STR_SIZE];
    int ret;

    if (url == NULL || year == NULL || month == NULL || day == NULL)
    {
        return -1;
    }

    ret = regcomp(&regex, DATE_PATTERN, REG_EXTENDED);
    if (ret != 0)
    {
        return -1;
    }

    ret = regexec(&regex, url, 1, matches, 0);
    if (ret != 0)
    {
        regfree(&regex);
        return -1;
    }

    if (matches[0].rm_so < 0 || matches[0].rm_eo < 0 ||
        matches[0].rm_eo <= matches[0].rm_so)
    {
        regfree(&regex);
        return -1;
    }

    size_t len = (size_t)(matches[0].rm_eo - matches[0].rm_so);
    if (len != 10 || len >= sizeof(date_str))
    {
        regfree(&regex);
        return -1;
    }

    if (snprintf(date_str, sizeof(date_str), "%.*s", (int)len,
                 url + matches[0].rm_so) < 0)
    {
        regfree(&regex);
        return -1;
    }

    if (sscanf(date_str, "%4d-%2d-%2d", year, month, day) != 3)
    {
        regfree(&regex);
        return -1;
    }

    if (*year < 1900 || *year > 2100 || *month < 1 || *month > 12 ||
        *day < 1 || *day > 31)
    {
        regfree(&regex);
        return -1;
    }

    regfree(&regex);
    return 0;
}

int main(void)
{
    const char *url = "https://example.com/2024-01-15/article.html";
    int year = 0;
    int month = 0;
    int day = 0;

    if (extract_date_from_url(url, &year, &month, &day) == 0)
    {
        printf("Year: %d\n", year);
        printf("Month: %d\n", month);
        printf("Day: %d\n", day);
    }
    else
    {
        fprintf(stderr, "Failed to extract date from URL\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

Key fixes applied:

1. **Statically-sized array (CWE-119/CWE-120):** The buffer size is now defined via the named constant `DATE_STR_SIZE` (16 bytes), providing headroom beyond the exact 11 bytes needed, and the explicit length check `len >= sizeof(date_str)` is retained before any copy occurs.

2. **`memcpy` overflow risk (CWE-120):** `memcpy` was replaced with `snprintf` using a precision-limited format (`%.*s`) and the destination size (`sizeof(date_str)`). This guarantees the copy is bounded and the result is always null-terminated, eliminating any possibility of a buffer overflow.

3. **Additional hardening:** Added validation that `rm_eo > rm_so` before computing the match length, preventing a zero-length or negative-length subtraction edge case.
