#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_STRING_LEN 1024

char *snake_to_camel(const char *input) {
    char *result;
    size_t input_len;
    size_t i;
    size_t j = 0;
    int capitalize_next = 0;

    if (input == NULL) {
        return NULL;
    }

    input_len = strnlen(input, MAX_STRING_LEN);
    if (input_len >= MAX_STRING_LEN) {
        return NULL;
    }

    result = malloc(input_len + 1);
    if (result == NULL) {
        return NULL;
    }

    for (i = 0; i < input_len; i++) {
        unsigned char c = (unsigned char)input[i];

        if (c == '_' && i + 1 < input_len &&
            input[i + 1] >= 'a' && input[i + 1] <= 'z') {
            capitalize_next = 1;
            continue;
        }

        if (capitalize_next) {
            result[j++] = (char)toupper(c);
            capitalize_next = 0;
        } else {
            result[j++] = (char)c;
        }
    }

    result[j] = '\0';
    return result;
}

int main(void) {
    const char *test_cases[] = {
        "hello_world",
        "snake_case_string_example",
        "single",
        "multiple__underscores",
        NULL
    };
    int i;

    for (i = 0; test_cases[i] != NULL; i++) {
        char *camel = snake_to_camel(test_cases[i]);
        if (camel != NULL) {
            printf("Input:  %s\n", test_cases[i]);
            printf("Output: %s\n\n", camel);
            free(camel);
        } else {
            printf("Conversion failed for: %s\n\n", test_cases[i]);
        }
    }

    return 0;
}
