#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define DICT_MAX_STRING_LEN 4096U

typedef struct {
    char *key;
    char *value;
} DictEntry;

typedef struct {
    DictEntry *entries;
    size_t size;
    size_t capacity;
} Dictionary;

static size_t dict_bounded_strlen(const char *s, size_t limit) {
    size_t len = 0;

    while (len < limit && s[len] != '\0') {
        len++;
    }

    return len;
}

static char *dict_safe_strdup(const char *src) {
    size_t len;
    char *copy;

    if (src == NULL) {
        return NULL;
    }

    len = dict_bounded_strlen(src, DICT_MAX_STRING_LEN + 1);
    if (len > DICT_MAX_STRING_LEN) {
        return NULL;
    }

    copy = (char *)malloc(len + 1);
    if (copy == NULL) {
        return NULL;
    }

    memcpy(copy, src, len + 1);

    return copy;
}

int dict_init(Dictionary *dict, size_t capacity) {
    if (dict == NULL) {
        return -1;
    }
    dict->entries = NULL;
    dict->size = 0;
    dict->capacity = 0;

    if (capacity > 0) {
        if (capacity > SIZE_MAX / sizeof(DictEntry)) {
            return -1;
        }
        dict->entries = (DictEntry *)calloc(capacity, sizeof(DictEntry));
        if (dict->entries == NULL) {
            return -1;
        }
        dict->capacity = capacity;
    }
    return 0;
}

int dict_set(Dictionary *dict, size_t index, const char *key, const char *value) {
    char *new_key = NULL;
    char *new_value = NULL;

    if (dict == NULL || dict->entries == NULL || index >= dict->capacity) {
        return -1;
    }

    if (key != NULL) {
        new_key = dict_safe_strdup(key);
        if (new_key == NULL) {
            return -1;
        }
    }

    if (value != NULL) {
        new_value = dict_safe_strdup(value);
        if (new_value == NULL) {
            free(new_key);
            return -1;
        }
    }

    free(dict->entries[index].key);
    free(dict->entries[index].value);

    dict->entries[index].key = new_key;
    dict->entries[index].value = new_value;

    if (index >= dict->size) {
        dict->size = index + 1;
    }

    return 0;
}

void dict_free(Dictionary *dict) {
    if (dict != NULL && dict->entries != NULL) {
        for (size_t i = 0; i < dict->capacity; i++) {
            free(dict->entries[i].key);
            free(dict->entries[i].value);
        }
        free(dict->entries);
        dict->entries = NULL;
        dict->size = 0;
        dict->capacity = 0;
    }
}

int dict_is_empty(const Dictionary *dict) {
    if (dict == NULL) {
        return 1;
    }
    return dict->size == 0;
}

int all_dicts_empty(const Dictionary *dicts, size_t count) {
    if (dicts == NULL || count == 0) {
        return 1;
    }

    for (size_t i = 0; i < count; i++) {
        if (!dict_is_empty(&dicts[i])) {
            return 0;
        }
    }
    return 1;
}

int main(void) {
    Dictionary *dict_list = NULL;
    const size_t dict_count = 3;
    int result;

    dict_list = (Dictionary *)calloc(dict_count, sizeof(Dictionary));
    if (dict_list == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    for (size_t i = 0; i < dict_count; i++) {
        if (dict_init(&dict_list[i], 10) != 0) {
            for (size_t j = 0; j < i; j++) {
                dict_free(&dict_list[j]);
            }
            free(dict_list);
            fprintf(stderr, "Dictionary initialization failed\n");
            return EXIT_FAILURE;
        }
    }

    result = all_dicts_empty(dict_list, dict_count);
    printf("All dictionaries empty: %s\n", result ? "true" : "false");

    if (dict_set(&dict_list[1], 0, "test", "") != 0) {
        for (size_t i = 0; i < dict_count; i++) {
            dict_free(&dict_list[i]);
        }
        free(dict_list);
        fprintf(stderr, "Failed to set dictionary entry\n");
        return EXIT_FAILURE;
    }

    result = all_dicts_empty(dict_list, dict_count);
    printf("All dictionaries empty: %s\n", result ? "true" : "false");

    for (size_t i = 0; i < dict_count; i++) {
        dict_free(&dict_list[i]);
    }
    free(dict_list);

    return EXIT_SUCCESS;
}
