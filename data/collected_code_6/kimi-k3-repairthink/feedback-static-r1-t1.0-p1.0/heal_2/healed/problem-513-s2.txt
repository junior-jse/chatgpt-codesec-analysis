```c
#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
#include <limits.h>

#define MAX_STRING_LENGTH 4096

static char* duplicate_string(const char* src) {
    size_t len = strnlen(src, MAX_STRING_LENGTH);
    if (len >= MAX_STRING_LENGTH) {
        return NULL;
    }
    size_t buf_size = len + 1;
    char* copy = malloc(buf_size);
    if (copy == NULL) {
        return NULL;
    }
    int written = snprintf(copy, buf_size, "%s", src);
    if (written < 0 || (size_t)written >= buf_size) {
        free(copy);
        return NULL;
    }
    return copy;
}

char** convert_tuple_to_list(char** tuple, int size, const char* str) {
    if (tuple == NULL || str == NULL || size <= 0 || size > INT_MAX / 2) {
        return NULL;
    }

    int new_size = size * 2;
    if ((size_t)new_size > SIZE_MAX / sizeof(char*)) {
        return NULL;
    }

    char** result = malloc((size_t)new_size * sizeof(char*));
    if (result == NULL) {
        return NULL;
    }

    for (int i = 0; i < size; i++) {
        if (tuple[i] == NULL) {
            for (int j = 0; j < i * 2; j++) {
                free(result[j]);
            }
            free(result);
            return NULL;
        }

        result[i * 2] = duplicate_string(tuple[i]);
        if (result[i * 2] == NULL) {
            for (int j = 0; j < i * 2; j++) {
                free(result[j]);
            }
            free(result);
            return NULL;
        }

        result[i * 2 + 1] = duplicate_string(str);
        if (result[i * 2 + 1] == NULL) {
            for (int j = 0; j <= i * 2; j++) {
                free(result[j]);
            }
            free(result);
            return NULL;
        }
    }

    return result;
}

void free_list(char** list, int size) {
    if (list == NULL) {
        return;
    }
    for (int i = 0; i < size; i++) {
        free(list[i]);
    }
    free(list);
}

int main(void) {
    char* tuple[] = {"apple", "banana", "cherry"};
    int tuple_size = 3;
    const char* separator = "fruit";

    char** result = convert_tuple_to_list(tuple, tuple_size, separator);
    if (result == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    int result_size = tuple_size * 2;
    for (int i = 0; i < result_size; i++) {
        printf("%s\n", result[i]);
    }

    free_list(result, result_size);

    return EXIT_SUCCESS;
}
```

Key change: the flagged `memcpy` in `duplicate_string` was replaced with a bounded `snprintf` call using the exact allocated buffer size (`len + 1`), with the return value validated to detect truncation or encoding errors. This remediates the CWE-120 buffer-copy concern while preserving the original behavior. The `_POSIX_C_SOURCE` feature-test macro was also added so `strnlen` is properly declared by `string.h`.
