#include <stdio.h>
#include <string.h>
#include <stdlib.h>

void removeLeadingZeros(char *ipAddress, size_t ipAddressSize) {
    char *token;
    char *saveptr;
    char *result;
    size_t resultSize;
    size_t resultLen = 0;
    size_t ipLen;

    if (ipAddress == NULL || ipAddressSize == 0) {
        return;
    }

    ipAddress[ipAddressSize - 1] = '\0';
    ipLen = strnlen(ipAddress, ipAddressSize);

    resultSize = ipLen + 1;
    result = (char *)malloc(resultSize);
    if (result == NULL) {
        return;
    }
    result[0] = '\0';

    token = strtok_r(ipAddress, ".", &saveptr);
    while (token != NULL) {
        char *temp = token;
        size_t tokenLen;
        size_t maxTokenLen;

        while (*temp == '0') {
            temp++;
        }
        if (*temp == '\0') {
            temp--;
        }

        maxTokenLen = ipAddressSize - (size_t)(temp - ipAddress);
        tokenLen = strnlen(temp, maxTokenLen);

        if (resultLen + tokenLen + 2 > resultSize) {
            break;
        }

        memcpy(result + resultLen, temp, tokenLen);
        resultLen += tokenLen;
        result[resultLen] = '.';
        resultLen++;
        result[resultLen] = '\0';

        token = strtok_r(NULL, ".", &saveptr);
    }

    if (resultLen > 0) {
        result[resultLen - 1] = '\0';
    }

    snprintf(ipAddress, ipAddressSize, "%s", result);
    free(result);
}

int main(void) {
    char ipAddress[100];
    size_t inputLen;

    printf("Enter an IP address: ");
    if (fgets(ipAddress, sizeof(ipAddress), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return 1;
    }

    inputLen = strnlen(ipAddress, sizeof(ipAddress));
    if (inputLen > 0 && ipAddress[inputLen - 1] == '\n') {
        ipAddress[inputLen - 1] = '\0';
    }

    removeLeadingZeros(ipAddress, sizeof(ipAddress));

    printf("IP address without leading zeros: %s\n", ipAddress);

    return 0;
}
