#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <stdbool.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>
#include <sys/types.h>

#define MIN_LENGTH 8
#define MAX_LENGTH 128

bool isValidPassword(const char *password) {
    if (password == NULL) {
        return false;
    }

    size_t length = strnlen(password, MAX_LENGTH + 1);

    if (length < MIN_LENGTH || length > MAX_LENGTH) {
        return false;
    }

    bool hasUpper = false;
    bool hasLower = false;
    bool hasDigit = false;
    bool hasSpecial = false;

    for (size_t i = 0; i < length; i++) {
        unsigned char c = (unsigned char)password[i];

        if (isupper(c)) {
            hasUpper = true;
        } else if (islower(c)) {
            hasLower = true;
        } else if (isdigit(c)) {
            hasDigit = true;
        } else if (ispunct(c)) {
            hasSpecial = true;
        } else if (isspace(c)) {
            return false;
        }
    }

    return hasUpper && hasLower && hasDigit && hasSpecial;
}

static void secureClear(void *buffer, size_t size) {
    volatile unsigned char *p = (volatile unsigned char *)buffer;

    while (size-- > 0) {
        *p++ = 0;
    }
}

int main(void) {
    char *password = NULL;
    size_t bufferSize = 0;
    ssize_t bytesRead;

    printf("Enter password: ");
    fflush(stdout);

    bytesRead = getline(&password, &bufferSize, stdin);

    if (bytesRead < 0) {
        fprintf(stderr, "Error reading input\n");
        free(password);
        return EXIT_FAILURE;
    }

    if (bytesRead > 0 && password[bytesRead - 1] == '\n') {
        password[bytesRead - 1] = '\0';
        bytesRead--;
    }

    if (bytesRead > MAX_LENGTH) {
        fprintf(stderr, "Password too long\n");
        secureClear(password, bufferSize);
        free(password);
        return EXIT_FAILURE;
    }

    bool valid = isValidPassword(password);

    secureClear(password, bufferSize);
    free(password);

    if (valid) {
        printf("Password is valid\n");
        return EXIT_SUCCESS;
    }

    printf("Password is invalid\n");
    return EXIT_FAILURE;
}
