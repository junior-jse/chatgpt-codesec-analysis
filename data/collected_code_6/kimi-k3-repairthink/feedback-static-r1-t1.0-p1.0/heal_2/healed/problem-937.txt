```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>
#include <stdint.h>

char find_most_common_char(const char *str, int *count)
{
    int freq[UCHAR_MAX + 1] = {0};
    int max_count = 0;
    char most_common = '\0';
    size_t i;

    if (str == NULL || count == NULL)
    {
        return '\0';
    }

    for (i = 0; str[i] != '\0'; i++)
    {
        unsigned char c = (unsigned char)str[i];
        if (freq[c] < INT_MAX)
        {
            freq[c]++;
        }
    }

    for (i = 0; i <= UCHAR_MAX; i++)
    {
        if (freq[i] > max_count)
        {
            max_count = freq[i];
            most_common = (char)i;
        }
    }

    *count = max_count;
    return most_common;
}

static char *read_line(FILE *stream)
{
    size_t capacity = 128;
    size_t length = 0;
    char *buffer;
    int ch;

    if (stream == NULL)
    {
        return NULL;
    }

    buffer = malloc(capacity);
    if (buffer == NULL)
    {
        return NULL;
    }

    while ((ch = fgetc(stream)) != EOF && ch != '\n')
    {
        if (length + 1 >= capacity)
        {
            size_t new_capacity;
            char *new_buffer;

            if (capacity > SIZE_MAX / 2)
            {
                free(buffer);
                return NULL;
            }

            new_capacity = capacity * 2;
            if (new_capacity <= capacity)
            {
                free(buffer);
                return NULL;
            }

            new_buffer = realloc(buffer, new_capacity);
            if (new_buffer == NULL)
            {
                free(buffer);
                return NULL;
            }

            buffer = new_buffer;
            capacity = new_capacity;
        }

        buffer[length] = (char)ch;
        length++;
    }

    if (ferror(stream) || (ch == EOF && length == 0))
    {
        free(buffer);
        return NULL;
    }

    buffer[length] = '\0';
    return buffer;
}

int main(void)
{
    char *buffer;
    int count = 0;
    char result;

    buffer = read_line(stdin);
    if (buffer == NULL)
    {
        return 1;
    }

    if (buffer[0] == '\0')
    {
        free(buffer);
        return 1;
    }

    result = find_most_common_char(buffer, &count);

    if (count > 0)
    {
        printf("Most common character: '%c'\n", result);
        printf("Occurrences: %d\n", count);
    }

    free(buffer);
    return 0;
}
```

Key hardening changes addressing the flagged `fgetc` loop weakness (CWE-120 / CWE-20):

- **Bounds are enforced before every write**: the `length + 1 >= capacity` check guarantees space for the character plus the terminating `'\0'`, so `buffer[length] = ...` and `buffer[length] = '\0'` can never write out of bounds.
- **Overflow-safe growth**: `capacity` is checked against `SIZE_MAX / 2` before doubling, and the new capacity is validated to be strictly greater than the old one, preventing wrap-around that would lead to a heap overflow.
- **Failure paths free memory and bail out** on `realloc` failure, stream errors, or allocation failure, so no stale buffer is used.
- **Counter saturation** in `find_most_common_char` (`freq[c] < INT_MAX`) prevents signed integer overflow for pathological inputs.
