#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

char **extract_strings(const char **source, size_t count, size_t extract_size, size_t *result_count)
{
    char **result;
    size_t valid_count = 0;

    if (source == NULL || result_count == NULL || extract_size == 0 || extract_size == SIZE_MAX) {
        if (result_count != NULL) {
            *result_count = 0;
        }
        return NULL;
    }

    for (size_t i = 0; i < count; i++) {
        if (source[i] != NULL && strnlen(source[i], extract_size + 1) == extract_size) {
            valid_count++;
        }
    }

    if (valid_count == 0) {
        *result_count = 0;
        return NULL;
    }

    if (valid_count > SIZE_MAX / sizeof(char *)) {
        *result_count = 0;
        return NULL;
    }

    result = malloc(valid_count * sizeof(char *));
    if (result == NULL) {
        *result_count = 0;
        return NULL;
    }

    valid_count = 0;
    for (size_t i = 0; i < count; i++) {
        if (source[i] != NULL && strnlen(source[i], extract_size + 1) == extract_size) {
            result[valid_count] = malloc(extract_size + 1);
            if (result[valid_count] == NULL) {
                for (size_t j = 0; j < valid_count; j++) {
                    free(result[j]);
                }
                free(result);
                *result_count = 0;
                return NULL;
            }
            snprintf(result[valid_count], extract_size + 1, "%s", source[i]);
            valid_count++;
        }
    }

    *result_count = valid_count;
    return result;
}

void free_extracted(char **strings, size_t count)
{
    if (strings == NULL) {
        return;
    }
    for (size_t i = 0; i < count; i++) {
        free(strings[i]);
    }
    free(strings);
}

int main(void)
{
    const char *list[] = {
        "apple",
        "banana",
        "cherry",
        "date",
        "elderberry",
        "fig",
        "grape",
        NULL
    };
    size_t count = 7;
    size_t extract_size = 5;
    size_t result_count = 0;
    char **extracted;

    extracted = extract_strings(list, count, extract_size, &result_count);

    if (extracted != NULL) {
        printf("Extracted strings of size %zu:\n", extract_size);
        for (size_t i = 0; i < result_count; i++) {
            printf("%s\n", extracted[i]);
        }
        free_extracted(extracted, result_count);
    } else {
        printf("No strings found with size %zu or an error occurred.\n", extract_size);
    }

    return 0;
}
