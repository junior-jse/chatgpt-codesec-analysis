#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int max_abs(int a, int b) {
    int abs_a = abs(a);
    int abs_b = abs(b);
    return abs_a > abs_b ? abs_a : abs_b;
}

void find_closest_triplet(int *arr1, int n1, int *arr2, int n2, int *arr3, int n3, int *res1, int *res2, int *res3) {
    int i = 0, j = 0, k = 0;
    int min_diff = INT_MAX;
    int best_i = 0, best_j = 0, best_k = 0;

    while (i < n1 && j < n2 && k < n3) {
        int min_val = arr1[i] < arr2[j] ? (arr1[i] < arr3[k] ? arr1[i] : arr3[k]) : (arr2[j] < arr3[k] ? arr2[j] : arr3[k]);
        int max_val = arr1[i] > arr2[j] ? (arr1[i] > arr3[k] ? arr1[i] : arr3[k]) : (arr2[j] > arr3[k] ? arr2[j] : arr3[k]);
        int diff = max_val - min_val;

        if (diff < min_diff) {
            min_diff = diff;
            best_i = i;
            best_j = j;
            best_k = k;
        }

        if (min_val == arr1[i]) {
            i++;
        } else if (min_val == arr2[j]) {
            j++;
        } else {
            k++;
        }
    }

    *res1 = arr1[best_i];
    *res2 = arr2[best_j];
    *res3 = arr3[best_k];
}

int main(void) {
    int arr1[] = {1, 4, 10};
    int arr2[] = {2, 15, 20};
    int arr3[] = {10, 12};

    int n1 = sizeof(arr1) / sizeof(arr1[0]);
    int n2 = sizeof(arr2) / sizeof(arr2[0]);
    int n3 = sizeof(arr3) / sizeof(arr3[0]);

    int res1, res2, res3;

    find_closest_triplet(arr1, n1, arr2, n2, arr3, n3, &res1, &res2, &res3);

    printf("%d %d %d\n", res1, res2, res3);

    return 0;
}
