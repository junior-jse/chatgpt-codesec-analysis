#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int* find_min_indices(const int *arr, size_t size, size_t *out_count) {
    if (arr == NULL || size == 0 || out_count == NULL) {
        return NULL;
    }

    int min_val = INT_MAX;
    size_t count = 0;
    size_t i;

    for (i = 0; i < size; i++) {
        if (arr[i] < min_val) {
            min_val = arr[i];
        }
    }

    for (i = 0; i < size; i++) {
        if (arr[i] == min_val) {
            count++;
        }
    }

    int *indices = (int*)malloc(count * sizeof(int));
    if (indices == NULL) {
        return NULL;
    }

    size_t idx = 0;
    for (i = 0; i < size; i++) {
        if (arr[i] == min_val) {
            indices[idx++] = (int)i;
        }
    }

    *out_count = count;
    return indices;
}

int main(void) {
    int arr[] = {5, 2, 8, 2, 9, 1, 3, 1};
    size_t size = sizeof(arr) / sizeof(arr[0]);
    size_t count = 0;

    int *indices = find_min_indices(arr, size, &count);

    if (indices != NULL) {
        printf("Minimum value indices: ");
        for (size_t i = 0; i < count; i++) {
            printf("%d", indices[i]);
            if (i < count - 1) {
                printf(", ");
            }
        }
        printf("\n");
        free(indices);
    } else {
        printf("Error: could not determine minimum indices.\n");
    }

    return 0;
}
