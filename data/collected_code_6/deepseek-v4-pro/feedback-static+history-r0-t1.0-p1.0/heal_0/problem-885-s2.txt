#include <stdio.h>
#include <stdbool.h>
#include <string.h>
#include <limits.h>

bool is_isomorphic(const char *s, const char *t) {
    if (s == NULL || t == NULL) {
        return false;
    }
    
    size_t len_s = strlen(s);
    size_t len_t = strlen(t);
    
    if (len_s != len_t) {
        return false;
    }
    
    int map_s_to_t[UCHAR_MAX + 1];
    int map_t_to_s[UCHAR_MAX + 1];
    
    for (int i = 0; i <= UCHAR_MAX; i++) {
        map_s_to_t[i] = -1;
        map_t_to_s[i] = -1;
    }
    
    for (size_t i = 0; i < len_s; i++) {
        unsigned char c1 = (unsigned char)s[i];
        unsigned char c2 = (unsigned char)t[i];
        
        if (map_s_to_t[c1] == -1 && map_t_to_s[c2] == -1) {
            map_s_to_t[c1] = c2;
            map_t_to_s[c2] = c1;
        } else if (map_s_to_t[c1] != c2 || map_t_to_s[c2] != c1) {
            return false;
        }
    }
    
    return true;
}

int main(void) {
    const char *str1 = "egg";
    const char *str2 = "add";
    const char *str3 = "foo";
    const char *str4 = "bar";
    const char *str5 = "paper";
    const char *str6 = "title";
    
    printf("\"%s\" and \"%s\" are %s\n", str1, str2, 
           is_isomorphic(str1, str2) ? "isomorphic" : "not isomorphic");
    printf("\"%s\" and \"%s\" are %s\n", str3, str4, 
           is_isomorphic(str3, str4) ? "isomorphic" : "not isomorphic");
    printf("\"%s\" and \"%s\" are %s\n", str5, str6, 
           is_isomorphic(str5, str6) ? "isomorphic" : "not isomorphic");
    
    return 0;
}
