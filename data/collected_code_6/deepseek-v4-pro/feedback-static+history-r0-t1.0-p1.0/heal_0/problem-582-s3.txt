#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    char *key;
    int value;
    struct Node *next;
} Node;

typedef struct {
    Node **buckets;
    size_t size;
} Dictionary;

Dictionary *create_dictionary(size_t size) {
    Dictionary *dict = (Dictionary *)malloc(sizeof(Dictionary));
    if (!dict) return NULL;
    
    dict->buckets = (Node **)calloc(size, sizeof(Node *));
    if (!dict->buckets) {
        free(dict);
        return NULL;
    }
    
    dict->size = size;
    return dict;
}

void destroy_dictionary(Dictionary *dict) {
    if (!dict) return;
    
    for (size_t i = 0; i < dict->size; i++) {
        Node *current = dict->buckets[i];
        while (current) {
            Node *temp = current;
            current = current->next;
            free(temp->key);
            free(temp);
        }
    }
    
    free(dict->buckets);
    free(dict);
}

int dictionary_is_empty(const Dictionary *dict) {
    if (!dict) return 1;
    
    for (size_t i = 0; i < dict->size; i++) {
        if (dict->buckets[i] != NULL) {
            return 0;
        }
    }
    
    return 1;
}

int main(void) {
    Dictionary *dict = create_dictionary(10);
    if (!dict) {
        fprintf(stderr, "Failed to create dictionary\n");
        return 1;
    }
    
    if (dictionary_is_empty(dict)) {
        printf("Dictionary is empty\n");
    } else {
        printf("Dictionary is not empty\n");
    }
    
    destroy_dictionary(dict);
    
    return 0;
}
