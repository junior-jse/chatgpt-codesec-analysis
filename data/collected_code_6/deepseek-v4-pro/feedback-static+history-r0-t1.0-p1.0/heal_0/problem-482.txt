#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

void find_sequences(const char *str) {
    regex_t regex;
    regmatch_t match;
    const char *pattern = "[A-Z][a-z]+";
    const char *cursor = str;
    int ret;

    ret = regcomp(&regex, pattern, REG_EXTENDED);
    if (ret) {
        size_t errbuf_size = regerror(ret, &regex, NULL, 0);
        char *errbuf = (char *)malloc(errbuf_size);
        if (errbuf) {
            regerror(ret, &regex, errbuf, errbuf_size);
            fprintf(stderr, "Regex compilation failed: %s\n", errbuf);
            free(errbuf);
        }
        return;
    }

    while (regexec(&regex, cursor, 1, &match, 0) == 0) {
        int start = match.rm_so;
        int end = match.rm_eo;
        char *seq = (char *)malloc(end - start + 1);
        if (!seq) {
            regfree(&regex);
            return;
        }
        strncpy(seq, cursor + start, end - start);
        seq[end - start] = '\0';
        printf("%s\n", seq);
        free(seq);
        cursor += end;
        if (*cursor == '\0') {
            break;
        }
    }

    regfree(&regex);
}

int main(int argc, char *argv[]) {
    if (argc != 2) {
        fprintf(stderr, "Usage: %s <string>\n", argv[0]);
        return EXIT_FAILURE;
    }

    find_sequences(argv[1]);
    return EXIT_SUCCESS;
}
