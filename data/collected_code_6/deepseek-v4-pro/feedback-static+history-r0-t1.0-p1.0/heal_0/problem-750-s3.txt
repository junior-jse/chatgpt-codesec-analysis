```c
#include <stdio.h>
#include <stdlib.h>

typedef struct Tuple {
    int first;
    int second;
} Tuple;

typedef struct ListNode {
    Tuple data;
    struct ListNode *next;
} ListNode;

typedef struct List {
    ListNode *head;
    ListNode *tail;
    size_t size;
} List;

List *list_create(void) {
    List *list = malloc(sizeof(*list));
    if (!list) return NULL;
    list->head = NULL;
    list->tail = NULL;
    list->size = 0;
    return list;
}

int list_append(List *list, Tuple tuple) {
    if (!list) return -1;
    
    ListNode *node = malloc(sizeof(*node));
    if (!node) return -1;
    
    node->data = tuple;
    node->next = NULL;
    
    if (list->tail) {
        list->tail->next = node;
        list->tail = node;
    } else {
        list->head = node;
        list->tail = node;
    }
    
    list->size++;
    return 0;
}

void list_destroy(List *list) {
    if (!list) return;
    
    ListNode *current = list->head;
    while (current) {
        ListNode *next = current->next;
        free(current);
        current = next;
    }
    
    free(list);
}

int main(void) {
    List *list = list_create();
    if (!list) {
        fprintf(stderr, "Failed to create list\n");
        return EXIT_FAILURE;
    }
    
    Tuple t1 = {10, 20};
    Tuple t2 = {30, 40};
    Tuple t3 = {50, 60};
    
    if (list_append(list, t1) != 0 ||
        list_append(list, t2) != 0 ||
        list_append(list, t3) != 0) {
        fprintf(stderr, "Failed to append tuple\n");
        list_destroy(list);
        return EXIT_FAILURE;
    }
    
    ListNode *current = list->head;
    while (current) {
        printf("(%d, %d)\n", current->data.first, current->data.second);
        current = current->next;
    }
    
    list_destroy(list);
    return EXIT_SUCCESS;
}
```
