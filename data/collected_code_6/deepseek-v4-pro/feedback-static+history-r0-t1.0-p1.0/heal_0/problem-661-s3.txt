#include <stdio.h>
#include <stdlib.h>

int maxSumWithoutThreeConsecutive(int arr[], int n) {
    if (n == 0) return 0;
    if (n == 1) return arr[0];
    if (n == 2) return arr[0] + arr[1];

    int *dp = (int *)malloc(n * sizeof(int));
    if (dp == NULL) {
        return -1;
    }

    dp[0] = arr[0];
    dp[1] = arr[0] + arr[1];
    dp[2] = arr[1] + arr[2];
    if (arr[0] + arr[2] > dp[2]) dp[2] = arr[0] + arr[2];
    if (arr[0] + arr[1] > dp[2]) dp[2] = arr[0] + arr[1];

    for (int i = 3; i < n; i++) {
        dp[i] = dp[i - 1];
        if (dp[i - 2] + arr[i] > dp[i]) dp[i] = dp[i - 2] + arr[i];
        if (dp[i - 3] + arr[i - 1] + arr[i] > dp[i]) dp[i] = dp[i - 3] + arr[i - 1] + arr[i];
    }

    int result = dp[n - 1];
    free(dp);
    return result;
}

int main() {
    int arr1[] = {1, 2, 3, 4, 5};
    int n1 = sizeof(arr1) / sizeof(arr1[0]);
    printf("%d\n", maxSumWithoutThreeConsecutive(arr1, n1));

    int arr2[] = {100, 100, 100, 100, 100};
    int n2 = sizeof(arr2) / sizeof(arr2[0]);
    printf("%d\n", maxSumWithoutThreeConsecutive(arr2, n2));

    int arr3[] = {5, 3, 4, 11, 2};
    int n3 = sizeof(arr3) / sizeof(arr3[0]);
    printf("%d\n", maxSumWithoutThreeConsecutive(arr3, n3));

    int arr4[] = {10};
    int n4 = sizeof(arr4) / sizeof(arr4[0]);
    printf("%d\n", maxSumWithoutThreeConsecutive(arr4, n4));

    int arr5[] = {1, 2};
    int n5 = sizeof(arr5) / sizeof(arr5[0]);
    printf("%d\n", maxSumWithoutThreeConsecutive(arr5, n5));

    return 0;
}
