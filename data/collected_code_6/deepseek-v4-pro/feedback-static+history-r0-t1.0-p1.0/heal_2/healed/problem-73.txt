#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

static char **resize_result(char **result, int *capacity, int size) {
    int new_capacity = *capacity * 2;
    char **temp = (char **)realloc(result, new_capacity * sizeof(char *));
    if (!temp) {
        return NULL;
    }
    *capacity = new_capacity;
    return temp;
}

static void cleanup(char **result, int size, regex_t *regex) {
    if (result) {
        for (int i = 0; i < size; i++) {
            free(result[i]);
        }
        free(result);
    }
    if (regex) {
        regfree(regex);
    }
}

char **split_regex(const char *str, const char *pattern, int *count) {
    regex_t regex;
    regmatch_t match;
    char **result = NULL;
    int capacity = 10;
    int size = 0;
    const char *cursor;
    const char *end;
    
    if (!str || !pattern || !count) {
        if (count) *count = 0;
        return NULL;
    }
    
    if (str[0] == '\0') {
        *count = 0;
        return NULL;
    }
    
    cursor = str;
    end = str + strlen(str);
    
    if (regcomp(&regex, pattern, REG_EXTENDED)) {
        *count = 0;
        return NULL;
    }
    
    result = (char **)malloc(capacity * sizeof(char *));
    if (!result) {
        regfree(&regex);
        *count = 0;
        return NULL;
    }
    
    while (cursor <= end && regexec(&regex, cursor, 1, &match, 0) == 0) {
        const char *token_start = cursor;
        size_t token_len;
        
        if (match.rm_so == -1) {
            token_len = end - cursor;
        } else {
            token_len = match.rm_so;
        }
        
        if (size == capacity) {
            char **temp = resize_result(result, &capacity, size);
            if (!temp) {
                cleanup(result, size, &regex);
                *count = 0;
                return NULL;
            }
            result = temp;
        }
        
        result[size] = (char *)malloc(token_len + 1);
        if (!result[size]) {
            cleanup(result, size, &regex);
            *count = 0;
            return NULL;
        }
        
        if (token_len > 0) {
            memcpy(result[size], token_start, token_len);
        }
        result[size][token_len] = '\0';
        size++;
        
        if (match.rm_so == -1) {
            break;
        }
        
        cursor += match.rm_so + match.rm_eo;
        
        if (match.rm_so == 0 && match.rm_eo == 0) {
            if (cursor < end) {
                cursor++;
            } else {
                break;
            }
        }
    }
    
    if (cursor < end) {
        size_t remaining_len = end - cursor;
        
        if (remaining_len > 0) {
            if (size == capacity) {
                char **temp = resize_result(result, &capacity, size);
                if (!temp) {
                    cleanup(result, size, &regex);
                    *count = 0;
                    return NULL;
                }
                result = temp;
            }
            
            result[size] = (char *)malloc(remaining_len + 1);
            if (!result[size]) {
                cleanup(result, size, &regex);
                *count = 0;
                return NULL;
            }
            
            memcpy(result[size], cursor, remaining_len);
            result[size][remaining_len] = '\0';
            size++;
        }
    }
    
    regfree(&regex);
    *count = size;
    return result;
}

void free_split_result(char **result, int count) {
    if (result) {
        for (int i = 0; i < count; i++) {
            free(result[i]);
        }
        free(result);
    }
}

int main() {
    const char *text = "apple,banana;cherry:date|elderberry";
    const char *pattern = "[,;:|]";
    int count = 0;
    
    char **tokens = split_regex(text, pattern, &count);
    
    if (tokens) {
        printf("Split results (%d tokens):\n", count);
        for (int i = 0; i < count; i++) {
            printf("Token %d: %s\n", i, tokens[i]);
        }
        free_split_result(tokens, count);
    } else {
        printf("Failed to split string\n");
    }
    
    return 0;
}
