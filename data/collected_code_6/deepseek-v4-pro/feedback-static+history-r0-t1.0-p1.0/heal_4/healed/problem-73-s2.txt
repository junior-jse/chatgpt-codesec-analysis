#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

static size_t safe_strlen(const char *s, size_t max_len) {
    if (!s) return 0;
    size_t len = 0;
    while (len < max_len && s[len] != '\0') {
        len++;
    }
    return len;
}

static char *safe_strndup(const char *s, size_t max_len) {
    if (!s) return NULL;
    size_t len = safe_strlen(s, max_len);
    if (len >= max_len) return NULL;
    char *dup = malloc(len + 1);
    if (!dup) return NULL;
    if (len > 0) {
        memcpy(dup, s, len);
    }
    dup[len] = '\0';
    return dup;
}

static char *safe_strdup_with_limit(const char *s, size_t max_len) {
    if (!s) return NULL;
    size_t len = safe_strlen(s, max_len);
    if (len >= max_len) return NULL;
    char *dup = malloc(len + 1);
    if (!dup) return NULL;
    if (len > 0) {
        memcpy(dup, s, len);
    }
    dup[len] = '\0';
    return dup;
}

char **split_string(const char *input, size_t input_len, const char *delimiters, size_t delim_len, int *count) {
    if (!input || !delimiters || !count) {
        return NULL;
    }

    if (delim_len == 0) {
        return NULL;
    }

    regex_t regex;
    size_t pattern_size = delim_len + 5;
    char *pattern = malloc(pattern_size);
    if (!pattern) {
        return NULL;
    }
    
    int written = snprintf(pattern, pattern_size, "[%s]+", delimiters);
    if (written < 0 || (size_t)written >= pattern_size) {
        free(pattern);
        return NULL;
    }

    int ret = regcomp(&regex, pattern, REG_EXTENDED);
    free(pattern);
    if (ret != 0) {
        return NULL;
    }

    char **tokens = NULL;
    int token_count = 0;
    const char *current = input;
    size_t remaining_len = input_len;
    regmatch_t match;

    while (remaining_len > 0 && *current != '\0') {
        if (regexec(&regex, current, 1, &match, 0) == 0) {
            if (match.rm_so > 0) {
                if ((size_t)match.rm_so > remaining_len) {
                    for (int i = 0; i < token_count; i++) {
                        free(tokens[i]);
                    }
                    free(tokens);
                    regfree(&regex);
                    return NULL;
                }
                char **new_tokens = realloc(tokens, (token_count + 1) * sizeof(char *));
                if (!new_tokens) {
                    for (int i = 0; i < token_count; i++) {
                        free(tokens[i]);
                    }
                    free(tokens);
                    regfree(&regex);
                    return NULL;
                }
                tokens = new_tokens;
                tokens[token_count] = safe_strndup(current, (size_t)match.rm_so);
                if (!tokens[token_count]) {
                    for (int i = 0; i < token_count; i++) {
                        free(tokens[i]);
                    }
                    free(tokens);
                    regfree(&regex);
                    return NULL;
                }
                token_count++;
            }
            size_t advance = (size_t)match.rm_eo;
            if (advance > remaining_len) {
                advance = remaining_len;
            }
            current += advance;
            remaining_len -= advance;
        } else {
            if (remaining_len > 0 && *current != '\0') {
                char **new_tokens = realloc(tokens, (token_count + 1) * sizeof(char *));
                if (!new_tokens) {
                    for (int i = 0; i < token_count; i++) {
                        free(tokens[i]);
                    }
                    free(tokens);
                    regfree(&regex);
                    return NULL;
                }
                tokens = new_tokens;
                tokens[token_count] = safe_strdup_with_limit(current, remaining_len);
                if (!tokens[token_count]) {
                    for (int i = 0; i < token_count; i++) {
                        free(tokens[i]);
                    }
                    free(tokens);
                    regfree(&regex);
                    return NULL;
                }
                token_count++;
            }
            break;
        }
    }

    regfree(&regex);
    *count = token_count;
    return tokens;
}

void free_tokens(char **tokens, int count) {
    if (!tokens) return;
    for (int i = 0; i < count; i++) {
        free(tokens[i]);
    }
    free(tokens);
}

int main(int argc, char *argv[]) {
    (void)argc;
    (void)argv;
    
    const char *input = "Hello,world;this:is|a test,string;with:multiple delimiters";
    size_t input_len = safe_strlen(input, 4096);
    const char *delimiters = ",;:| ";
    size_t delim_len = safe_strlen(delimiters, 256);
    int count = 0;

    char **tokens = split_string(input, input_len, delimiters, delim_len, &count);

    if (!tokens) {
        fprintf(stderr, "Failed to split string\n");
        return 1;
    }

    for (int i = 0; i < count; i++) {
        printf("Token %d: %s\n", i, tokens[i]);
    }

    free_tokens(tokens, count);
    return 0;
}
