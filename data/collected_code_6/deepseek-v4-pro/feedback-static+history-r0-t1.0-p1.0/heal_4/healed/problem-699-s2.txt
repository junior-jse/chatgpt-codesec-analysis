int min_swaps_to_convert(const char *s1, const char *s2) {
    if (s1 == NULL || s2 == NULL) {
        return -1;
    }

    size_t len1 = strnlen(s1, 1025);
    size_t len2 = strnlen(s2, 1025);

    if (len1 != len2) {
        return -1;
    }

    if (len1 > 1024 || len1 == 1025) {
        return -1;
    }

    size_t len = len1;
    int count_s1 = 0;
    int count_s2 = 0;
    int mismatches = 0;
    int ones_mismatch = 0;

    for (size_t i = 0; i < len; i++) {
        if (s1[i] == '1') {
            count_s1++;
        }
        if (s2[i] == '1') {
            count_s2++;
        }
        if (s1[i] != s2[i]) {
            mismatches++;
            if (s1[i] == '1') {
                ones_mismatch++;
            }
        }
    }

    if (count_s1 != count_s2) {
        return -1;
    }

    return ones_mismatch < (mismatches - ones_mismatch) ? ones_mismatch : (mismatches - ones_mismatch);
}
