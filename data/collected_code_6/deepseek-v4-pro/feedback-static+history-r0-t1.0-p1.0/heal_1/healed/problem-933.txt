#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

static size_t count_snake_length(const char *input) {
    size_t len = 0;
    int prev_lower = 0;

    if (input == NULL) {
        return 0;
    }

    while (*input) {
        if (isupper((unsigned char)*input)) {
            if (prev_lower) {
                len++;
            }
            prev_lower = 0;
        } else {
            prev_lower = islower((unsigned char)*input) || isdigit((unsigned char)*input);
        }
        len++;
        input++;
    }
    return len;
}

char *camel_to_snake(const char *input) {
    size_t output_len;
    char *output;
    char *out_ptr;
    int prev_lower = 0;

    if (input == NULL) {
        return NULL;
    }

    output_len = count_snake_length(input);
    if (output_len == 0 && *input == '\0') {
        output = (char *)malloc(1);
        if (output == NULL) {
            return NULL;
        }
        output[0] = '\0';
        return output;
    }

    output = (char *)malloc(output_len + 1);
    if (output == NULL) {
        return NULL;
    }

    out_ptr = output;

    while (*input) {
        if (isupper((unsigned char)*input)) {
            if (prev_lower) {
                *out_ptr++ = '_';
            }
            *out_ptr++ = (char)tolower((unsigned char)*input);
            prev_lower = 0;
        } else {
            *out_ptr++ = *input;
            prev_lower = islower((unsigned char)*input) || isdigit((unsigned char)*input);
        }
        input++;
    }

    *out_ptr = '\0';
    return output;
}

int main(void) {
    const char *tests[] = {
        "camelCase",
        "helloWorld",
        "thisIsATest",
        "already_snake",
        "mixedCase_WithSnake",
        "UPPERCASE",
        "camel123Case",
        "c",
        "",
        NULL
    };
    size_t num_tests = sizeof(tests) / sizeof(tests[0]);

    for (size_t i = 0; i < num_tests; i++) {
        char *result = camel_to_snake(tests[i]);
        if (result != NULL) {
            printf("%s -> %s\n", tests[i] ? tests[i] : "(null)", result);
            free(result);
        } else {
            printf("%s -> (conversion failed)\n", tests[i] ? tests[i] : "(null)");
        }
    }

    return 0;
}
