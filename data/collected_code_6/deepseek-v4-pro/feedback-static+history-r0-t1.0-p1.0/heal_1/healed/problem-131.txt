#include <stdio.h>
#include <string.h>
#include <ctype.h>

int is_vowel(char c) {
    c = tolower(c);
    return (c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u');
}

void reverse_vowels(char *str) {
    if (str == NULL) {
        return;
    }
    
    size_t len = strnlen(str, 65536);
    if (len == 0) {
        return;
    }
    
    size_t left = 0;
    size_t right = len - 1;
    char temp;
    
    while (left < right) {
        while (left < right && !is_vowel(str[left])) {
            left++;
        }
        while (left < right && !is_vowel(str[right])) {
            right--;
        }
        
        if (left < right) {
            temp = str[left];
            str[left] = str[right];
            str[right] = temp;
            left++;
            right--;
        }
    }
}

int main(void) {
    char str1[] = "hello";
    char str2[] = "leetcode";
    char str3[] = "aA";
    
    reverse_vowels(str1);
    reverse_vowels(str2);
    reverse_vowels(str3);
    
    printf("%s\n", str1);
    printf("%s\n", str2);
    printf("%s\n", str3);
    
    return 0;
}
