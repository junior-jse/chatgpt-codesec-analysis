#include <stdio.h>
#include <stdlib.h>

int* find_max_sublist(const int *arr, int n, int *start, int *len) {
    if (arr == NULL || n <= 0 || start == NULL || len == NULL) {
        return NULL;
    }

    int max_len = 1;
    int curr_len = 1;
    int max_start = 0;
    int curr_start = 0;

    for (int i = 1; i < n; i++) {
        if (arr[i] == arr[i-1] + 1) {
            curr_len++;
        } else {
            if (curr_len > max_len) {
                max_len = curr_len;
                max_start = curr_start;
            }
            curr_len = 1;
            curr_start = i;
        }
    }

    if (curr_len > max_len) {
        max_len = curr_len;
        max_start = curr_start;
    }

    *start = max_start;
    *len = max_len;

    int *result = (int*)malloc(max_len * sizeof(int));
    if (result == NULL) {
        return NULL;
    }

    for (int i = 0; i < max_len; i++) {
        result[i] = arr[max_start + i];
    }

    return result;
}

int main(void) {
    int test_cases[][10] = {
        {1, 2, 3, 5, 6, 7, 8, 2, 3, 4},
        {5, 4, 3, 2, 1},
        {1, 3, 5, 7, 9},
        {2, 3, 4, 5, 6},
        {1}
    };

    const int sizes[] = {10, 5, 5, 5, 1};

    for (int t = 0; t < 5; t++) {
        int start = 0;
        int len = 0;
        int *sublist = find_max_sublist(test_cases[t], sizes[t], &start, &len);

        printf("Test %d: ", t + 1);
        if (sublist != NULL) {
            printf("Start=%d, Length=%d, Sublist: [", start, len);
            for (int i = 0; i < len; i++) {
                if (i > 0) printf(", ");
                printf("%d", sublist[i]);
            }
            printf("]\n");
            free(sublist);
        } else {
            printf("Error: Invalid input or allocation failure\n");
        }
    }

    return 0;
}
