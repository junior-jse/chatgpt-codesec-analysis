#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <stdint.h>

int shortest_supersequence_length(const char *str1, const char *str2) {
    if (str1 == NULL || str2 == NULL) {
        return -1;
    }
    
    size_t m = strnlen(str1, SIZE_MAX);
    size_t n = strnlen(str2, SIZE_MAX);
    
    if (m > (size_t)(INT_MAX - 1) || n > (size_t)(INT_MAX - 1)) {
        return -1;
    }
    
    int m_int = (int)m;
    int n_int = (int)n;
    
    int **dp = (int **)malloc((size_t)(m_int + 1) * sizeof(int *));
    if (dp == NULL) {
        return -1;
    }
    
    for (int i = 0; i <= m_int; i++) {
        dp[i] = (int *)malloc((size_t)(n_int + 1) * sizeof(int));
        if (dp[i] == NULL) {
            for (int j = 0; j < i; j++) {
                free(dp[j]);
            }
            free(dp);
            return -1;
        }
    }
    
    for (int i = 0; i <= m_int; i++) {
        dp[i][0] = i;
    }
    for (int j = 0; j <= n_int; j++) {
        dp[0][j] = j;
    }
    
    for (int i = 1; i <= m_int; i++) {
        for (int j = 1; j <= n_int; j++) {
            if (str1[i-1] == str2[j-1]) {
                dp[i][j] = dp[i-1][j-1] + 1;
            } else {
                dp[i][j] = (dp[i-1][j] < dp[i][j-1] ? dp[i-1][j] : dp[i][j-1]) + 1;
            }
        }
    }
    
    int result = dp[m_int][n_int];
    
    for (int i = 0; i <= m_int; i++) {
        free(dp[i]);
    }
    free(dp);
    
    return result;
}

int main(void) {
    const char *str1 = "AGGTAB";
    const char *str2 = "GXTXAYB";
    
    int length = shortest_supersequence_length(str1, str2);
    
    if (length >= 0) {
        printf("Length of shortest supersequence: %d\n", length);
    } else {
        printf("Memory allocation failed or invalid input\n");
    }
    
    return 0;
}
