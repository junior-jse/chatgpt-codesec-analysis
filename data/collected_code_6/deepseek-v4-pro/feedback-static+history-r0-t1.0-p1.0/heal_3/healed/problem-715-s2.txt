#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <errno.h>
#include <limits.h>

typedef struct {
    int *values;
    size_t size;
} Tuple;

static int safe_add(size_t a, size_t b, size_t *result) {
    if (a > SIZE_MAX - b) {
        return 0;
    }
    *result = a + b;
    return 1;
}

static int safe_mul(size_t a, size_t b, size_t *result) {
    if (b != 0 && a > SIZE_MAX / b) {
        return 0;
    }
    *result = a * b;
    return 1;
}

Tuple parse_tuple(const char *str) {
    Tuple result = {NULL, 0};
    if (str == NULL) {
        return result;
    }

    size_t len = 0;
    while (str[len] != '\0') {
        if (len == SIZE_MAX) {
            return result;
        }
        len++;
    }

    if (len < 2 || str[0] != '(' || str[len - 1] != ')') {
        return result;
    }

    size_t content_len = len - 2;
    size_t alloc_size;
    if (!safe_add(content_len, 1, &alloc_size)) {
        return result;
    }

    char *copy = (char *)malloc(alloc_size);
    if (copy == NULL) {
        return result;
    }

    if (content_len > 0) {
        memmove(copy, str + 1, content_len);
    }
    copy[content_len] = '\0';

    size_t capacity = 8;
    size_t values_size;
    if (!safe_mul(capacity, sizeof(int), &values_size)) {
        free(copy);
        return result;
    }

    int *values = (int *)malloc(values_size);
    if (values == NULL) {
        free(copy);
        return result;
    }

    char *saveptr;
    char *token = strtok_r(copy, ",", &saveptr);
    size_t count = 0;

    while (token != NULL) {
        while (*token == ' ') {
            token++;
        }

        if (*token == '\0') {
            token = strtok_r(NULL, ",", &saveptr);
            continue;
        }

        if (count == capacity) {
            size_t new_capacity;
            if (!safe_mul(capacity, 2, &new_capacity)) {
                free(values);
                free(copy);
                return result;
            }

            size_t new_values_size;
            if (!safe_mul(new_capacity, sizeof(int), &new_values_size)) {
                free(values);
                free(copy);
                return result;
            }

            int *temp = (int *)realloc(values, new_values_size);
            if (temp == NULL) {
                free(values);
                free(copy);
                return result;
            }
            values = temp;
            capacity = new_capacity;
        }

        char *endptr;
        errno = 0;
        long val = strtol(token, &endptr, 10);

        if (errno == ERANGE || val > INT_MAX || val < INT_MIN) {
            free(values);
            free(copy);
            return result;
        }

        if (endptr == token) {
            free(values);
            free(copy);
            return result;
        }

        while (*endptr == ' ') {
            endptr++;
        }

        if (*endptr != '\0') {
            free(values);
            free(copy);
            return result;
        }

        values[count++] = (int)val;
        token = strtok_r(NULL, ",", &saveptr);
    }

    free(copy);

    result.values = values;
    result.size = count;

    return result;
}

void free_tuple(Tuple *tuple) {
    if (tuple != NULL && tuple->values != NULL) {
        free(tuple->values);
        tuple->values = NULL;
        tuple->size = 0;
    }
}

int main(void) {
    const char *input = "(1, 2, 3, 4, 5)";
    Tuple tuple = parse_tuple(input);

    if (tuple.values == NULL) {
        fprintf(stderr, "Failed to parse tuple\n");
        return EXIT_FAILURE;
    }

    for (size_t i = 0; i < tuple.size; i++) {
        printf("%d ", tuple.values[i]);
    }
    printf("\n");

    free_tuple(&tuple);
    return EXIT_SUCCESS;
}
