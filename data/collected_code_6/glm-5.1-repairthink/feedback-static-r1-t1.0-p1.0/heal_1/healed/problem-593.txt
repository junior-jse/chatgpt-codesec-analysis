#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char *remove_leading_zeroes_ip(const char *ip) {
    size_t max_len = 16;
    size_t len = strnlen(ip, max_len);
    if (len == 0 || len == max_len) {
        return NULL;
    }

    char *result = (char *)malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    size_t res_idx = 0;
    size_t i = 0;

    while (i < len) {
        if (ip[i] == '.') {
            result[res_idx++] = '.';
            i++;
        } else {
            size_t start = i;
            while (i < len && ip[i] != '.') {
                i++;
            }
            
            size_t first_non_zero = start;
            while (first_non_zero < i && ip[first_non_zero] == '0') {
                first_non_zero++;
            }

            if (first_non_zero == i) {
                result[res_idx++] = '0';
            } else {
                for (size_t j = first_non_zero; j < i; j++) {
                    result[res_idx++] = ip[j];
                }
            }
        }
    }

    result[res_idx] = '\0';
    return result;
}

int main(void) {
    const char *ip1 = "192.168.001.010";
    const char *ip2 = "000.000.000.000";
    const char *ip3 = "010.001.100.200";

    char *res1 = remove_leading_zeroes_ip(ip1);
    if (res1 != NULL) {
        printf("%s\n", res1);
        free(res1);
    }

    char *res2 = remove_leading_zeroes_ip(ip2);
    if (res2 != NULL) {
        printf("%s\n", res2);
        free(res2);
    }

    char *res3 = remove_leading_zeroes_ip(ip3);
    if (res3 != NULL) {
        printf("%s\n", res3);
        free(res3);
    }

    return 0;
}
