#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int is_balanced(const char *expression) {
    if (expression == NULL) {
        return 0;
    }

    size_t capacity = 16;
    char *stack = (char *)malloc(capacity);
    if (stack == NULL) {
        return 0;
    }

    int top = -1;

    for (size_t i = 0; expression[i] != '\0'; i++) {
        char ch = expression[i];
        if (ch == '(' || ch == '{' || ch == '[') {
            if ((size_t)(top + 1) >= capacity) {
                capacity *= 2;
                char *new_stack = (char *)realloc(stack, capacity);
                if (new_stack == NULL) {
                    free(stack);
                    return 0;
                }
                stack = new_stack;
            }
            stack[++top] = ch;
        } else if (ch == ')' || ch == '}' || ch == ']') {
            if (top == -1) {
                free(stack);
                return 0;
            }
            char open = stack[top--];
            if (!((ch == ')' && open == '(') ||
                  (ch == '}' && open == '{') ||
                  (ch == ']' && open == '['))) {
                free(stack);
                return 0;
            }
        }
    }

    int result = (top == -1) ? 1 : 0;
    free(stack);
    return result;
}

int main(int argc, const char * const argv[]) {
    if (argc != 2) {
        fprintf(stderr, "Usage: %s <expression>\n", argv[0]);
        return 1;
    }

    if (is_balanced(argv[1])) {
        printf("Balanced\n");
    } else {
        printf("Not Balanced\n");
    }

    return 0;
}
