#include <stdio.h>
#include <stdlib.h>

int max_sum_no_three_consecutive(int arr[], int n) {
    if (n == 0) {
        return 0;
    }
    if (n == 1) {
        return arr[0] > 0 ? arr[0] : 0;
    }
    if (n == 2) {
        int sum = arr[0] + arr[1];
        return sum > 0 ? sum : 0;
    }

    int *dp = (int *)malloc(n * sizeof(int));
    if (dp == NULL) {
        return 0;
    }

    dp[0] = arr[0] > 0 ? arr[0] : 0;
    int sum1 = arr[0] + arr[1];
    dp[1] = sum1 > 0 ? sum1 : 0;
    
    int max_val = arr[0] > arr[1] ? arr[0] : arr[1];
    dp[2] = max_val > 0 ? max_val + (arr[2] > 0 ? arr[2] : 0) : (arr[2] > 0 ? arr[2] : 0);
    if (dp[1] > dp[2]) {
        dp[2] = dp[1];
    }

    for (int i = 3; i < n; i++) {
        int val = arr[i] > 0 ? arr[i] : 0;
        int max1 = dp[i - 1];
        int max2 = dp[i - 2] + val;
        int max3 = dp[i - 3] + (arr[i - 1] > 0 ? arr[i - 1] : 0) + val;
        
        int current_max = max1;
        if (max2 > current_max) {
            current_max = max2;
        }
        if (max3 > current_max) {
            current_max = max3;
        }
        dp[i] = current_max;
    }

    int result = dp[n - 1];
    free(dp);
    return result;
}

int main() {
    int arr[] = {1, 2, 3, 4, 5, 6, 7, 8};
    int n = sizeof(arr) / sizeof(arr[0]);
    int max_sum = max_sum_no_three_consecutive(arr, n);
    printf("%d\n", max_sum);
    return 0;
}
