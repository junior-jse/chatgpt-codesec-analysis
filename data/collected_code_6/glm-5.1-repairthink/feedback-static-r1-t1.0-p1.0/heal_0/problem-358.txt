#include <stdio.h>
#include <stdlib.h>

int* find_modulo_division(const int* list1, const int* list2, size_t size) {
    int* result = (int*)malloc(size * sizeof(int));
    if (result == NULL) {
        return NULL;
    }

    for (size_t i = 0; i < size; i++) {
        if (list2[i] == 0) {
            free(result);
            return NULL;
        }
        result[i] = list1[i] % list2[i];
    }

    return result;
}

int main(void) {
    int list1[] = {10, 25, 32, 14, 50};
    int list2[] = {3, 4, 5, 2, 6};
    size_t size = sizeof(list1) / sizeof(list1[0]);

    int* result = find_modulo_division(list1, list2, size);
    if (result == NULL) {
        return EXIT_FAILURE;
    }

    for (size_t i = 0; i < size; i++) {
        printf("%d ", result[i]);
    }
    printf("\n");

    free(result);
    result = NULL;

    return EXIT_SUCCESS;
}
