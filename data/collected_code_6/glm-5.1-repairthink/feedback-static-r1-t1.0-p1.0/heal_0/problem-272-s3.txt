#include <stdlib.h>
#include <stdio.h>

typedef struct {
    int a;
    int b;
} Tuple;

typedef struct {
    Tuple *data;
    size_t len;
    size_t cap;
} TupleList;

TupleList *tuple_list_create(size_t cap) {
    TupleList *list = (TupleList *)malloc(sizeof(TupleList));
    if (list == NULL) {
        return NULL;
    }
    list->data = (Tuple *)malloc(cap * sizeof(Tuple));
    if (list->data == NULL) {
        free(list);
        return NULL;
    }
    list->len = 0;
    list->cap = cap;
    return list;
}

int tuple_list_push(TupleList *list, Tuple t) {
    if (list == NULL) {
        return -1;
    }
    if (list->len >= list->cap) {
        size_t new_cap = list->cap == 0 ? 1 : list->cap * 2;
        Tuple *new_data = (Tuple *)realloc(list->data, new_cap * sizeof(Tuple));
        if (new_data == NULL) {
            return -1;
        }
        list->data = new_data;
        list->cap = new_cap;
    }
    list->data[list->len] = t;
    list->len++;
    return 0;
}

int tuple_list_pop_rear(TupleList *list, Tuple *out) {
    if (list == NULL || out == NULL) {
        return -1;
    }
    if (list->len == 0) {
        return -1;
    }
    *out = list->data[list->len - 1];
    list->len--;
    return 0;
}

void tuple_list_destroy(TupleList *list) {
    if (list != NULL) {
        free(list->data);
        free(list);
    }
}

int main(void) {
    TupleList *list = tuple_list_create(2);
    if (list == NULL) {
        return EXIT_FAILURE;
    }

    Tuple t1 = {1, 2};
    Tuple t2 = {3, 4};
    Tuple t3 = {5, 6};

    if (tuple_list_push(list, t1) != 0) {
        tuple_list_destroy(list);
        return EXIT_FAILURE;
    }
    if (tuple_list_push(list, t2) != 0) {
        tuple_list_destroy(list);
        return EXIT_FAILURE;
    }
    if (tuple_list_push(list, t3) != 0) {
        tuple_list_destroy(list);
        return EXIT_FAILURE;
    }

    Tuple rear;
    if (tuple_list_pop_rear(list, &rear) == 0) {
        printf("Rear element: (%d, %d)\n", rear.a, rear.b);
    } else {
        tuple_list_destroy(list);
        return EXIT_FAILURE;
    }

    if (tuple_list_pop_rear(list, &rear) == 0) {
        printf("Rear element: (%d, %d)\n", rear.a, rear.b);
    } else {
        tuple_list_destroy(list);
        return EXIT_FAILURE;
    }

    if (tuple_list_pop_rear(list, &rear) == 0) {
        printf("Rear element: (%d, %d)\n", rear.a, rear.b);
    } else {
        tuple_list_destroy(list);
        return EXIT_FAILURE;
    }

    if (tuple_list_pop_rear(list, &rear) != 0) {
        printf("List is empty, no rear element to extract.\n");
    }

    tuple_list_destroy(list);
    return EXIT_SUCCESS;
}
