#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct Node {
    char *key;
    int value;
    struct Node *next;
} Node;

typedef struct {
    Node **buckets;
    size_t size;
} Dictionary;

unsigned long hash(const char *str) {
    unsigned long hash = 5381;
    int c;
    while ((c = *str++)) {
        hash = ((hash << 5) + hash) + c;
    }
    return hash;
}

Dictionary *dict_create(size_t size) {
    Dictionary *dict = malloc(sizeof(Dictionary));
    if (!dict) return NULL;
    dict->size = size;
    dict->buckets = calloc(size, sizeof(Node *));
    if (!dict->buckets) {
        free(dict);
        return NULL;
    }
    return dict;
}

int dict_insert(Dictionary *dict, const char *key, int value) {
    if (!dict || !key) return 0;
    size_t index = hash(key) % dict->size;
    Node *current = dict->buckets[index];
    while (current) {
        if (strcmp(current->key, key) == 0) {
            current->value = value;
            return 1;
        }
        current = current->next;
    }
    Node *new_node = malloc(sizeof(Node));
    if (!new_node) return 0;
    new_node->key = strdup(key);
    if (!new_node->key) {
        free(new_node);
        return 0;
    }
    new_node->value = value;
    new_node->next = dict->buckets[index];
    dict->buckets[index] = new_node;
    return 1;
}

Dictionary *dict_merge(Dictionary *dict1, Dictionary *dict2) {
    if (!dict1 || !dict2) return NULL;
    Dictionary *merged = dict_create(dict1->size + dict2->size);
    if (!merged) return NULL;
    for (size_t i = 0; i < dict1->size; i++) {
        Node *current = dict1->buckets[i];
        while (current) {
            if (!dict_insert(merged, current->key, current->value)) {
                dict_free(merged);
                return NULL;
            }
            current = current->next;
        }
    }
    for (size_t i = 0; i < dict2->size; i++) {
        Node *current = dict2->buckets[i];
        while (current) {
            if (!dict_insert(merged, current->key, current->value)) {
                dict_free(merged);
                return NULL;
            }
            current = current->next;
        }
    }
    return merged;
}

void dict_free(Dictionary *dict) {
    if (!dict) return;
    for (size_t i = 0; i < dict->size; i++) {
        Node *current = dict->buckets[i];
        while (current) {
            Node *temp = current;
            current = current->next;
            free(temp->key);
            free(temp);
        }
    }
    free(dict->buckets);
    free(dict);
}

void dict_print(Dictionary *dict) {
    if (!dict) return;
    for (size_t i = 0; i < dict->size; i++) {
        Node *current = dict->buckets[i];
        while (current) {
            printf("%s: %d\n", current->key, current->value);
            current = current->next;
        }
    }
}

int main(void) {
    Dictionary *dict1 = dict_create(16);
    if (!dict1) return EXIT_FAILURE;
    Dictionary *dict2 = dict_create(16);
    if (!dict2) {
        dict_free(dict1);
        return EXIT_FAILURE;
    }
    if (!dict_insert(dict1, "apple", 1) || !dict_insert(dict1, "banana", 2)) {
        dict_free(dict1);
        dict_free(dict2);
        return EXIT_FAILURE;
    }
    if (!dict_insert(dict2, "cherry", 3) || !dict_insert(dict2, "apple", 4)) {
        dict_free(dict1);
        dict_free(dict2);
        return EXIT_FAILURE;
    }
    Dictionary *merged = dict_merge(dict1, dict2);
    if (!merged) {
        dict_free(dict1);
        dict_free(dict2);
        return EXIT_FAILURE;
    }
    dict_print(merged);
    dict_free(dict1);
    dict_free(dict2);
    dict_free(merged);
    return EXIT_SUCCESS;
}
