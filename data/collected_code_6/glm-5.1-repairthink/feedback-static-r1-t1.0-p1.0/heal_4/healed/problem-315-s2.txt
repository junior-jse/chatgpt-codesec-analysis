#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

char* find_first_max_even_word(const char* str, size_t* out_len) {
    if (str == NULL || out_len == NULL) {
        return NULL;
    }

    *out_len = 0;
    size_t i = 0;
    size_t best_start = 0;
    size_t best_len = 0;

    while (str[i] != '\0') {
        size_t start = i;
        while (str[i] != '\0' && str[i] != ' ') {
            i++;
        }
        size_t word_len = i - start;

        if (word_len > 0 && word_len % 2 == 0) {
            if (word_len > best_len) {
                best_len = word_len;
                best_start = start;
                *out_len = word_len;
            }
        }

        if (str[i] != '\0') {
            i++;
        }
    }

    if (best_len == 0) {
        return NULL;
    }

    if (best_len == SIZE_MAX) {
        return NULL;
    }

    size_t alloc_size = best_len + 1;

    char* result = malloc(alloc_size);
    if (result == NULL) {
        return NULL;
    }

    memcpy(result, str + best_start, alloc_size - 1);
    result[best_len] = '\0';

    return result;
}

int main(void) {
    const char* text = "This is an example string with some long even words like eight";
    size_t max_len = 0;
    
    char* word = find_first_max_even_word(text, &max_len);
    
    if (word != NULL) {
        printf("First max even word: %s\n", word);
        printf("Length: %zu\n", max_len);
        free(word);
    } else {
        printf("No even length word found.\n");
    }
    
    return 0;
}
