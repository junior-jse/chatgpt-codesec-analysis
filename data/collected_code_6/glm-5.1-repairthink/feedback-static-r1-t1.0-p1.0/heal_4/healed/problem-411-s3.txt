#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>
#include <ctype.h>

char *snake_to_camel(const char *snake_str, size_t len) {
    if (snake_str == NULL) {
        return NULL;
    }

    size_t safe_len = strnlen(snake_str, len);
    char *safe_str = malloc(safe_len + 1);
    if (safe_str == NULL) {
        return NULL;
    }

    snprintf(safe_str, safe_len + 1, "%s", snake_str);

    regex_t regex;
    regmatch_t matches[2];
    int ret;

    ret = regcomp(&regex, "_([a-z])", REG_EXTENDED);
    if (ret != 0) {
        free(safe_str);
        return NULL;
    }

    size_t max_len = safe_len + 1;
    char *camel_str = malloc(max_len);
    if (camel_str == NULL) {
        regfree(&regex);
        free(safe_str);
        return NULL;
    }

    size_t i = 0;
    const char *cursor = safe_str;

    while (1) {
        ret = regexec(&regex, cursor, 2, matches, 0);
        if (ret == REG_NOMATCH || matches[0].rm_so == -1 || matches[1].rm_so == -1) {
            size_t remaining = max_len - i;
            snprintf(camel_str + i, remaining, "%s", cursor);
            break;
        }

        size_t copy_len = matches[0].rm_so;
        size_t remaining = max_len - i;
        
        if (copy_len > remaining) {
            copy_len = remaining;
        }

        for (size_t j = 0; j < copy_len; j++) {
            camel_str[i + j] = cursor[j];
        }
        i += copy_len;

        remaining = max_len - i;
        if (remaining > 0) {
            camel_str[i++] = toupper((unsigned char)cursor[matches[1].rm_so]);
        }

        cursor += matches[0].rm_eo;
    }

    regfree(&regex);
    free(safe_str);
    return camel_str;
}

int main(void) {
    char input[] = "convert_this_snake_case_string";
    size_t len = sizeof(input) - 1;
    char *output = snake_to_camel(input, len);

    if (output != NULL) {
        printf("%s\n", output);
        free(output);
    } else {
        fprintf(stderr, "Conversion failed\n");
        return 1;
    }

    return 0;
}
