#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define INITIAL_CAPACITY 16

typedef struct {
    char *key;
    char **values;
    size_t count;
    size_t capacity;
} DictEntry;

typedef struct {
    DictEntry *entries;
    size_t count;
    size_t capacity;
} Dictionary;

int dict_add(Dictionary *dict, const char *key, const char *value) {
    if (dict == NULL || key == NULL || value == NULL) {
        return -1;
    }

    DictEntry *entry = NULL;
    for (size_t i = 0; i < dict->count; i++) {
        if (strcmp(dict->entries[i].key, key) == 0) {
            entry = &dict->entries[i];
            break;
        }
    }

    if (entry == NULL) {
        if (dict->count >= dict->capacity) {
            size_t new_cap = dict->capacity == 0 ? INITIAL_CAPACITY : dict->capacity * 2;
            DictEntry *new_entries = realloc(dict->entries, new_cap * sizeof(DictEntry));
            if (new_entries == NULL) {
                return -1;
            }
            dict->entries = new_entries;
            dict->capacity = new_cap;
        }
        
        entry = &dict->entries[dict->count];
        entry->key = strdup(key);
        if (entry->key == NULL) {
            return -1;
        }
        
        entry->values = NULL;
        entry->count = 0;
        entry->capacity = 0;
        dict->count++;
    }

    if (entry->count >= entry->capacity) {
        size_t new_cap = entry->capacity == 0 ? INITIAL_CAPACITY : entry->capacity * 2;
        char **new_values = realloc(entry->values, new_cap * sizeof(char *));
        if (new_values == NULL) {
            return -1;
        }
        entry->values = new_values;
        entry->capacity = new_cap;
    }

    char *val_copy = strdup(value);
    if (val_copy == NULL) {
        return -1;
    }

    entry->values[entry->count] = val_copy;
    entry->count++;

    return 0;
}

void dict_free(Dictionary *dict) {
    if (dict == NULL) {
        return;
    }
    for (size_t i = 0; i < dict->count; i++) {
        free(dict->entries[i].key);
        for (size_t j = 0; j < dict->entries[i].count; j++) {
            free(dict->entries[i].values[j]);
        }
        free(dict->entries[i].values);
    }
    free(dict->entries);
    dict->entries = NULL;
    dict->count = 0;
    dict->capacity = 0;
}

int main(void) {
    Dictionary dict;
    memset(&dict, 0, sizeof(Dictionary));

    const char *keys[] = {"fruit", "color", "fruit", "color", "fruit"};
    const char *values[] = {"apple", "red", "banana", "green", "cherry"};
    size_t seq_len = sizeof(keys) / sizeof(keys[0]);

    for (size_t i = 0; i < seq_len; i++) {
        if (dict_add(&dict, keys[i], values[i]) != 0) {
            dict_free(&dict);
            return EXIT_FAILURE;
        }
    }

    for (size_t i = 0; i < dict.count; i++) {
        printf("%s: [", dict.entries[i].key);
        for (size_t j = 0; j < dict.entries[i].count; j++) {
            printf("%s", dict.entries[i].values[j]);
            if (j + 1 < dict.entries[i].count) {
                printf(", ");
            }
        }
        printf("]\n");
    }

    dict_free(&dict);

    return EXIT_SUCCESS;
}
