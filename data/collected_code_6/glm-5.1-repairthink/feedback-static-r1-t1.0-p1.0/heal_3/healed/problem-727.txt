#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

char *remove_non_alnum(const char *str) {
    if (str == NULL) {
        return NULL;
    }

    regex_t regex;
    int ret = regcomp(&regex, "[^a-zA-Z0-9]", REG_EXTENDED);
    if (ret != 0) {
        return NULL;
    }

    size_t len = strlen(str);
    size_t result_size = len + 1;
    char *result = malloc(result_size);
    if (result == NULL) {
        regfree(&regex);
        return NULL;
    }

    size_t result_idx = 0;
    size_t str_offset = 0;

    while (str_offset < len) {
        regmatch_t matches;
        ret = regexec(&regex, str + str_offset, 1, &matches, 0);
        
        if (ret == REG_NOMATCH) {
            size_t rem_len = len - str_offset;
            size_t copy_len = rem_len;
            if (result_idx + copy_len >= result_size) {
                copy_len = result_size - result_idx - 1;
            }
            memcpy(result + result_idx, str + str_offset, copy_len);
            result_idx += copy_len;
            break;
        }

        if (ret != 0) {
            free(result);
            regfree(&regex);
            return NULL;
        }

        if (matches.rm_so == -1) {
            break;
        }

        size_t copy_len = matches.rm_so;
        if (result_idx + copy_len >= result_size) {
            copy_len = result_size - result_idx - 1;
        }
        
        if (copy_len > 0) {
            memcpy(result + result_idx, str + str_offset, copy_len);
        }
        result_idx += copy_len;
        str_offset += matches.rm_eo;
    }

    if (result_idx >= result_size) {
        result_idx = result_size - 1;
    }
    result[result_idx] = '\0';

    char *final_result = realloc(result, result_idx + 1);
    if (final_result == NULL) {
        final_result = result;
    }

    regfree(&regex);

    return final_result;
}

}
