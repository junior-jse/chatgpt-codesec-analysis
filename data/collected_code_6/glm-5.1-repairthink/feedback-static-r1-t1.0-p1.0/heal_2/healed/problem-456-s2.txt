#include <stdio.h>
#include <stdlib.h>
#include <string.h>

void reverse_string(char *str, size_t size) {
    if (str == NULL || size == 0) {
        return;
    }
    size_t len = strnlen(str, size);
    if (len <= 1) {
        return;
    }
    size_t left = 0;
    size_t right = len - 1;
    while (left < right) {
        char temp = str[left];
        str[left] = str[right];
        str[right] = temp;
        left++;
        right--;
    }
}

void reverse_string_list(char **list, size_t *sizes, size_t count) {
    if (list == NULL) {
        return;
    }
    for (size_t i = 0; i < count; i++) {
        size_t current_size = (sizes != NULL) ? sizes[i] : 0;
        reverse_string(list[i], current_size);
    }
}

int main(void) {
    char s0[] = "hello";
    char s1[] = "world";
    char s2[] = "C";
    char s3[] = "programming";
    
    char *list[] = {
        s0,
        s1,
        s2,
        s3
    };
    
    size_t sizes[] = {
        sizeof(s0),
        sizeof(s1),
        sizeof(s2),
        sizeof(s3)
    };

    size_t count = sizeof(list) / sizeof(list[0]);

    reverse_string_list(list, sizes, count);

    for (size_t i = 0; i < count; i++) {
        printf("%s\n", list[i]);
    }

    return 0;
}
