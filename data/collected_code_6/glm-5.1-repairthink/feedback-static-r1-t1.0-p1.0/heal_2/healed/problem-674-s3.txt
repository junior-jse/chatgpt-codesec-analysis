#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdbool.h>

char *remove_duplicate_words(const char *input) {
    if (input == NULL) {
        return NULL;
    }

    size_t len = strlen(input);
    char *temp = (char *)malloc(len + 1);
    if (temp == NULL) {
        return NULL;
    }

    memcpy(temp, input, len + 1);

    size_t max_words = len / 2 + 1;
    char **words = (char **)malloc(max_words * sizeof(char *));
    if (words == NULL) {
        free(temp);
        return NULL;
    }

    size_t word_count = 0;
    char *token = strtok(temp, " \t\n\r");
    while (token != NULL) {
        bool found = false;
        for (size_t i = 0; i < word_count; i++) {
            if (strcmp(words[i], token) == 0) {
                found = true;
                break;
            }
        }

        if (!found) {
            words[word_count] = token;
            word_count++;
        }

        token = strtok(NULL, " \t\n\r");
    }

    size_t result_size = 0;
    for (size_t i = 0; i < word_count; i++) {
        result_size += strlen(words[i]);
    }
    if (word_count > 0) {
        result_size += word_count - 1;
    }

    char *result = (char *)malloc(result_size + 1);
    if (result == NULL) {
        free(temp);
        free(words);
        return NULL;
    }

    size_t offset = 0;
    for (size_t i = 0; i < word_count; i++) {
        size_t word_len = strlen(words[i]);
        if (offset + word_len <= result_size) {
            memcpy(result + offset, words[i], word_len);
            offset += word_len;
        }
        if (i < word_count - 1) {
            if (offset < result_size) {
                result[offset++] = ' ';
            }
        }
    }
    result[offset] = '\0';

    free(temp);
    free(words);

    return result;
}

int main(void) {
    char str[] = "hello world hello there world again";
    char *cleaned = remove_duplicate_words(str);

    if (cleaned != NULL) {
        printf("Original: %s\n", str);
        printf("Cleaned: %s\n", cleaned);
        free(cleaned);
    }

    const char *null_test = remove_duplicate_words(NULL);
    if (null_test == NULL) {
        printf("NULL input handled safely.\n");
    }

    return 0;
}
