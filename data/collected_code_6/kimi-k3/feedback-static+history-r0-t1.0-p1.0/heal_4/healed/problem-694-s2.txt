#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdbool.h>

#define MAX_DICT_SIZE 100
#define MAX_STR_LEN 50

typedef struct {
    char key[MAX_STR_LEN];
    char value[MAX_STR_LEN];
} DictionaryEntry;

static bool is_valid_string(const char *str, size_t max_len) {
    if (str == NULL) {
        return false;
    }
    size_t i;
    for (i = 0; i < max_len; i++) {
        if (str[i] == '\0') {
            return true;
        }
    }
    return false;
}

static int is_unique(const char values[][MAX_STR_LEN], size_t count, const char *value) {
    size_t i;
    if (values == NULL || value == NULL) {
        return 0;
    }
    for (i = 0; i < count; i++) {
        if (strncmp(values[i], value, MAX_STR_LEN) == 0) {
            return 0;
        }
    }
    return 1;
}

static int safe_copy_string(char *dest, size_t dest_size, const char *src) {
    if (dest == NULL || src == NULL || dest_size == 0) {
        return -1;
    }
    if (!is_valid_string(src, dest_size)) {
        return -1;
    }
    size_t src_len = strnlen(src, dest_size - 1);
    memcpy(dest, src, src_len);
    dest[src_len] = '\0';
    return 0;
}

int extract_unique_values(const DictionaryEntry *dict, size_t dict_size, 
                          char unique_values[][MAX_STR_LEN], size_t max_unique) {
    size_t unique_count = 0;
    size_t i;
    
    if (dict == NULL || unique_values == NULL || dict_size == 0 || max_unique == 0) {
        return -1;
    }
    
    if (dict_size > MAX_DICT_SIZE || max_unique > MAX_DICT_SIZE) {
        return -1;
    }
    
    for (i = 0; i < dict_size; i++) {
        if (unique_count >= max_unique) {
            break;
        }
        if (!is_valid_string(dict[i].value, MAX_STR_LEN)) {
            continue;
        }
        if (is_unique(unique_values, unique_count, dict[i].value)) {
            if (safe_copy_string(unique_values[unique_count], MAX_STR_LEN, dict[i].value) != 0) {
                return -1;
            }
            unique_count++;
        }
    }
    
    return (int)unique_count;
}

int main(void) {
    DictionaryEntry dict[] = {
        {"key1", "value1"},
        {"key2", "value2"},
        {"key3", "value1"},
        {"key4", "value3"},
        {"key5", "value2"}
    };
    
    size_t dict_size = sizeof(dict) / sizeof(dict[0]);
    char unique_values[MAX_DICT_SIZE][MAX_STR_LEN];
    int unique_count;
    int i;
    
    memset(unique_values, 0, sizeof(unique_values));
    
    unique_count = extract_unique_values(dict, dict_size, unique_values, MAX_DICT_SIZE);
    
    if (unique_count < 0) {
        fprintf(stderr, "Error extracting unique values\n");
        return EXIT_FAILURE;
    }
    
    printf("Unique values:\n");
    for (i = 0; i < unique_count; i++) {
        if (is_valid_string(unique_values[i], MAX_STR_LEN)) {
            printf("%s\n", unique_values[i]);
        }
    }
    
    return EXIT_SUCCESS;
}
