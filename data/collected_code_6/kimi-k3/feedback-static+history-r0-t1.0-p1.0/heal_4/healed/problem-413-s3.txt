#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define NAME_SIZE 50

typedef struct {
    int id;
    char name[NAME_SIZE];
    double value;
} Tuple;

int extract_nth_element(const Tuple *list, size_t size, size_t n, Tuple *result) {
    if (list == NULL || result == NULL) {
        return -1;
    }
    if (n >= size) {
        return -1;
    }
    *result = list[n];
    return 0;
}

static void init_tuple(Tuple *t, int id, const char *name, double value) {
    if (t == NULL) {
        return;
    }
    t->id = id;
    t->value = value;
    if (name != NULL) {
        size_t len = strnlen(name, NAME_SIZE - 1);
        if (len >= NAME_SIZE) {
            len = NAME_SIZE - 1;
        }
        strncpy(t->name, name, len);
        t->name[len] = '\0';
    } else {
        t->name[0] = '\0';
    }
}

int main(void) {
    Tuple list[5];
    size_t size = sizeof(list) / sizeof(list[0]);
    size_t n = 2;
    Tuple result;

    init_tuple(&list[0], 1, "Alice", 100.5);
    init_tuple(&list[1], 2, "Bob", 200.75);
    init_tuple(&list[2], 3, "Charlie", 300.25);
    init_tuple(&list[3], 4, "Diana", 400.0);
    init_tuple(&list[4], 5, "Eve", 500.5);

    if (extract_nth_element(list, size, n, &result) == 0) {
        printf("Element at index %zu: id=%d, name=%s, value=%.2f\n",
               n, result.id, result.name, result.value);
    } else {
        fprintf(stderr, "Error: Invalid index or NULL pointer\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
