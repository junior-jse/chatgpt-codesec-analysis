#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node *next;
} Node;

Node *shift_first_to_end(Node *head)
{
    Node *new_head;
    Node *last;

    if (head == NULL || head->next == NULL) {
        return head;
    }

    new_head = head->next;
    last = head;

    while (last->next != NULL) {
        last = last->next;
    }

    last->next = head;
    head->next = NULL;

    return new_head;
}

Node *create_node(int data)
{
    Node *node = malloc(sizeof(Node));

    if (node == NULL) {
        return NULL;
    }

    node->data = data;
    node->next = NULL;

    return node;
}

void free_list(Node *head)
{
    Node *current = head;
    Node *next;

    while (current != NULL) {
        next = current->next;
        free(current);
        current = next;
    }
}

void print_list(const Node *head)
{
    const Node *current = head;

    while (current != NULL) {
        printf("%d", current->data);
        if (current->next != NULL) {
            printf(" -> ");
        }
        current = current->next;
    }
    printf("\n");
}

int main(void)
{
    Node *head = NULL;
    Node *tail = NULL;
    Node *node;
    int values[] = {1, 2, 3, 4, 5};
    size_t i;
    int status = EXIT_SUCCESS;

    for (i = 0; i < sizeof(values) / sizeof(values[0]); i++) {
        node = create_node(values[i]);
        if (node == NULL) {
            free_list(head);
            fprintf(stderr, "Memory allocation failed\n");
            return EXIT_FAILURE;
        }

        if (head == NULL) {
            head = node;
            tail = node;
        } else {
            tail->next = node;
            tail = node;
        }
    }

    printf("Original list: ");
    print_list(head);

    head = shift_first_to_end(head);

    printf("After shift:   ");
    print_list(head);

    free_list(head);

    return status;
}
