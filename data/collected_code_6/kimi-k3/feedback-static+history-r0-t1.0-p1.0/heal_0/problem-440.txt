static void find_adverbs(const char *sentence)
{
    char buffer[MAX_SENTENCE_LEN];
    char word[MAX_WORD_LEN];
    size_t i = 0;
    size_t word_start = 0;
    size_t word_len = 0;
    int found = 0;
    if (sentence == NULL) {
        fprintf(stderr, "Error: null sentence pointer\n");
        return;
    }
    if (strlen(sentence) >= MAX_SENTENCE_LEN) {
        fprintf(stderr, "Error: sentence too long\n");
        return;
    }
    strcpy(buffer, sentence);
    while (buffer[i] != '\0') {
        if (isalpha((unsigned char)buffer[i])) {
            word_start = i;
            word_len = 0;
            while (isalpha((unsigned char)buffer[i]) && word_len < MAX_WORD_LEN - 1) {
                word[word_len] = (char)tolower((unsigned char)buffer[i]);
                word_len++;
                i++;
            }
            word[word_len] = '\0';
            if (is_adverb(word)) {
                printf("Adverb: %-20s Position: %zu\n", word, word_start);
                found = 1;
            }
        } else {
            i++;
        }
    }
    if (!found) {
        printf("No adverbs found.\n");
    }
}

    sentence[strcspn(sentence, "\n")] = '\0';
    if (strlen(sentence) == 0) {
        fprintf(stderr, "Error: empty input\n");
        return EXIT_FAILURE;
    }
    find_adverbs(sentence);
    return EXIT_SUCCESS;
}
