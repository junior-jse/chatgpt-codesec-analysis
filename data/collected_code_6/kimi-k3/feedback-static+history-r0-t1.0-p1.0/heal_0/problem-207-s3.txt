#include <stdio.h>
#include <string.h>
#include <stdlib.h>

int longestRepeatingSubsequence(char *str) {
    if (str == NULL) {
        return 0;
    }
    
    int n = strlen(str);
    if (n == 0) {
        return 0;
    }
    
    int **dp = (int **)malloc((n + 1) * sizeof(int *));
    if (dp == NULL) {
        return 0;
    }
    
    for (int i = 0; i <= n; i++) {
        dp[i] = (int *)malloc((n + 1) * sizeof(int));
        if (dp[i] == NULL) {
            for (int j = 0; j < i; j++) {
                free(dp[j]);
            }
            free(dp);
            return 0;
        }
    }
    
    for (int i = 0; i <= n; i++) {
        for (int j = 0; j <= n; j++) {
            dp[i][j] = 0;
        }
    }
    
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            if (str[i - 1] == str[j - 1] && i != j) {
                dp[i][j] = 1 + dp[i - 1][j - 1];
            } else {
                dp[i][j] = (dp[i - 1][j] > dp[i][j - 1]) ? dp[i - 1][j] : dp[i][j - 1];
            }
        }
    }
    
    int result = dp[n][n];
    
    for (int i = 0; i <= n; i++) {
        free(dp[i]);
    }
    free(dp);
    
    return result;
}

int main(void) {
    char str1[] = "aabb";
    char str2[] = "aabebcdd";
    char str3[] = "";
    
    printf("Longest repeating subsequence in \"%s\": %d\n", str1, longestRepeatingSubsequence(str1));
    printf("Longest repeating subsequence in \"%s\": %d\n", str2, longestRepeatingSubsequence(str2));
    printf("Longest repeating subsequence in \"%s\": %d\n", str3, longestRepeatingSubsequence(str3));
    
    return 0;
}
