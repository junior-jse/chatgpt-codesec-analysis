#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int maxProfit(int k, int* prices, int pricesSize) {
    if (prices == NULL || pricesSize < 2 || k <= 0) {
        return 0;
    }

    if (k >= pricesSize / 2) {
        int profit = 0;
        for (int i = 1; i < pricesSize; i++) {
            if (prices[i] > prices[i - 1]) {
                profit += prices[i] - prices[i - 1];
            }
        }
        return profit;
    }

    int* buy = (int*)malloc((size_t)(k + 1) * sizeof(int));
    int* sell = (int*)malloc((size_t)(k + 1) * sizeof(int));
    
    if (buy == NULL || sell == NULL) {
        free(buy);
        free(sell);
        return 0;
    }

    for (int i = 0; i <= k; i++) {
        buy[i] = INT_MIN;
        sell[i] = 0;
    }

    for (int i = 0; i < pricesSize; i++) {
        for (int j = k; j >= 1; j--) {
            if (sell[j - 1] != INT_MIN) {
                int new_buy = sell[j - 1] - prices[i];
                if (new_buy > buy[j]) {
                    buy[j] = new_buy;
                }
            }
            
            if (buy[j] != INT_MIN) {
                int new_sell = buy[j] + prices[i];
                if (new_sell > sell[j]) {
                    sell[j] = new_sell;
                }
            }
        }
    }

    int result = sell[k];
    free(buy);
    free(sell);
    return result;
}

int main(void) {
    int prices1[] = {2, 4, 1};
    int size1 = 3;
    int k1 = 2;
    printf("Test 1: %d\n", maxProfit(k1, prices1, size1));

    int prices2[] = {3, 2, 6, 5, 0, 3};
    int size2 = 6;
    int k2 = 2;
    printf("Test 2: %d\n", maxProfit(k2, prices2, size2));

    int prices3[] = {1, 2, 3, 4, 5};
    int size3 = 5;
    int k3 = 2;
    printf("Test 3: %d\n", maxProfit(k3, prices3, size3));

    return 0;
}
