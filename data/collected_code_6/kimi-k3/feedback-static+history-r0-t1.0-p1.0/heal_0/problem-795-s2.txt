#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    int id;
    double price;
} Item;

typedef struct {
    Item *items;
    size_t size;
    size_t capacity;
} MaxHeap;

static void swap_items(Item *a, Item *b) {
    Item temp = *a;
    *a = *b;
    *b = temp;
}

static void max_heapify(MaxHeap *heap, size_t idx) {
    size_t largest = idx;
    size_t left = 2 * idx + 1;
    size_t right = 2 * idx + 2;

    if (left < heap->size && heap->items[left].price > heap->items[largest].price)
        largest = left;
    if (right < heap->size && heap->items[right].price > heap->items[largest].price)
        largest = right;

    if (largest != idx) {
        swap_items(&heap->items[idx], &heap->items[largest]);
        max_heapify(heap, largest);
    }
}

static void build_max_heap(MaxHeap *heap) {
    if (heap->size == 0) return;
    for (size_t i = heap->size / 2; i-- > 0;)
        max_heapify(heap, i);
}

static int max_heap_insert(MaxHeap *heap, Item item) {
    if (heap->size >= heap->capacity) {
        size_t new_cap = heap->capacity == 0 ? 8 : heap->capacity * 2;
        Item *new_items = realloc(heap->items, new_cap * sizeof(Item));
        if (new_items == NULL) return -1;
        heap->items = new_items;
        heap->capacity = new_cap;
    }
    heap->items[heap->size] = item;
    size_t idx = heap->size++;
    while (idx > 0) {
        size_t parent = (idx - 1) / 2;
        if (heap->items[parent].price >= heap->items[idx].price) break;
        swap_items(&heap->items[parent], &heap->items[idx]);
        idx = parent;
    }
    return 0;
}

static Item max_heap_extract_max(MaxHeap *heap) {
    Item max_item = heap->items[0];
    heap->items[0] = heap->items[--heap->size];
    max_heapify(heap, 0);
    return max_item;
}

static int compare_items_asc(const void *a, const void *b) {
    const Item *ia = (const Item *)a;
    const Item *ib = (const Item *)b;
    if (ia->price < ib->price) return -1;
    if (ia->price > ib->price) return 1;
    return 0;
}

Item *find_n_cheap_items(const Item *dataset, size_t dataset_size, size_t n, size_t *result_size) {
    if (dataset == NULL || result_size == NULL || n == 0) {
        if (result_size != NULL) *result_size = 0;
        return NULL;
    }

    if (n >= dataset_size) {
        Item *result = malloc(dataset_size * sizeof(Item));
        if (result == NULL) {
            *result_size = 0;
            return NULL;
        }
        memcpy(result, dataset, dataset_size * sizeof(Item));
        qsort(result, dataset_size, sizeof(Item), compare_items_asc);
        *result_size = dataset_size;
        return result;
    }

    MaxHeap heap = {0};
    heap.items = malloc(n * sizeof(Item));
    if (heap.items == NULL) {
        *result_size = 0;
        return NULL;
    }
    heap.capacity = n;

    for (size_t i = 0; i < n; i++) {
        heap.items[i] = dataset[i];
    }
    heap.size = n;
    build_max_heap(&heap);

    for (size_t i = n; i < dataset_size; i++) {
        if (dataset[i].price < heap.items[0].price) {
            heap.items[0] = dataset[i];
            max_heapify(&heap, 0);
        }
    }

    qsort(heap.items, heap.size, sizeof(Item), compare_items_asc);
    *result_size = heap.size;
    return heap.items;
}

int main(void) {
    Item dataset[] = {
        {1, 15.99},
        {2, 5.49},
        {3, 25.00},
        {4, 3.99},
        {5, 12.50},
        {6, 1.99},
        {7, 8.75},
        {8, 30.00},
        {9, 2.49},
        {10, 18.25}
    };
    size_t dataset_size = sizeof(dataset) / sizeof(dataset[0]);
    size_t n = 4;
    size_t result_size = 0;

    Item *cheapest = find_n_cheap_items(dataset, dataset_size, n, &result_size);
    if (cheapest == NULL) {
        fprintf(stderr, "Failed to find cheapest items\n");
        return EXIT_FAILURE;
    }

    printf("The %zu cheapest items:\n", result_size);
    for (size_t i = 0; i < result_size; i++) {
        printf("ID: %d, Price: %.2f\n", cheapest[i].id, cheapest[i].price);
    }

    free(cheapest);
    return EXIT_SUCCESS;
}
