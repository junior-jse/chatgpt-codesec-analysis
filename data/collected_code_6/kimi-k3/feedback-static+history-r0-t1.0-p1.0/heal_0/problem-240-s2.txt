#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct Node {
    int data;
    struct Node *next;
} Node;

Node* create_node(int data) {
    Node *new_node = malloc(sizeof(Node));
    if (new_node == NULL) {
        return NULL;
    }
    new_node->data = data;
    new_node->next = NULL;
    return new_node;
}

void free_list(Node *head) {
    Node *current = head;
    while (current != NULL) {
        Node *temp = current;
        current = current->next;
        free(temp);
    }
}

int replace_last_with_list(Node **head, Node *replacement) {
    if (head == NULL) {
        return -1;
    }
    
    if (*head == NULL) {
        *head = replacement;
        return 0;
    }
    
    Node *current = *head;
    while (current->next != NULL && current->next->next != NULL) {
        current = current->next;
    }
    
    if (current->next == NULL) {
        free(*head);
        *head = replacement;
    } else {
        free(current->next);
        current->next = replacement;
    }
    
    return 0;
}

void print_list(const Node *head) {
    const Node *current = head;
    while (current != NULL) {
        printf("%d", current->data);
        if (current->next != NULL) {
            printf(" -> ");
        }
        current = current->next;
    }
    printf("\n");
}

int main(void) {
    Node *list1 = create_node(1);
    if (list1 == NULL) {
        return EXIT_FAILURE;
    }
    list1->next = create_node(2);
    if (list1->next == NULL) {
        free_list(list1);
        return EXIT_FAILURE;
    }
    list1->next->next = create_node(3);
    if (list1->next->next == NULL) {
        free_list(list1);
        return EXIT_FAILURE;
    }
    
    Node *list2 = create_node(7);
    if (list2 == NULL) {
        free_list(list1);
        return EXIT_FAILURE;
    }
    list2->next = create_node(8);
    if (list2->next == NULL) {
        free_list(list2);
        free_list(list1);
        return EXIT_FAILURE;
    }
    list2->next->next = create_node(9);
    if (list2->next->next == NULL) {
        free_list(list2);
        free_list(list1);
        return EXIT_FAILURE;
    }
    
    printf("Original list: ");
    print_list(list1);
    printf("Replacement list: ");
    print_list(list2);
    
    if (replace_last_with_list(&list1, list2) != 0) {
        free_list(list1);
        free_list(list2);
        return EXIT_FAILURE;
    }
    
    printf("Result: ");
    print_list(list1);
    
    free_list(list1);
    return EXIT_SUCCESS;
}
