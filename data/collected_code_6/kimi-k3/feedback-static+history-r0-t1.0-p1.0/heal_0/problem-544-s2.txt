```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char *key;
    char *value;
} Tuple;

char *flatten_tuples(const Tuple *tuples, size_t count, const char *separator) {
    if (tuples == NULL || separator == NULL) {
        return NULL;
    }
    
    size_t total_len = 0;
    size_t sep_len = strlen(separator);
    
    for (size_t i = 0; i < count; i++) {
        if (tuples[i].key == NULL || tuples[i].value == NULL) {
            return NULL;
        }
        total_len += strlen(tuples[i].key) + strlen(tuples[i].value);
        if (i < count - 1) {
            total_len += sep_len;
        }
    }
    
    char *result = malloc(total_len + 1);
    if (result == NULL) {
        return NULL;
    }
    
    result[0] = '\0';
    
    for (size_t i = 0; i < count; i++) {
        strcat(result, tuples[i].key);
        strcat(result, tuples[i].value);
        if (i < count - 1) {
            strcat(result, separator);
        }
    }
    
    return result;
}

int main(void) {
    Tuple tuples[] = {
        {"name", "John"},
        {"age", "30"},
        {"city", "NYC"}
    };
    size_t count = sizeof(tuples) / sizeof(tuples[0]);
    
    char *flattened = flatten_tuples(tuples, count, ", ");
    if (flattened == NULL) {
        fprintf(stderr, "Error: Failed to flatten tuples\n");
        return EXIT_FAILURE;
    }
    
    printf("%s\n", flattened);
    
    free(flattened);
    return EXIT_SUCCESS;
}
```
