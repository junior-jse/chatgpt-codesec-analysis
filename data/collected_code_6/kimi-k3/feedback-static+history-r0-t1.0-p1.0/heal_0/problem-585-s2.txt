#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    int id;
    double price;
} Item;

typedef struct {
    Item *data;
    size_t size;
    size_t capacity;
} MinHeap;

static void swap_items(Item *a, Item *b)
{
    Item tmp = *a;
    *a = *b;
    *b = tmp;
}

static int heap_init(MinHeap *heap, size_t capacity)
{
    if (heap == NULL || capacity == 0) {
        return -1;
    }
    heap->data = malloc(capacity * sizeof(*heap->data));
    if (heap->data == NULL) {
        return -1;
    }
    heap->size = 0;
    heap->capacity = capacity;
    return 0;
}

static void heap_destroy(MinHeap *heap)
{
    if (heap != NULL) {
        free(heap->data);
        heap->data = NULL;
        heap->size = 0;
        heap->capacity = 0;
    }
}

static void heapify_down(MinHeap *heap, size_t idx)
{
    size_t smallest = idx;
    size_t left = 2 * idx + 1;
    size_t right = 2 * idx + 2;

    if (left < heap->size && heap->data[left].price < heap->data[smallest].price) {
        smallest = left;
    }
    if (right < heap->size && heap->data[right].price < heap->data[smallest].price) {
        smallest = right;
    }
    if (smallest != idx) {
        swap_items(&heap->data[idx], &heap->data[smallest]);
        heapify_down(heap, smallest);
    }
}

static void heapify_up(MinHeap *heap, size_t idx)
{
    while (idx > 0) {
        size_t parent = (idx - 1) / 2;
        if (heap->data[parent].price <= heap->data[idx].price) {
            break;
        }
        swap_items(&heap->data[parent], &heap->data[idx]);
        idx = parent;
    }
}

static int heap_push(MinHeap *heap, const Item *item)
{
    if (heap == NULL || item == NULL || heap->size >= heap->capacity) {
        return -1;
    }
    heap->data[heap->size] = *item;
    heapify_up(heap, heap->size);
    heap->size++;
    return 0;
}

static int heap_pop(MinHeap *heap, Item *out)
{
    if (heap == NULL || out == NULL || heap->size == 0) {
        return -1;
    }
    *out = heap->data[0];
    heap->size--;
    if (heap->size > 0) {
        heap->data[0] = heap->data[heap->size];
        heapify_down(heap, 0);
    }
    return 0;
}

static int compare_items_desc(const void *a, const void *b)
{
    const Item *ia = (const Item *)a;
    const Item *ib = (const Item *)b;
    if (ia->price < ib->price) {
        return 1;
    }
    if (ia->price > ib->price) {
        return -1;
    }
    return 0;
}

int find_n_expensive(const Item *dataset, size_t count, size_t n, Item *result)
{
    MinHeap heap;
    size_t i;
    size_t result_count;

    if (dataset == NULL || result == NULL || n == 0 || n > count) {
        return -1;
    }

    if (heap_init(&heap, n) != 0) {
        return -1;
    }

    for (i = 0; i < count; i++) {
        if (heap.size < n) {
            if (heap_push(&heap, &dataset[i]) != 0) {
                heap_destroy(&heap);
                return -1;
            }
        } else if (dataset[i].price > heap.data[0].price) {
            Item tmp;
            if (heap_pop(&heap, &tmp) != 0) {
                heap_destroy(&heap);
                return -1;
            }
            if (heap_push(&heap, &dataset[i]) != 0) {
                heap_destroy(&heap);
                return -1;
            }
        }
    }

    result_count = heap.size;
    for (i = 0; i < result_count; i++) {
        if (heap_pop(&heap, &result[i]) != 0) {
            heap_destroy(&heap);
            return -1;
        }
    }

    qsort(result, result_count, sizeof(*result), compare_items_desc);
    heap_destroy(&heap);
    return (int)result_count;
}

int main(void)
{
    Item dataset[] = {
        {1, 19.99},
        {2, 5.49},
        {3, 99.95},
        {4, 42.00},
        {5, 7.25},
        {6, 150.00},
        {7, 33.33},
        {8, 1.99}
    };
    size_t count = sizeof(dataset) / sizeof(dataset[0]);
    size_t n = 3;
    Item *result;
    int found;
    int i;

    result = malloc(n * sizeof(*result));
    if (result == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    found = find_n_expensive(dataset, count, n, result);
    if (found < 0) {
        fprintf(stderr, "Failed to find top items\n");
        free(result);
        return EXIT_FAILURE;
    }

    printf("Top %d expensive items:\n", found);
    for (i = 0; i < found; i++) {
        printf("ID: %d, Price: %.2f\n", result[i].id, result[i].price);
    }

    free(result);
    return EXIT_SUCCESS;
}
