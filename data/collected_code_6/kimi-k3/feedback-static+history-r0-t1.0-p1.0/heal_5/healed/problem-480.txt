#include <stdio.h>
#include <string.h>
#include <limits.h>
#include <stdlib.h>

#define BUFFER_SIZE 1024

char find_max_occurrence(const char *str, size_t *max_count)
{
    unsigned int counts[UCHAR_MAX + 1] = {0};
    size_t i;
    unsigned char max_char = 0;
    unsigned int max = 0;

    if (str == NULL || *str == '\0') {
        if (max_count != NULL) {
            *max_count = 0;
        }
        return '\0';
    }

    for (i = 0; str[i] != '\0'; i++) {
        unsigned char c = (unsigned char)str[i];
        counts[c]++;
        if (counts[c] > max) {
            max = counts[c];
            max_char = c;
        }
    }

    if (max_count != NULL) {
        *max_count = (size_t)max;
    }

    return (char)max_char;
}

int main(void)
{
    char *buffer = NULL;
    size_t buffer_size = 0;
    ssize_t read_len;
    size_t count = 0;
    char result;
    size_t len;

    printf("Enter a string: ");

    read_len = getline(&buffer, &buffer_size, stdin);
    
    if (read_len == -1) {
        fprintf(stderr, "Error reading input\n");
        free(buffer);
        return 1;
    }

    len = strcspn(buffer, "\n");
    buffer[len] = '\0';

    result = find_max_occurrence(buffer, &count);

    if (count > 0) {
        printf("Maximum occurring character: '%c' (appears %zu times)\n", result, count);
    } else {
        printf("String is empty\n");
    }

    free(buffer);
    return 0;
}
