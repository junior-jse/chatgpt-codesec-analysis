#include <stdio.h>
#include <string.h>
#include <stdlib.h>

int isReverse(const char *s1, const char *s2) {
    if (s1 == NULL || s2 == NULL) {
        return 0;
    }
    
    size_t len1 = strnlen(s1, 1024);
    size_t len2 = strnlen(s2, 1024);
    
    if (len1 != len2) {
        return 0;
    }
    
    for (size_t i = 0; i < len1; i++) {
        if (s1[i] != s2[len1 - 1 - i]) {
            return 0;
        }
    }
    
    return 1;
}

int countReversePairs(const char * const list[], int size) {
    int count = 0;
    
    if (list == NULL || size <= 0) {
        return 0;
    }
    
    for (int i = 0; i < size; i++) {
        if (list[i] == NULL) {
            continue;
        }
        for (int j = i + 1; j < size; j++) {
            if (list[j] == NULL) {
                continue;
            }
            if (isReverse(list[i], list[j])) {
                count++;
            }
        }
    }
    
    return count;
}

int main(void) {
    const char *list[] = {"abc", "def", "cba", "fed", "ghi"};
    int size = sizeof(list) / sizeof(list[0]);
    
    int result = countReversePairs(list, size);
    printf("Number of reverse pairs: %d\n", result);
    
    return 0;
}
