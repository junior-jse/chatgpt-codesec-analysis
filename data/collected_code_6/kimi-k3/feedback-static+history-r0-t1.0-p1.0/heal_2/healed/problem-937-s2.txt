#include <stdio.h>
#include <string.h>
#include <limits.h>
#include <stdlib.h>
#include <ctype.h>

char find_most_common_char(const char *str, int *count) {
    int char_count[UCHAR_MAX + 1] = {0};
    size_t i;
    int max_count = 0;
    char most_common = '\0';
    
    if (str == NULL || count == NULL) {
        if (count != NULL) {
            *count = 0;
        }
        return '\0';
    }
    
    for (i = 0; str[i] != '\0'; i++) {
        unsigned char c = (unsigned char)str[i];
        char_count[c]++;
    }
    
    for (i = 0; i <= UCHAR_MAX; i++) {
        if (char_count[i] > max_count) {
            max_count = char_count[i];
            most_common = (char)i;
        }
    }
    
    *count = max_count;
    return most_common;
}

int main(void) {
    char *buffer = NULL;
    size_t buffer_size = 0;
    ssize_t len;
    int count = 0;
    char result;
    
    printf("Enter a string: ");
    
    len = getline(&buffer, &buffer_size, stdin);
    
    if (len == -1) {
        fprintf(stderr, "Error reading input\n");
        free(buffer);
        return EXIT_FAILURE;
    }
    
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
        len--;
    }
    
    if (len == 0) {
        fprintf(stderr, "Empty string provided\n");
        free(buffer);
        return EXIT_FAILURE;
    }
    
    result = find_most_common_char(buffer, &count);
    
    if (count > 0) {
        if (isprint((unsigned char)result)) {
            printf("Most common character: '%c' (appears %d times)\n", result, count);
        } else {
            printf("Most common character: (non-printable, ASCII %d) (appears %d times)\n", 
                   (unsigned char)result, count);
        }
    } else {
        printf("No characters found\n");
    }
    
    free(buffer);
    return EXIT_SUCCESS;
}
