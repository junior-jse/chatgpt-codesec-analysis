#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node *next;
} Node;

Node *create_node(int data) {
    Node *node = (Node *)malloc(sizeof(Node));
    if (node == NULL) {
        exit(EXIT_FAILURE);
    }
    node->data = data;
    node->next = NULL;
    return node;
}

Node *subtract_lists(Node *list1, Node *list2) {
    Node dummy;
    Node *tail = &dummy;
    dummy.next = NULL;

    while (list1 != NULL) {
        if (list2 != NULL) {
            Node *node = create_node(list1->data - list2->data);
            tail->next = node;
            tail = node;
            list1 = list1->next;
            list2 = list2->next;
        } else {
            Node *node = create_node(list1->data);
            tail->next = node;
            tail = node;
            list1 = list1->next;
        }
    }

    return dummy.next;
}

void free_list(Node *head) {
    Node *current = head;
    while (current != NULL) {
        Node *next = current->next;
        free(current);
        current = next;
    }
}

void print_list(Node *head) {
    Node *current = head;
    while (current != NULL) {
        printf("%d ", current->data);
        current = current->next;
    }
    printf("\n");
}

Node *append(Node *head, int data) {
    if (head == NULL) {
        return create_node(data);
    }
    Node *current = head;
    while (current->next != NULL) {
        current = current->next;
    }
    current->next = create_node(data);
    return head;
}

int main(void) {
    Node *list1 = NULL;
    list1 = append(list1, 10);
    list1 = append(list1, 20);
    list1 = append(list1, 30);
    list1 = append(list1, 40);

    Node *list2 = NULL;
    list2 = append(list2, 1);
    list2 = append(list2, 2);
    list2 = append(list2, 3);

    Node *result = subtract_lists(list1, list2);

    print_list(result);

    free_list(list1);
    free_list(list2);
    free_list(result);

    return 0;
}
