#include <stdio.h>
#include <stdlib.h>

void rotate_left(int *list, size_t size, size_t k) {
    if (size == 0) {
        return;
    }

    k = k % size;

    if (k == 0) {
        return;
    }

    int *temp = (int *)malloc(k * sizeof(int));
    if (temp == NULL) {
        return;
    }

    for (size_t i = 0; i < k; i++) {
        temp[i] = list[i];
    }

    for (size_t i = 0; i < size - k; i++) {
        list[i] = list[i + k];
    }

    for (size_t i = 0; i < k; i++) {
        list[size - k + i] = temp[i];
    }

    free(temp);
}

int main(void) {
    int arr[] = {1, 2, 3, 4, 5, 6, 7};
    size_t size = sizeof(arr) / sizeof(arr[0]);
    size_t k = 3;

    rotate_left(arr, size, k);

    for (size_t i = 0; i < size; i++) {
        printf("%d ", arr[i]);
    }

    printf("\n");

    return 0;
}
