#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *next;
};

struct Node *create_node(int data) {
    struct Node *new_node = (struct Node *)malloc(sizeof(struct Node));
    if (new_node == NULL) {
        exit(EXIT_FAILURE);
    }
    new_node->data = data;
    new_node->next = NULL;
    return new_node;
}

struct Node *add_two_lists(struct Node *l1, struct Node *l2) {
    struct Node dummy;
    dummy.next = NULL;
    struct Node *current = &dummy;
    int carry = 0;

    while (l1 != NULL || l2 != NULL || carry != 0) {
        int sum = carry;
        if (l1 != NULL) {
            sum += l1->data;
            l1 = l1->next;
        }
        if (l2 != NULL) {
            sum += l2->data;
            l2 = l2->next;
        }
        carry = sum / 10;
        struct Node *new_node = create_node(sum % 10);
        current->next = new_node;
        current = current->next;
    }

    return dummy.next;
}

void free_list(struct Node *head) {
    struct Node *temp;
    while (head != NULL) {
        temp = head;
        head = head->next;
        free(temp);
    }
}

void print_list(struct Node *head) {
    while (head != NULL) {
        printf("%d ", head->data);
        head = head->next;
    }
    printf("\n");
}

int main(void) {
    struct Node *l1 = create_node(9);
    l1->next = create_node(9);
    l1->next->next = create_node(9);

    struct Node *l2 = create_node(1);
    l2->next = create_node(8);

    struct Node *result = add_two_lists(l1, l2);

    print_list(result);

    free_list(l1);
    free_list(l2);
    free_list(result);

    return 0;
}
