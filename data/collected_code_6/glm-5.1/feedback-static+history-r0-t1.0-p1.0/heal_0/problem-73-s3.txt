#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

char **split_string_regex(const char *str, const char *pattern, size_t *count) {
    regex_t regex;
    regmatch_t match;
    int ret;
    size_t capacity = 10;
    size_t size = 0;
    char **result = NULL;
    const char *cursor = str;
    size_t remaining_len;

    if (str == NULL || pattern == NULL || count == NULL) {
        return NULL;
    }

    *count = 0;

    ret = regcomp(&regex, pattern, REG_EXTENDED);
    if (ret != 0) {
        return NULL;
    }

    result = (char **)malloc(capacity * sizeof(char *));
    if (result == NULL) {
        regfree(&regex);
        return NULL;
    }

    while (regexec(&regex, cursor, 1, &match, 0) == 0) {
        if (size >= capacity) {
            capacity *= 2;
            char **temp = (char **)realloc(result, capacity * sizeof(char *));
            if (temp == NULL) {
                for (size_t i = 0; i < size; i++) {
                    free(result[i]);
                }
                free(result);
                regfree(&regex);
                return NULL;
            }
            result = temp;
        }

        size_t token_len = match.rm_so;
        if (token_len > 0) {
            result[size] = (char *)malloc(token_len + 1);
            if (result[size] == NULL) {
                for (size_t i = 0; i < size; i++) {
                    free(result[i]);
                }
                free(result);
                regfree(&regex);
                return NULL;
            }
            memcpy(result[size], cursor, token_len);
            result[size][token_len] = '\0';
            size++;
        }

        cursor += match.rm_eo;
    }

    remaining_len = strlen(cursor);
    if (remaining_len > 0) {
        if (size >= capacity) {
            capacity += 1;
            char **temp = (char **)realloc(result, capacity * sizeof(char *));
            if (temp == NULL) {
                for (size_t i = 0; i < size; i++) {
                    free(result[i]);
                }
                free(result);
                regfree(&regex);
                return NULL;
            }
            result = temp;
        }
        result[size] = (char *)malloc(remaining_len + 1);
        if (result[size] == NULL) {
            for (size_t i = 0; i < size; i++) {
                free(result[i]);
            }
            free(result);
            regfree(&regex);
            return NULL;
        }
        memcpy(result[size], cursor, remaining_len);
        result[size][remaining_len] = '\0';
        size++;
    }

    regfree(&regex);

    char **final_result = (char **)realloc(result, (size + 1) * sizeof(char *));
    if (final_result == NULL) {
        for (size_t i = 0; i < size; i++) {
            free(result[i]);
        }
        free(result);
        return NULL;
    }
    result = final_result;
    result[size] = NULL;

    *count = size;
    return result;
}

void free_split_result(char **result) {
    if (result == NULL) {
        return;
    }
    for (size_t i = 0; result[i] != NULL; i++) {
        free(result[i]);
    }
    free(result);
}

int main(void) {
    const char *str = "hello;world|test,string";
    const char *pattern = "[;|,]";
    size_t count = 0;

    char **tokens = split_string_regex(str, pattern, &count);
    if (tokens == NULL) {
        return EXIT_FAILURE;
    }

    for (size_t i = 0; i < count; i++) {
        printf("%s\n", tokens[i]);
    }

    free_split_result(tokens);

    return EXIT_SUCCESS;
}
