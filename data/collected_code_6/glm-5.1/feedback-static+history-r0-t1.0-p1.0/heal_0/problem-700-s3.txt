```c
#include <stdio.h>
#include <stdlib.h>

int count_in_range(int *list, size_t size, int lower_bound, int upper_bound) {
    if (list == NULL) {
        return 0;
    }

    int count = 0;
    for (size_t i = 0; i < size; i++) {
        if (list[i] >= lower_bound && list[i] <= upper_bound) {
            count++;
        }
    }
    return count;
}

int main(void) {
    int data[] = {1, 5, 12, 15, 20, 25, 30};
    size_t data_size = sizeof(data) / sizeof(data[0]);

    int lower = 10;
    int upper = 25;

    int result = count_in_range(data, data_size, lower, upper);

    printf("%d\n", result);

    return 0;
}
```
