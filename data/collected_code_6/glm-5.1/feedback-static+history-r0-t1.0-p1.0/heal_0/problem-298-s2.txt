#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int is_list;
    union {
        int value;
        struct {
            struct Node *items;
            size_t count;
        } list;
    } data;
    struct Node *next;
} Node;

typedef struct {
    Node *head;
    size_t count;
} NestedList;

typedef struct {
    int *values;
    size_t count;
    size_t capacity;
} FlatList;

Node *create_int_node(int val) {
    Node *n = (Node *)malloc(sizeof(Node));
    if (!n) exit(EXIT_FAILURE);
    n->is_list = 0;
    n->data.value = val;
    n->next = NULL;
    return n;
}

Node *create_list_node(Node *items, size_t count) {
    Node *n = (Node *)malloc(sizeof(Node));
    if (!n) exit(EXIT_FAILURE);
    n->is_list = 1;
    n->data.list.items = items;
    n->data.list.count = count;
    n->next = NULL;
    return n;
}

NestedList create_nested_list(Node *head, size_t count) {
    NestedList nl;
    nl.head = head;
    nl.count = count;
    return nl;
}

FlatList create_flat_list() {
    FlatList fl;
    fl.values = (int *)malloc(sizeof(int) * 16);
    if (!fl.values) exit(EXIT_FAILURE);
    fl.count = 0;
    fl.capacity = 16;
    return fl;
}

void flat_list_append(FlatList *fl, int val) {
    if (fl->count == fl->capacity) {
        fl->capacity *= 2;
        fl->values = (int *)realloc(fl->values, sizeof(int) * fl->capacity);
        if (!fl->values) exit(EXIT_FAILURE);
    }
    fl->values[fl->count++] = val;
}

int exists_in_list(const FlatList *fl, int val) {
    for (size_t i = 0; i < fl->count; i++) {
        if (fl->values[i] == val) return 1;
    }
    return 0;
}

void flatten_nested(const Node *node, FlatList *fl) {
    while (node) {
        if (node->is_list) {
            for (size_t i = 0; i < node->data.list.count; i++) {
                flatten_nested(&node->data.list.items[i], fl);
            }
        } else {
            flat_list_append(fl, node->data.value);
        }
        node = node->next;
    }
}

FlatList find_common_elements(const NestedList *nl1, const NestedList *nl2) {
    FlatList fl1 = create_flat_list();
    flatten_nested(nl1->head, &fl1);

    FlatList fl2 = create_flat_list();
    flatten_nested(nl2->head, &fl2);

    FlatList result = create_flat_list();
    
    for (size_t i = 0; i < fl1.count; i++) {
        if (exists_in_list(&fl2, fl1.values[i]) && !exists_in_list(&result, fl1.values[i])) {
            flat_list_append(&result, fl1.values[i]);
        }
    }

    free(fl1.values);
    free(fl2.values);

    return result;
}

void free_nested_nodes(Node *node) {
    while (node) {
        Node *next = node->next;
        if (node->is_list) {
            for (size_t i = 0; i < node->data.list.count; i++) {
                free_nested_nodes(&node->data.list.items[i]);
            }
            free(node->data.list.items);
        }
        node = next;
    }
}

int main(void) {
    Node *sub1_head = create_int_node(1);
    sub1_head->next = create_int_node(2);
    
    Node *sub1 = create_list_node(sub1_head, 1);
    sub1->next = create_int_node(3);
    sub1->next->next = create_int_node(4);
    
    NestedList nl1 = create_nested_list(sub1, 2);

    Node *sub2_head = create_int_node(2);
    sub2_head->next = create_int_node(5);
    
    Node *sub2 = create_list_node(sub2_head, 1);
    sub2->next = create_int_node(6);
    sub2->next->next = create_int_node(3);
    
    NestedList nl2 = create_nested_list(sub2, 2);

    FlatList common = find_common_elements(&nl1, &nl2);

    for (size_t i = 0; i < common.count; i++) {
        printf("%d ", common.values[i]);
    }
    printf("\n");

    free_nested_nodes(nl1.head);
    free_nested_nodes(nl2.head);
    free(common.values);

    return 0;
}
