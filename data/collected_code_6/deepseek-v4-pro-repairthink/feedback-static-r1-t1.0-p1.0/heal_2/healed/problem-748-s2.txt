#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

char *add_spaces_before_capitals(const char *input) {
    if (input == NULL) {
        return NULL;
    }

    size_t input_len = strlen(input);
    size_t spaces_needed = 0;
    const char *p = input;

    while (*p) {
        if (isupper((unsigned char)*p) && p != input) {
            const char *prev = p - 1;
            if (prev >= input && *prev != ' ') {
                spaces_needed++;
            }
        }
        p++;
    }

    size_t result_len = input_len + spaces_needed + 1;
    char *result = malloc(result_len);
    if (result == NULL) {
        return NULL;
    }

    const char *src = input;
    char *dst = result;

    while (*src) {
        if (isupper((unsigned char)*src) && src != input) {
            const char *prev = src - 1;
            if (prev >= input && *prev != ' ') {
                if (dst > result && *(dst - 1) != ' ') {
                    *dst++ = ' ';
                } else if (dst == result) {
                    *dst++ = ' ';
                }
            }
        }
        *dst++ = *src++;
    }
    *dst = '\0';

    return result;
}

int main(void) {
    const char *test1 = "HelloWorldExample";

    const char *test3 = "already separated words";
    const char *test4 = "";
    const char *test5 = "UPPERCASE";
    const char *test6 = "noSpacesHere";
