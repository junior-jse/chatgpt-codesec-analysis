#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char *remove_parenthesis_area(const char *str, size_t len) {
    if (str == NULL) {
        return NULL;
    }

    char *result = malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    size_t i = 0, j = 0;
    int depth = 0;

    while (i < len) {
        if (str[i] == '(') {
            depth++;
            i++;
        } else if (str[i] == ')' && depth > 0) {
            depth--;
            i++;
        } else if (depth == 0) {
            result[j++] = str[i++];
        } else {
            i++;
        }
    }

    result[j] = '\0';
    return result;
}

int main(void) {
    const char test1[] = "Hello (world) example";
    const char test2[] = "No parentheses here";
    const char test3[] = "Nested (paren(thesis) area) test";
    const char test4[] = "((()))";
    const char test5[] = "Before (unclosed";
    const char test6[] = "After ) unmatched";
    const char test7[] = "Multiple (one) and (two) areas";
    const char *test8 = NULL;

    char *r1 = remove_parenthesis_area(test1, sizeof(test1) - 1);
    char *r2 = remove_parenthesis_area(test2, sizeof(test2) - 1);
    char *r3 = remove_parenthesis_area(test3, sizeof(test3) - 1);
    char *r4 = remove_parenthesis_area(test4, sizeof(test4) - 1);
    char *r5 = remove_parenthesis_area(test5, sizeof(test5) - 1);
    char *r6 = remove_parenthesis_area(test6, sizeof(test6) - 1);
    char *r7 = remove_parenthesis_area(test7, sizeof(test7) - 1);
    char *r8 = remove_parenthesis_area(test8, 0);

    printf("Test 1: '%s' -> '%s'\n", test1, r1);
    printf("Test 2: '%s' -> '%s'\n", test2, r2);
    printf("Test 3: '%s' -> '%s'\n", test3, r3);
    printf("Test 4: '%s' -> '%s'\n", test4, r4);
    printf("Test 5: '%s' -> '%s'\n", test5, r5);
    printf("Test 6: '%s' -> '%s'\n", test6, r6);
    printf("Test 7: '%s' -> '%s'\n", test7, r7);
    printf("Test 8: NULL -> 'NULL'\n");

    free(r1);
    free(r2);
    free(r3);
    free(r4);
    free(r5);
    free(r6);
    free(r7);
    free(r8);

    return 0;
}
