#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

const int* find_max_sum_list(int** lists, const int* sizes, int num_lists) {
    if (lists == NULL || sizes == NULL || num_lists <= 0) {
        return NULL;
    }

    int max_sum = INT_MIN;
    int max_index = -1;

    for (int i = 0; i < num_lists; i++) {
        if (lists[i] == NULL || sizes[i] <= 0) {
            continue;
        }

        int current_sum = 0;
        for (int j = 0; j < sizes[i]; j++) {
            current_sum += lists[i][j];
        }

        if (current_sum > max_sum) {
            max_sum = current_sum;
            max_index = i;
        }
    }

    if (max_index == -1) {
        return NULL;
    }

    return lists[max_index];
}

int main() {
    int list1[] = {1, 2, 3, 4, 5};
    int list2[] = {-5, -4, -3, -2, -1};
    int list3[] = {10, 20};
    int list4[] = {7, 8, 9, 10};

    int* lists[] = {list1, list2, list3, list4};
    const int sizes[] = {5, 5, 2, 4};
    int num_lists = 4;

    const int* max_list = find_max_sum_list(lists, sizes, num_lists);

    if (max_list != NULL) {
        int max_idx = -1;
        for (int i = 0; i < num_lists; i++) {
            if (max_list == lists[i]) {
                max_idx = i;
                break;
            }
        }

        if (max_idx != -1) {
            printf("Maximum sum list: ");
            for (int i = 0; i < sizes[max_idx]; i++) {
                printf("%d ", max_list[i]);
            }
            printf("\n");
        }
    } else {
        printf("No valid list found\n");
    }

    return 0;
}
