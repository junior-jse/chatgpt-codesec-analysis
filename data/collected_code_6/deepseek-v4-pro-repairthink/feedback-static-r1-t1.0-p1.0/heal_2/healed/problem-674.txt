#define _POSIX_C_SOURCE 200809L
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char **words;
    int count;
    int capacity;
} WordList;

static void init_wordlist(WordList *wl) {
    wl->words = NULL;
    wl->count = 0;
    wl->capacity = 0;
}

static void free_wordlist(WordList *wl) {
    for (int i = 0; i < wl->count; i++) {
        free(wl->words[i]);
    }
    free(wl->words);
    wl->words = NULL;
    wl->count = 0;
    wl->capacity = 0;
}

static int is_duplicate(const WordList *wl, const char *word) {
    for (int i = 0; i < wl->count; i++) {
        if (strcmp(wl->words[i], word) == 0) {
            return 1;
        }
    }
    return 0;
}

static int add_word(WordList *wl, const char *word) {
    if (wl->count >= wl->capacity) {
        int newcap = (wl->capacity == 0) ? 16 : wl->capacity * 2;
        char **tmp = realloc(wl->words, newcap * sizeof(char *));
        if (tmp == NULL) {
            return -1;
        }
        wl->words = tmp;
        wl->capacity = newcap;
    }
    wl->words[wl->count] = strdup(word);
    if (wl->words[wl->count] == NULL) {
        return -1;
    }
    wl->count++;
    return 0;
}

void remove_duplicate_words(char *str) {
    if (str == NULL || *str == '\0') {
        return;
    }

    WordList seen;
    init_wordlist(&seen);

    size_t len = strlen(str);
    char *result = malloc(len + 1);
    if (result == NULL) {
        free_wordlist(&seen);
        return;
    }
    result[0] = '\0';
    size_t offset = 0;
    int first = 1;

    char *token = strtok(str, " \t\n\r");
    while (token != NULL) {
        if (!is_duplicate(&seen, token)) {
            if (add_word(&seen, token) != 0) {
                free(result);
                free_wordlist(&seen);
                return;
            }
            if (first) {
                int written = snprintf(result + offset, len + 1 - offset, "%s", token);
                if (written < 0) {
                    free(result);
                    free_wordlist(&seen);
                    return;
                }
                offset += written;
                first = 0;
            } else {
                int written = snprintf(result + offset, len + 1 - offset, " %s", token);
                if (written < 0) {
                    free(result);
                    free_wordlist(&seen);
                    return;
                }
                offset += written;
            }
        }
        token = strtok(NULL, " \t\n\r");
    }

    if (offset < len + 1) {
        memmove(str, result, offset + 1);
    }
    free(result);
    free_wordlist(&seen);
}

int main(void) {
    char str[] = "hello world hello again world test hello";
    printf("Original: %s\n", str);
    remove_duplicate_words(str);
    printf("After removing duplicates: %s\n", str);
    return 0;
}
