#include <stdio.h>
#include <stdlib.h>
#include <string.h>

void reverse_string(char *str, size_t max_size) {
    if (str == NULL || max_size == 0) return;
    size_t length = strnlen(str, max_size);
    if (length >= max_size) length = max_size - 1;
    for (size_t i = 0; i < length / 2; i++) {
        char temp = str[i];
        str[i] = str[length - 1 - i];
        str[length - 1 - i] = temp;
    }
}

void reverse_strings(char **list, size_t *sizes, size_t count) {
    if (list == NULL || sizes == NULL || count == 0) return;
    for (size_t i = 0; i < count; i++) {
        if (list[i] != NULL && sizes[i] > 0) {
            reverse_string(list[i], sizes[i]);
        }
    }
}

int main(void) {
    const char *initial[] = {
        "hello",
        "world",
        "example",
        "programming"
    };
    size_t num_strings = sizeof(initial) / sizeof(initial[0]);

    char **list = malloc(num_strings * sizeof(char *));
    size_t *sizes = malloc(num_strings * sizeof(size_t));
    if (list == NULL || sizes == NULL) {
        free(list);
        free(sizes);
        return 1;
    }

    for (size_t i = 0; i < num_strings; i++) {
        size_t len = strlen(initial[i]);
        sizes[i] = len + 1;
        list[i] = malloc(sizes[i]);
        if (list[i] == NULL) {
            for (size_t j = 0; j < i; j++) free(list[j]);
            free(list);
            free(sizes);
            return 1;
        }
        snprintf(list[i], sizes[i], "%s", initial[i]);
    }

    reverse_strings(list, sizes, num_strings);

    for (size_t i = 0; i < num_strings; i++) {
        printf("%s\n", list[i]);
        free(list[i]);
    }
    free(list);
    free(sizes);

    return 0;
}
