#include <stdio.h>
#include <stdlib.h>

int count_lists(void **lists, int num_lists) {
    int count = 0;
    for (int i = 0; i < num_lists; i++) {
        if (lists[i] != NULL) {
            count++;
        }
    }
    return count;
}

int main(void) {
    void *lists[3];
    int a = 0, b = 0, c = 0;
    void *list1 = &a;
    void *list2 = &b;
    void *list3 = &c;

    lists[0] = list1;
    lists[1] = list2;
    lists[2] = list3;

    printf("Number of lists: %d\n", count_lists(lists, 3));

    lists[1] = NULL;
    printf("Number of lists after removing one: %d\n", count_lists(lists, 3));

    return 0;
}
