#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

void find_max_product_pair(const int arr[], int n, int *num1, int *num2) {
    if (n < 2) {
        *num1 = 0;
        *num2 = 0;
        return;
    }

    int max1 = INT_MIN, max2 = INT_MIN;
    int min1 = INT_MAX, min2 = INT_MAX;

    for (int i = 0; i < n; i++) {
        if (arr[i] > max1) {
            max2 = max1;
            max1 = arr[i];
        } else if (arr[i] > max2) {
            max2 = arr[i];
        }

        if (arr[i] < min1) {
            min2 = min1;
            min1 = arr[i];
        } else if (arr[i] < min2) {
            min2 = arr[i];
        }
    }

    if ((long long)max1 * max2 >= (long long)min1 * min2) {
        *num1 = max1;
        *num2 = max2;
    } else {
        *num1 = min1;
        *num2 = min2;
    }
}

int main(void) {
    int arr[] = {-10, -3, 5, 6, -2};
    int n = sizeof(arr) / sizeof(arr[0]);
    int num1, num2;

    find_max_product_pair(arr, n, &num1, &num2);
    printf("%d %d\n", num1, num2);

    return 0;
}
