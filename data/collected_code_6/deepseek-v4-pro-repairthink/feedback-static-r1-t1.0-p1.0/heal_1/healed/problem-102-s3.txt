#include <stdio.h>
#include <stdlib.h>
#include <ctype.h>

char *snake_to_camel(const char *snake) {
    if (snake == NULL) {
        return NULL;
    }

    size_t bufsize = 16;
    char *camel = (char *)malloc(bufsize);
    if (camel == NULL) {
        return NULL;
    }

    size_t j = 0;
    int capitalize_next = 0;
    const char *s = snake;

    if (*s == '_') {
        if (j + 2 > bufsize) {
            bufsize *= 2;
            char *tmp = (char *)realloc(camel, bufsize);
            if (tmp == NULL) {
                free(camel);
                return NULL;
            }
            camel = tmp;
        }
        camel[j++] = '_';
        s++;
    }

    for (size_t i = 0; s[i] != '\0'; i++) {
        char c = s[i];
        if (c == '_') {
            capitalize_next = 1;
        } else {
            char out;
            if (capitalize_next) {
                out = (char)toupper((unsigned char)c);
                capitalize_next = 0;
            } else {
                out = c;
            }

            if (j + 2 > bufsize) {
                bufsize *= 2;
                char *tmp = (char *)realloc(camel, bufsize);
                if (tmp == NULL) {
                    free(camel);
                    return NULL;
                }
                camel = tmp;
            }
            camel[j++] = out;
        }
    }

    if (j + 1 > bufsize) {
        bufsize = j + 1;
        char *tmp = (char *)realloc(camel, bufsize);
        if (tmp == NULL) {
            free(camel);
            return NULL;
        }
        camel = tmp;
    }
    camel[j] = '\0';
    return camel;
}

int main(void) {
