#include <stdio.h>
#include <string.h>
#include <limits.h>

char find_first_repeated_char(const char *str) {
    int count[UCHAR_MAX + 1] = {0};
    int first_index[UCHAR_MAX + 1];
    size_t i;
    
    if (str == NULL) {
        return '\0';
    }
    
    for (i = 0; i <= UCHAR_MAX; i++) {
        first_index[i] = -1;
    }
    
    for (i = 0; str[i] != '\0'; i++) {
        unsigned char uc = (unsigned char)str[i];
        count[uc]++;
        if (first_index[uc] == -1) {
            first_index[uc] = (int)i;
        }
    }
    
    int min_index = INT_MAX;
    char result = '\0';
    
    for (i = 0; i <= UCHAR_MAX; i++) {
        if (count[i] > 1 && first_index[i] < min_index) {
            min_index = first_index[i];
            result = (char)i;
        }
    }
    
    return result;
}

int main(void) {
    const char *test_cases[] = {
        "hello",
        "abcdef",
        "aabbcc",
        "",
        "programming",
        "racecar",
        "xyzxyz"
    };
    
    size_t num_tests = sizeof(test_cases) / sizeof(test_cases[0]);
    
    for (size_t i = 0; i < num_tests; i++) {
        char result = find_first_repeated_char(test_cases[i]);
        if (result != '\0') {
            printf("String: \"%s\", First repeated character: '%c'\n", 
                   test_cases[i], result);
        } else {
            printf("String: \"%s\", No repeated character found\n", 
                   test_cases[i]);
        }
    }
    
    return 0;
}
