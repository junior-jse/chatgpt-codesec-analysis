```c
#include <stdio.h>
#include <string.h>
#include <stdbool.h>

bool all_unique_chars(const char *str) {
    if (str == NULL) {
        return true;
    }
    
    bool seen[256] = {false};
    
    while (*str) {
        unsigned char c = (unsigned char)*str;
        if (seen[c]) {
            return false;
        }
        seen[c] = true;
        str++;
    }
    
    return true;
}

int main(void) {
    const char *test_strings[] = {
        "abcdefg",
        "abcdefa",
        "",
        "a",
        "aa",
        "abc defg",
        NULL
    };
    
    for (int i = 0; test_strings[i] != NULL; i++) {
        printf("\"%s\" -> %s\n", 
               test_strings[i], 
               all_unique_chars(test_strings[i]) ? "unique" : "not unique");
    }
    
    return 0;
}
```
