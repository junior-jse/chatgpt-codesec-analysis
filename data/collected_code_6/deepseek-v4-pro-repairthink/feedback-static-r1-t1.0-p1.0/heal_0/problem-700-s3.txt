#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node *next;
} Node;

Node *create_node(int data) {
    Node *new_node = (Node *)malloc(sizeof(Node));
    if (new_node == NULL) {
        return NULL;
    }
    new_node->data = data;
    new_node->next = NULL;
    return new_node;
}

int count_in_range(Node *head, int low, int high) {
    int count = 0;
    Node *current = head;
    
    if (low > high) {
        return 0;
    }
    
    while (current != NULL) {
        if (current->data >= low && current->data <= high) {
            count++;
        }
        current = current->next;
    }
    
    return count;
}

void free_list(Node *head) {
    Node *current = head;
    Node *temp;
    
    while (current != NULL) {
        temp = current;
        current = current->next;
        free(temp);
    }
}

int main(void) {
    Node *head = NULL;
    Node *tail = NULL;
    int values[] = {5, 12, 3, 8, 15, 7, 20, 1, 9, 14};
    int i;
    int low = 5;
    int high = 15;
    int result;
    
    for (i = 0; i < 10; i++) {
        Node *new_node = create_node(values[i]);
        if (new_node == NULL) {
            free_list(head);
            return 1;
        }
        if (head == NULL) {
            head = new_node;
            tail = new_node;
        } else {
            tail->next = new_node;
            tail = new_node;
        }
    }
    
    result = count_in_range(head, low, high);
    
    printf("Count of elements in range [%d, %d]: %d\n", low, high, result);
    
    free_list(head);
    
    return 0;
}
