#include <stdio.h>
#include <stdlib.h>
#include <string.h>

static int get_max(int arr[], int n) {
    int max = arr[0];
    for (int i = 1; i < n; i++) {
        if (arr[i] > max) {
            max = arr[i];
        }
    }
    return max;
}

static void counting_sort(int arr[], int n, int exp) {
    int *output = (int *)malloc(n * sizeof(int));
    if (output == NULL) {
        exit(EXIT_FAILURE);
    }

    int count[10] = {0};

    for (int i = 0; i < n; i++) {
        count[(arr[i] / exp) % 10]++;
    }

    for (int i = 1; i < 10; i++) {
        count[i] += count[i - 1];
    }

    for (int i = n - 1; i >= 0; i--) {
        int digit = (arr[i] / exp) % 10;
        output[count[digit] - 1] = arr[i];
        count[digit]--;
    }

    memcpy(arr, output, n * sizeof(int));
    free(output);
}

void radix_sort(int arr[], int n) {
    if (arr == NULL || n <= 1) {
        return;
    }

    int max = get_max(arr, n);

    for (int exp = 1; max / exp > 0; exp *= 10) {
        counting_sort(arr, n, exp);
    }
}

int main(void) {
    int arr[] = {170, 45, 75, 90, 802, 24, 2, 66};
    int n = (int)(sizeof(arr) / sizeof(arr[0]));

    radix_sort(arr, n);

    for (int i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");

    return 0;
}
