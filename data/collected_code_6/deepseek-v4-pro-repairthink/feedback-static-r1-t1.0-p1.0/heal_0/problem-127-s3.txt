#include <stdio.h>
#include <limits.h>
#include <stdint.h>

int multiply(int a, int b) {
    int result = 0;
    int negative = 0;

    if (a == 0 || b == 0) {
        return 0;
    }

    if (a < 0) {
        negative = !negative;
        if (a == INT_MIN) {
            if (b == 1) return INT_MIN;
            if (b == -1) return INT_MAX;
        }
        a = -a;
    }
    if (b < 0) {
        negative = !negative;
        if (b == INT_MIN) {
            if (a == 1) return INT_MIN;
            if (a == -1) return INT_MAX;
        }
        b = -b;
    }

    while (b > 0) {
        if (b & 1) {
            result += a;
        }
        a <<= 1;
        b >>= 1;
    }

    return negative ? -result : result;
}

int main(void) {
    int x = 12;
    int y = 13;
    printf("%d\n", multiply(x, y));
    return 0;
}
