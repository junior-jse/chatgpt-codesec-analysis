#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char **shortlist_words(char **words, int word_count, size_t n, int *result_count) {
    if (words == NULL || result_count == NULL) {
        return NULL;
    }

    char **result = NULL;
    int count = 0;

    for (int i = 0; i < word_count; i++) {
        if (words[i] != NULL && strlen(words[i]) > n) {
            char **temp = realloc(result, (count + 1) * sizeof(char *));
            if (temp == NULL) {
                free(result);
                return NULL;
            }
            result = temp;
            result[count++] = words[i];
        }
    }

    *result_count = count;
    return result;
}

int main(void) {
    char *words[] = {"apple", "banana", "kiwi", "strawberry", "pear", "grapefruit"};
    int word_count = sizeof(words) / sizeof(words[0]);
    int result_count = 0;

    char **shortlisted = shortlist_words(words, word_count, 5, &result_count);

    if (shortlisted != NULL || result_count == 0) {
        for (int i = 0; i < result_count; i++) {
            printf("%s\n", shortlisted[i]);
        }
    }

    free(shortlisted);

    return 0;
}
