#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct WordCount {
    char *word;
    int count;
    struct WordCount *next;
} WordCount;

static WordCount *find_word(WordCount *head, const char *word) {
    WordCount *current = head;
    while (current != NULL) {
        if (strcmp(current->word, word) == 0) {
            return current;
        }
        current = current->next;
    }
    return NULL;
}

static WordCount *add_word(WordCount *head, const char *word) {
    WordCount *new_node = (WordCount *)malloc(sizeof(WordCount));
    if (new_node == NULL) {
        return NULL;
    }
    new_node->word = (char *)malloc(strlen(word) + 1);
    if (new_node->word == NULL) {
        free(new_node);
        return NULL;
    }
    strcpy(new_node->word, word);
    new_node->count = 1;
    new_node->next = head;
    return new_node;
}

static void free_list(WordCount *head) {
    WordCount *current = head;
    while (current != NULL) {
        WordCount *next = current->next;
        free(current->word);
        free(current);
        current = next;
    }
}

char *most_frequent_word(char *strings[], int size) {
    if (strings == NULL || size <= 0) {
        return NULL;
    }

    WordCount *head = NULL;
    for (int i = 0; i < size; i++) {
        if (strings[i] == NULL) {
            continue;
        }
        WordCount *found = find_word(head, strings[i]);
        if (found != NULL) {
            found->count++;
        } else {
            WordCount *new_head = add_word(head, strings[i]);
            if (new_head == NULL) {
                free_list(head);
                return NULL;
            }
            head = new_head;
        }
    }

    if (head == NULL) {
        return NULL;
    }

    WordCount *current = head;
    WordCount *max_node = head;
    while (current != NULL) {
        if (current->count > max_node->count) {
            max_node = current;
        }
        current = current->next;
    }

    char *result = (char *)malloc(strlen(max_node->word) + 1);
    if (result == NULL) {
        free_list(head);
        return NULL;
    }
    strcpy(result, max_node->word);

    free_list(head);
    return result;
}

int main(void) {
    char *strings[] = {"apple", "banana", "apple", "orange", "banana", "apple", "grape"};
    int size = sizeof(strings) / sizeof(strings[0]);
    
    char *result = most_frequent_word(strings, size);
    if (result != NULL) {
        printf("%s\n", result);
        free(result);
    }

    return 0;
}
