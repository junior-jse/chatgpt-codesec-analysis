#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node* next;
} Node;

Node* create_node(int data) {
    Node* new_node = (Node*)malloc(sizeof(Node));
    if (new_node == NULL) {
        return NULL;
    }
    new_node->data = data;
    new_node->next = NULL;
    return new_node;
}

Node* append_node(Node* head, int data) {
    Node* new_node = create_node(data);
    if (new_node == NULL) {
        return NULL;
    }
    if (head == NULL) {
        return new_node;
    }
    Node* current = head;
    while (current->next != NULL) {
        current = current->next;
    }
    current->next = new_node;
    return head;
}

void free_list(Node* head) {
    Node* current = head;
    while (current != NULL) {
        Node* temp = current;
        current = current->next;
        free(temp);
    }
}

Node* list_from_array(int arr[], int size) {
    Node* head = NULL;
    for (int i = 0; i < size; i++) {
        head = append_node(head, arr[i]);
        if (head == NULL) {
            return NULL;
        }
    }
    return head;
}

Node* modulo_division_lists(Node* list1, Node* list2) {
    if (list1 == NULL || list2 == NULL) {
        return NULL;
    }

    Node* result = NULL;
    Node* current1 = list1;
    Node* current2 = list2;

    while (current1 != NULL && current2 != NULL) {
        if (current2->data == 0) {
            free_list(result);
            return NULL;
        }
        int mod_result = current1->data % current2->data;
        result = append_node(result, mod_result);
        if (result == NULL) {
            return NULL;
        }
        current1 = current1->next;
        current2 = current2->next;
    }

    if (current1 != NULL || current2 != NULL) {
        free_list(result);
        return NULL;
    }

    return result;
}

void print_list(Node* head) {
    if (head == NULL) {
        printf("(null)\n");
        return;
    }
    Node* current = head;
    while (current != NULL) {
        printf("%d", current->data);
        if (current->next != NULL) {
            printf(" -> ");
        }
        current = current->next;
    }
    printf("\n");
}

int main(void) {
    int arr1[] = {10, 20, 30, 40};
    int arr2[] = {3, 7, 6, 9};
    int size = sizeof(arr1) / sizeof(arr1[0]);

    Node* list1 = list_from_array(arr1, size);
    Node* list2 = list_from_array(arr2, size);

    if (list1 == NULL || list2 == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        free_list(list1);
        free_list(list2);
        return 1;
    }

    printf("List 1: ");
    print_list(list1);
    printf("List 2: ");
    print_list(list2);

    Node* result = modulo_division_lists(list1, list2);
    printf("Result: ");
    print_list(result);

    free_list(list1);
    free_list(list2);
    free_list(result);

    return 0;
}
