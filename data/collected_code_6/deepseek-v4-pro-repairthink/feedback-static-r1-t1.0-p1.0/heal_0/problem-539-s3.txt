#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    long long value;
    struct Node *next;
} Node;

typedef struct {
    Node *head;
    Node *tail;
    size_t size;
} List;

void list_init(List *list) {
    list->head = NULL;
    list->tail = NULL;
    list->size = 0;
}

int list_append(List *list, long long value) {
    Node *new_node = malloc(sizeof(Node));
    if (new_node == NULL) {
        return -1;
    }
    new_node->value = value;
    new_node->next = NULL;
    
    if (list->tail == NULL) {
        list->head = new_node;
        list->tail = new_node;
    } else {
        list->tail->next = new_node;
        list->tail = new_node;
    }
    list->size++;
    return 0;
}

void list_free(List *list) {
    Node *current = list->head;
    while (current != NULL) {
        Node *next = current->next;
        free(current);
        current = next;
    }
    list->head = NULL;
    list->tail = NULL;
    list->size = 0;
}

long long int_power(int base, int exponent) {
    long long result = 1;
    for (int i = 0; i < exponent; i++) {
        if (result > 9223372036854775807LL / base) {
            return 0;
        }
        result *= base;
    }
    return result;
}

List *list_map(List *input, long long (*func)(int, int)) {
    if (input == NULL || func == NULL) {
        return NULL;
    }
    
    List *result = malloc(sizeof(List));
    if (result == NULL) {
        return NULL;
    }
    list_init(result);
    
    Node *current = input->head;
    int index = 0;
    while (current != NULL) {
        long long mapped_value = func((int)current->value, index);
        if (list_append(result, mapped_value) != 0) {
            list_free(result);
            free(result);
            return NULL;
        }
        current = current->next;
        index++;
    }
    
    return result;
}

int main(void) {
    List input;
    list_init(&input);
    
    int values[] = {2, 3, 5, 7};
    for (size_t i = 0; i < sizeof(values) / sizeof(values[0]); i++) {
        if (list_append(&input, values[i]) != 0) {
            list_free(&input);
            return 1;
        }
    }
    
    List *result = list_map(&input, int_power);
    if (result == NULL) {
        list_free(&input);
        return 1;
    }
    
    printf("Powers list: ");
    Node *current = result->head;
    while (current != NULL) {
        printf("%lld ", current->value);
        current = current->next;
    }
    printf("\n");
    
    list_free(&input);
    list_free(result);
    free(result);
    
    return 0;
}
