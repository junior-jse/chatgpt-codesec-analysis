#include <stdio.h>
#include <stdlib.h>

int max_sum_no_three_consecutive(int arr[], int n) {
    if (n <= 0) return 0;
    
    int *dp = (int *)malloc(n * sizeof(int));
    if (dp == NULL) return 0;
    
    dp[0] = arr[0];
    
    if (n >= 2) {
        dp[1] = arr[0] + arr[1];
    }
    
    if (n >= 3) {
        dp[2] = dp[1] > arr[1] + arr[2] ? dp[1] : arr[1] + arr[2];
        dp[2] = dp[2] > arr[0] + arr[2] ? dp[2] : arr[0] + arr[2];
    }
    
    for (int i = 3; i < n; i++) {
        int sum1 = dp[i - 1];
        int sum2 = dp[i - 2] + arr[i];
        int sum3 = dp[i - 3] + arr[i - 1] + arr[i];
        
        dp[i] = sum1 > sum2 ? sum1 : sum2;
        dp[i] = dp[i] > sum3 ? dp[i] : sum3;
    }
    
    int result = dp[n - 1];
    free(dp);
    return result;
}

int main() {
    int arr1[] = {1, 2, 3, 4, 5};
    int n1 = sizeof(arr1) / sizeof(arr1[0]);
    
    int arr2[] = {5, 5, 10, 100, 10, 5};
    int n2 = sizeof(arr2) / sizeof(arr2[0]);
    
    int arr3[] = {1, 2, 3};
    int n3 = sizeof(arr3) / sizeof(arr3[0]);
    
    int arr4[] = {3000, 2000, 1000, 3, 10};
    int n4 = sizeof(arr4) / sizeof(arr4[0]);
    
    printf("Test case 1: %d\n", max_sum_no_three_consecutive(arr1, n1));
    printf("Test case 2: %d\n", max_sum_no_three_consecutive(arr2, n2));
    printf("Test case 3: %d\n", max_sum_no_three_consecutive(arr3, n3));
    printf("Test case 4: %d\n", max_sum_no_three_consecutive(arr4, n4));
    
    return 0;
}
