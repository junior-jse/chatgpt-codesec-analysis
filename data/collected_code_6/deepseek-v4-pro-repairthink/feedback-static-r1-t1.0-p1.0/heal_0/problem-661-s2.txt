#include <stdio.h>
#include <stdlib.h>

int maxSumWithoutThreeConsec(int arr[], int n) {
    if (n <= 0) return 0;
    if (n == 1) return arr[0];
    if (n == 2) return arr[0] + arr[1];

    int *dp = (int *)malloc(n * sizeof(int));
    if (dp == NULL) {
        return -1;
    }

    dp[0] = arr[0];
    dp[1] = arr[0] + arr[1];
    dp[2] = (arr[0] + arr[1] > arr[1] + arr[2]) ? 
            ((arr[0] + arr[1] > arr[0] + arr[2]) ? arr[0] + arr[1] : arr[0] + arr[2]) : 
            ((arr[1] + arr[2] > arr[0] + arr[2]) ? arr[1] + arr[2] : arr[0] + arr[2]);

    for (int i = 3; i < n; i++) {
        int sum1 = dp[i - 1];
        int sum2 = dp[i - 2] + arr[i];
        int sum3 = dp[i - 3] + arr[i - 1] + arr[i];
        
        int max = sum1;
        if (sum2 > max) max = sum2;
        if (sum3 > max) max = sum3;
        
        dp[i] = max;
    }

    int result = dp[n - 1];
    free(dp);
    return result;
}

int main() {
    int arr1[] = {100, 1000, 100, 1000, 1};
    int n1 = sizeof(arr1) / sizeof(arr1[0]);
    printf("Maximum sum: %d\n", maxSumWithoutThreeConsec(arr1, n1));

    int arr2[] = {1, 2, 3, 4, 5, 6, 7, 8};
    int n2 = sizeof(arr2) / sizeof(arr2[0]);
    printf("Maximum sum: %d\n", maxSumWithoutThreeConsec(arr2, n2));

    int arr3[] = {5, 5, 10, 100, 10, 5};
    int n3 = sizeof(arr3) / sizeof(arr3[0]);
    printf("Maximum sum: %d\n", maxSumWithoutThreeConsec(arr3, n3));

    return 0;
}
