#include <stdio.h>
#include <stdlib.h>

long long *memo;
size_t memo_size;

long long max_sum(long long n) {
    if (n == 0) return 0;
    if (n < 12) return n;

    if (n < (long long)memo_size && memo[n] != -1) return memo[n];

    long long result = max_sum(n/2) + max_sum(n/3) + max_sum(n/4) + max_sum(n/5);
    if (result < n) result = n;

    if (n < (long long)memo_size) memo[n] = result;

    return result;
}

int main(void) {
    long long n;

    while (scanf("%lld", &n) == 1) {
        memo_size = (n < 1000000) ? (size_t)n + 1 : 1000000;
        memo = (long long *)malloc(memo_size * sizeof(long long));
        if (memo == NULL) return 1;

        for (size_t i = 0; i < memo_size; i++) memo[i] = -1;

        printf("%lld\n", max_sum(n));

        free(memo);
        memo = NULL;
    }

    return 0;
}
