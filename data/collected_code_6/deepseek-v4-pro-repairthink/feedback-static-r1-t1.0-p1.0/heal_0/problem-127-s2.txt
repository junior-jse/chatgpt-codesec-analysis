#include <stdio.h>
#include <stdint.h>
#include <limits.h>

int multiply_without_operator(int a, int b) {
    int result = 0;
    int sign = 1;

    if (a < 0) {
        sign = -sign;
        a = -a;
    }
    if (b < 0) {
        sign = -sign;
        b = -b;
    }

    while (b > 0) {
        if (b & 1) {
            result += a;
        }
        a <<= 1;
        b >>= 1;
    }

    return sign > 0 ? result : -result;
}

int main(void) {
    int x = -7;
    int y = 6;
    printf("%d\n", multiply_without_operator(x, y));
    return 0;
}
