#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node* next;
} Node;

typedef struct List {
    Node* head;
    int length;
} List;

typedef struct ListContainer {
    List** lists;
    int count;
} ListContainer;

List* create_list(void) {
    List* list = (List*)malloc(sizeof(List));
    if (!list) return NULL;
    list->head = NULL;
    list->length = 0;
    return list;
}

void list_append(List* list, int value) {
    if (!list) return;
    Node* new_node = (Node*)malloc(sizeof(Node));
    if (!new_node) return;
    new_node->data = value;
    new_node->next = NULL;
    
    if (!list->head) {
        list->head = new_node;
    } else {
        Node* current = list->head;
        while (current->next) {
            current = current->next;
        }
        current->next = new_node;
    }
    list->length++;
}

void free_list(List* list) {
    if (!list) return;
    Node* current = list->head;
    while (current) {
        Node* temp = current;
        current = current->next;
        free(temp);
    }
    free(list);
}

void free_container(ListContainer* container) {
    if (!container) return;
    if (container->lists) {
        for (int i = 0; i < container->count; i++) {
            free_list(container->lists[i]);
        }
        free(container->lists);
    }
    free(container);
}

ListContainer* create_container(int count) {
    ListContainer* container = (ListContainer*)malloc(sizeof(ListContainer));
    if (!container) return NULL;
    container->lists = (List**)malloc(sizeof(List*) * count);
    if (!container->lists) {
        free(container);
        return NULL;
    }
    container->count = count;
    for (int i = 0; i < count; i++) {
        container->lists[i] = NULL;
    }
    return container;
}

List** find_max_length_lists(ListContainer* container, int* result_count) {
    if (!container || !result_count || container->count <= 0) {
        *result_count = 0;
        return NULL;
    }
    
    int max_length = 0;
    for (int i = 0; i < container->count; i++) {
        if (container->lists[i] && container->lists[i]->length > max_length) {
            max_length = container->lists[i]->length;
        }
    }
    
    if (max_length == 0) {
        *result_count = 0;
        return NULL;
    }
    
    *result_count = 0;
    for (int i = 0; i < container->count; i++) {
        if (container->lists[i] && container->lists[i]->length == max_length) {
            (*result_count)++;
        }
    }
    
    List** result = (List**)malloc(sizeof(List*) * (*result_count));
    if (!result) {
        *result_count = 0;
        return NULL;
    }
    
    int index = 0;
    for (int i = 0; i < container->count; i++) {
        if (container->lists[i] && container->lists[i]->length == max_length) {
            result[index++] = container->lists[i];
        }
    }
    
    return result;
}

int main(void) {
    ListContainer* container = create_container(4);
    if (!container) return 1;
    
    container->lists[0] = create_list();
    container->lists[1] = create_list();
    container->lists[2] = create_list();
    container->lists[3] = create_list();
    
    if (!container->lists[0] || !container->lists[1] || 
        !container->lists[2] || !container->lists[3]) {
        free_container(container);
        return 1;
    }
    
    list_append(container->lists[0], 1);
    list_append(container->lists[0], 2);
    
    list_append(container->lists[1], 3);
    
    list_append(container->lists[2], 4);
    list_append(container->lists[2], 5);
    list_append(container->lists[2], 6);
    
    list_append(container->lists[3], 7);
    list_append(container->lists[3], 8);
    list_append(container->lists[3], 9);
    
    int result_count = 0;
    List** max_lists = find_max_length_lists(container, &result_count);
    
    if (max_lists) {
        printf("Number of lists with maximum length: %d\n", result_count);
        free(max_lists);
    }
    
    free_container(container);
    return 0;
}
